 243     entries = (enc_entries, dyn_entries, dec_entries)
 244     outs = {'tokens': tokens, 'repfeat': repfeat, 'losses': losses}

 312   def _apply_replay_context(self, carry, data):
 313     (enc_carry, dyn_carry, dec_carry, prevact) = carry
 314     carry = (enc_carry, dyn_carry, dec_carry)
 315     stepid = data['stepid']
 316     obs = {k: data[k] for k in self.obs_space}
 317     prepend = lambda x, y: jnp.concatenate([x[:, None], y[:, :-1]], 1)
 318     prevact = {k: prepend(prevact[k], data[k]) for k in self.act_space}
 319     if not self.config.replay_context:
 320       return carry, obs, prevact, stepid
 321 
 322     K = self.config.replay_context
 323     nested = elements.tree.nestdict(data)
 324     entries = [nested.get(k, {}) for k in ('enc', 'dyn', 'dec')]
 325     lhs = lambda xs: jax.tree.map(lambda x: x[:, :K], xs)
 326     rhs = lambda xs: jax.tree.map(lambda x: x[:, K:], xs)
 327     rep_carry = (
 328         self.enc.truncate(lhs(entries[0]), enc_carry),
 329         self.dyn.truncate(lhs(entries[1]), dyn_carry),
 330         self.dec.truncate(lhs(entries[2]), dec_carry))
 331     rep_obs = {k: rhs(data[k]) for k in self.obs_space}
 332     rep_prevact = {k: data[k][:, K - 1: -1] for k in self.act_space}
 333     rep_stepid = rhs(stepid)
 334 
 335     first_chunk = (data['consec'][:, 0] == 0)
 336     carry, obs, prevact, stepid = jax.tree.map(
 337         lambda normal, replay: nn.where(first_chunk, replay, normal),
 338         (carry, rhs(obs), rhs(prevact), rhs(stepid)),
 339         (rep_carry, rep_obs, rep_prevact, rep_stepid))
 340     return carry, obs, prevact, stepid

 342   def _make_opt(
 343       self,
 344       lr: float = 4e-5,
 345       agc: float = 0.3,
 346       eps: float = 1e-20,
 347       beta1: float = 0.9,
 348       beta2: float = 0.999,
 349       momentum: bool = True,
 350       nesterov: bool = False,
 351       wd: float = 0.0,
 352       wdregex: str = r'/kernel$',
 353       schedule: str = 'const',
 354       warmup: int = 1000,
 355       anneal: int = 0,
 356   ):
 357     chain = []
 358     chain.append(embodied.jax.opt.clip_by_agc(agc))
 359     chain.append(embodied.jax.opt.scale_by_rms(beta2, eps))
 360     chain.append(embodied.jax.opt.scale_by_momentum(beta1, nesterov))
 361     if wd:
 362       assert not wdregex[0].isnumeric(), wdregex
 363       pattern = re.compile(wdregex)
 364       wdmask = lambda params: {k: bool(pattern.search(k)) for k in params}
 365       chain.append(optax.add_decayed_weights(wd, wdmask))
 366     assert anneal > 0 or schedule == 'const'
 367     if schedule == 'const':
 368       sched = optax.constant_schedule(lr)
 369     elif schedule == 'linear':
 370       sched = optax.linear_schedule(lr, 0.1 * lr, anneal - warmup)
 371     elif schedule == 'cosine':
 372       sched = optax.cosine_decay_schedule(lr, anneal - warmup, 0.1 * lr)
 373     else:
 374       raise NotImplementedError(schedule)
 375     if warmup:
 376       ramp = optax.linear_schedule(0.0, lr, warmup)
 377       sched = optax.join_schedules([ramp, sched], [warmup])
 378     chain.append(optax.scale_by_learning_rate(sched))
 379     return optax.chain(*chain)

 382 def imag_loss(
 383     act, rew, con,
 384     policy, value, slowvalue,
 385     retnorm, valnorm, advnorm,
 386     update,
 387     contdisc=True,
 388     slowtar=True,
 389     horizon=333,
 390     lam=0.95,
 391     actent=3e-4,
 392     slowreg=1.0,
 393 ):
 394   losses = {}
 395   metrics = {}
 396 
 397   voffset, vscale = valnorm.stats()
 398   val = value.pred() * vscale + voffset
 399   slowval = slowvalue.pred() * vscale + voffset
 400   tarval = slowval if slowtar else val
 401   disc = 1 if contdisc else 1 - 1 / horizon
 402   weight = jnp.cumprod(disc * con, 1) / disc
 403   last = jnp.zeros_like(con)
 404   term = 1 - con
 405   ret = lambda_return(last, term, rew, tarval, tarval, disc, lam)
 406 
 407   roffset, rscale = retnorm(ret, update)
 408   adv = (ret - tarval[:, :-1]) / rscale
 409   aoffset, ascale = advnorm(adv, update)
 410   adv_normed = (adv - aoffset) / ascale
 411   logpi = sum([v.logp(sg(act[k]))[:, :-1] for k, v in policy.items()])
 412   ents = {k: v.entropy()[:, :-1] for k, v in policy.items()}
 413   policy_loss = sg(weight[:, :-1]) * -(
 414       logpi * sg(adv_normed) + actent * sum(ents.values()))
 415   losses['policy'] = policy_loss
 416 
 417   voffset, vscale = valnorm(ret, update)
 418   tar_normed = (ret - voffset) / vscale
 419   tar_padded = jnp.concatenate([tar_normed, 0 * tar_normed[:, -1:]], 1)
 420   losses['value'] = sg(weight[:, :-1]) * (
 421       value.loss(sg(tar_padded)) +
 422       slowreg * value.loss(sg(slowvalue.pred())))[:, :-1]
 423 
 424   ret_normed = (ret - roffset) / rscale
 425   metrics['adv'] = adv.mean()
 426   metrics['adv_std'] = adv.std()
 427   metrics['adv_mag'] = jnp.abs(adv).mean()
 428   metrics['rew'] = rew.mean()
 429   metrics['con'] = con.mean()
 430   metrics['ret'] = ret_normed.mean()
 431   metrics['val'] = val.mean()
 432   metrics['tar'] = tar_normed.mean()
 433   metrics['weight'] = weight.mean()
 434   metrics['slowval'] = slowval.mean()
 435   metrics['ret_min'] = ret_normed.min()
 436   metrics['ret_max'] = ret_normed.max()
 437   metrics['ret_rate'] = (jnp.abs(ret_normed) >= 1.0).mean()
 438   for k in act:
 439     metrics[f'ent/{k}'] = ents[k].mean()
 440     if hasattr(policy[k], 'minent'):
 441       lo, hi = policy[k].minent, policy[k].maxent
 442       metrics[f'rand/{k}'] = (ents[k].mean() - lo) / (hi - lo)
 443 
 444   outs = {}
 445   outs['ret'] = ret
 446   return losses, outs, metrics

 449 def repl_loss(
 450     last, term, rew, boot,
 451     value, slowvalue, valnorm,
 452     update=True,
 453     slowreg=1.0,
 454     slowtar=True,
 455     horizon=333,
 456     lam=0.95,
 457 ):
 458   losses = {}
 459 
 460   voffset, vscale = valnorm.stats()
 461   val = value.pred() * vscale + voffset
 462   slowval = slowvalue.pred() * vscale + voffset
 463   tarval = slowval if slowtar else val
 464   disc = 1 - 1 / horizon
 465   weight = f32(~last)
 466   ret = lambda_return(last, term, rew, tarval, boot, disc, lam)
 467 
 468   voffset, vscale = valnorm(ret, update)
 469   ret_normed = (ret - voffset) / vscale
 470   ret_padded = jnp.concatenate([ret_normed, 0 * ret_normed[:, -1:]], 1)
 471   losses['repval'] = weight[:, :-1] * (
 472       value.loss(sg(ret_padded)) +
 473       slowreg * value.loss(sg(slowvalue.pred())))[:, :-1]
 474 
 475   outs = {}
 476   outs['ret'] = ret
 477   metrics = {}
 478 
 479   return losses, outs, metrics

 482 def lambda_return(last, term, rew, val, boot, disc, lam):
 483   chex.assert_equal_shape((last, term, rew, val, boot))
 484   rets = [boot[:, -1]]
 485   live = (1 - f32(term))[:, 1:] * disc
 486   cont = (1 - f32(last))[:, 1:] * lam
 487   interm = rew[:, 1:] + (1 - cont) * live * boot[:, 1:]
 488   for t in reversed(range(live.shape[1])):
 489     rets.append(interm[:, t] + live[:, t] * cont[:, t] * rets[-1])
 490   return jnp.stack(list(reversed(rets))[:-1], 1)
