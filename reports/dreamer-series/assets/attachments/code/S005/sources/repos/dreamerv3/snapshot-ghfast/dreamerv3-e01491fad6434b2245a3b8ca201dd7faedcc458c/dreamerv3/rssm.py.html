  61   def observe(self, carry, tokens, action, reset, training, single=False):
  62     carry, tokens, action = nn.cast((carry, tokens, action))
  63     if single:
  64       carry, (entry, feat) = self._observe(
  65           carry, tokens, action, reset, training)
  66       return carry, entry, feat
  67     else:
  68       unroll = jax.tree.leaves(tokens)[0].shape[1] if self.unroll else 1
  69       carry, (entries, feat) = nj.scan(
  70           lambda carry, inputs: self._observe(
  71               carry, *inputs, training),
  72           carry, (tokens, action, reset), unroll=unroll, axis=1)
  73       return carry, entries, feat

  75   def _observe(self, carry, tokens, action, reset, training):
  76     deter, stoch, action = nn.mask(
  77         (carry['deter'], carry['stoch'], action), ~reset)
  78     action = nn.DictConcat(self.act_space, 1)(action)
  79     action = nn.mask(action, ~reset)
  80     deter = self._core(deter, stoch, action)
  81     tokens = tokens.reshape((*deter.shape[:-1], -1))
  82     x = tokens if self.absolute else jnp.concatenate([deter, tokens], -1)
  83     for i in range(self.obslayers):
  84       x = self.sub(f'obs{i}', nn.Linear, self.hidden, **self.kw)(x)
  85       x = nn.act(self.act)(self.sub(f'obs{i}norm', nn.Norm, self.norm)(x))
  86     logit = self._logit('obslogit', x)
  87     stoch = nn.cast(self._dist(logit).sample(seed=nj.seed()))
  88     carry = dict(deter=deter, stoch=stoch)
  89     feat = dict(deter=deter, stoch=stoch, logit=logit)
  90     entry = dict(deter=deter, stoch=stoch)
  91     assert all(x.dtype == nn.COMPUTE_DTYPE for x in (deter, stoch, logit))
  92     return carry, (entry, feat)

  94   def imagine(self, carry, policy, length, training, single=False):
  95     if single:
  96       action = policy(sg(carry)) if callable(policy) else policy
  97       actemb = nn.DictConcat(self.act_space, 1)(action)
  98       deter = self._core(carry['deter'], carry['stoch'], actemb)
  99       logit = self._prior(deter)
 100       stoch = nn.cast(self._dist(logit).sample(seed=nj.seed()))
 101       carry = nn.cast(dict(deter=deter, stoch=stoch))
 102       feat = nn.cast(dict(deter=deter, stoch=stoch, logit=logit))
 103       assert all(x.dtype == nn.COMPUTE_DTYPE for x in (deter, stoch, logit))
 104       return carry, (feat, action)
 105     else:
 106       unroll = length if self.unroll else 1
 107       if callable(policy):
 108         carry, (feat, action) = nj.scan(
 109             lambda c, _: self.imagine(c, policy, 1, training, single=True),
 110             nn.cast(carry), (), length, unroll=unroll, axis=1)
 111       else:
 112         carry, (feat, action) = nj.scan(
 113             lambda c, a: self.imagine(c, a, 1, training, single=True),
 114             nn.cast(carry), nn.cast(policy), length, unroll=unroll, axis=1)
 115       # We can also return all carry entries but it might be expensive.
 116       # entries = dict(deter=feat['deter'], stoch=feat['stoch'])
 117       # return carry, entries, feat, action
 118       return carry, feat, action

 120   def loss(self, carry, tokens, acts, reset, training):
 121     metrics = {}
 122     carry, entries, feat = self.observe(carry, tokens, acts, reset, training)
 123     prior = self._prior(feat['deter'])
 124     post = feat['logit']
 125     dyn = self._dist(sg(post)).kl(self._dist(prior))
 126     rep = self._dist(post).kl(self._dist(sg(prior)))
 127     if self.free_nats:
 128       dyn = jnp.maximum(dyn, self.free_nats)
 129       rep = jnp.maximum(rep, self.free_nats)
 130     losses = {'dyn': dyn, 'rep': rep}
 131     metrics['dyn_ent'] = self._dist(prior).entropy().mean()
 132     metrics['rep_ent'] = self._dist(post).entropy().mean()
 133     return carry, entries, losses, feat, metrics

 135   def _core(self, deter, stoch, action):
 136     stoch = stoch.reshape((stoch.shape[0], -1))
 137     action /= sg(jnp.maximum(1, jnp.abs(action)))
 138     g = self.blocks
 139     flat2group = lambda x: einops.rearrange(x, '... (g h) -> ... g h', g=g)
 140     group2flat = lambda x: einops.rearrange(x, '... g h -> ... (g h)', g=g)
 141     x0 = self.sub('dynin0', nn.Linear, self.hidden, **self.kw)(deter)
 142     x0 = nn.act(self.act)(self.sub('dynin0norm', nn.Norm, self.norm)(x0))
 143     x1 = self.sub('dynin1', nn.Linear, self.hidden, **self.kw)(stoch)
 144     x1 = nn.act(self.act)(self.sub('dynin1norm', nn.Norm, self.norm)(x1))
 145     x2 = self.sub('dynin2', nn.Linear, self.hidden, **self.kw)(action)
 146     x2 = nn.act(self.act)(self.sub('dynin2norm', nn.Norm, self.norm)(x2))
 147     x = jnp.concatenate([x0, x1, x2], -1)[..., None, :].repeat(g, -2)
 148     x = group2flat(jnp.concatenate([flat2group(deter), x], -1))
 149     for i in range(self.dynlayers):
 150       x = self.sub(f'dynhid{i}', nn.BlockLinear, self.deter, g, **self.kw)(x)
 151       x = nn.act(self.act)(self.sub(f'dynhid{i}norm', nn.Norm, self.norm)(x))
 152     x = self.sub('dyngru', nn.BlockLinear, 3 * self.deter, g, **self.kw)(x)
 153     gates = jnp.split(flat2group(x), 3, -1)
 154     reset, cand, update = [group2flat(x) for x in gates]
 155     reset = jax.nn.sigmoid(reset)
 156     cand = jnp.tanh(reset * cand)
 157     update = jax.nn.sigmoid(update - 1)
 158     deter = update * cand + (1 - update) * deter
 159     return deter

 161   def _prior(self, feat):
 162     x = feat
 163     for i in range(self.imglayers):
 164       x = self.sub(f'prior{i}', nn.Linear, self.hidden, **self.kw)(x)
 165       x = nn.act(self.act)(self.sub(f'prior{i}norm', nn.Norm, self.norm)(x))
 166     return self._logit('priorlogit', x)

 168   def _logit(self, name, x):
 169     kw = dict(**self.kw, outscale=self.outscale)
 170     x = self.sub(name, nn.Linear, self.stoch * self.classes, **kw)(x)
 171     return x.reshape(x.shape[:-1] + (self.stoch, self.classes))

 173   def _dist(self, logits):
 174     out = embodied.jax.outs.OneHot(logits, self.unimix)
 175     out = embodied.jax.outs.Agg(out, 1, jnp.sum)
 176     return out

 210   def __call__(self, carry, obs, reset, training, single=False):
 211     bdims = 1 if single else 2
 212     outs = []
 213     bshape = reset.shape
 214 
 215     if self.veckeys:
 216       vspace = {k: self.obs_space[k] for k in self.veckeys}
 217       vecs = {k: obs[k] for k in self.veckeys}
 218       squish = nn.symlog if self.symlog else lambda x: x
 219       x = nn.DictConcat(vspace, 1, squish=squish)(vecs)
 220       x = x.reshape((-1, *x.shape[bdims:]))
 221       for i in range(self.layers):
 222         x = self.sub(f'mlp{i}', nn.Linear, self.units, **self.kw)(x)
 223         x = nn.act(self.act)(self.sub(f'mlp{i}norm', nn.Norm, self.norm)(x))
 224       outs.append(x)
 225 
 226     if self.imgkeys:
 227       K = self.kernel
 228       imgs = [obs[k] for k in sorted(self.imgkeys)]
 229       assert all(x.dtype == jnp.uint8 for x in imgs)
 230       x = nn.cast(jnp.concatenate(imgs, -1), force=True) / 255 - 0.5
 231       x = x.reshape((-1, *x.shape[bdims:]))
 232       for i, depth in enumerate(self.depths):
 233         if self.outer and i == 0:
 234           x = self.sub(f'cnn{i}', nn.Conv2D, depth, K, **self.kw)(x)
 235         elif self.strided:
 236           x = self.sub(f'cnn{i}', nn.Conv2D, depth, K, 2, **self.kw)(x)
 237         else:
 238           x = self.sub(f'cnn{i}', nn.Conv2D, depth, K, **self.kw)(x)
 239           B, H, W, C = x.shape
 240           x = x.reshape((B, H // 2, 2, W // 2, 2, C)).max((2, 4))
 241         x = nn.act(self.act)(self.sub(f'cnn{i}norm', nn.Norm, self.norm)(x))
 242       assert 3 <= x.shape[-3] <= 16, x.shape
 243       assert 3 <= x.shape[-2] <= 16, x.shape
 244       x = x.reshape((x.shape[0], -1))
 245       outs.append(x)
 246 
 247     x = jnp.concatenate(outs, -1)
 248     tokens = x.reshape((*bshape, *x.shape[1:]))
 249     entries = {}
 250     return carry, entries, tokens

 253 class Decoder(nj.Module):
 254 
 255   units: int = 1024
 256   norm: str = 'rms'
 257   act: str = 'gelu'
 258   outscale: float = 1.0
 259   depth: int = 64
 260   mults: tuple = (2, 3, 4, 4)
 261   layers: int = 3
 262   kernel: int = 5
 263   symlog: bool = True
 264   bspace: int = 8
 265   outer: bool = False
 266   strided: bool = False

 268   def __init__(self, obs_space, **kw):
 269     assert all(len(s.shape) <= 3 for s in obs_space.values()), obs_space
 270     self.obs_space = obs_space
 271     self.veckeys = [k for k, s in obs_space.items() if len(s.shape) <= 2]
 272     self.imgkeys = [k for k, s in obs_space.items() if len(s.shape) == 3]
 273     self.depths = tuple(self.depth * mult for mult in self.mults)
 274     self.imgdep = sum(obs_space[k].shape[-1] for k in self.imgkeys)
 275     self.imgres = self.imgkeys and obs_space[self.imgkeys[0]].shape[:-1]
 276     self.kw = kw
 277 
 278   @property
 279   def entry_space(self):
 280     return {}
 281 
 282   def initial(self, batch_size):
 283     return {}
 284 
 285   def truncate(self, entries, carry=None):
 286     return {}

 288   def __call__(self, carry, feat, reset, training, single=False):
 289     assert feat['deter'].shape[-1] % self.bspace == 0
 290     K = self.kernel
 291     recons = {}
 292     bshape = reset.shape
 293     inp = [nn.cast(feat[k]) for k in ('stoch', 'deter')]
 294     inp = [x.reshape((math.prod(bshape), -1)) for x in inp]
 295     inp = jnp.concatenate(inp, -1)
 296 
 297     if self.veckeys:
 298       spaces = {k: self.obs_space[k] for k in self.veckeys}
 299       o1, o2 = 'categorical', ('symlog_mse' if self.symlog else 'mse')
 300       outputs = {k: o1 if v.discrete else o2 for k, v in spaces.items()}
 301       kw = dict(**self.kw, act=self.act, norm=self.norm)
 302       x = self.sub('mlp', nn.MLP, self.layers, self.units, **kw)(inp)
 303       x = x.reshape((*bshape, *x.shape[1:]))
 304       kw = dict(**self.kw, outscale=self.outscale)
 305       outs = self.sub('vec', embodied.jax.DictHead, spaces, outputs, **kw)(x)
 306       recons.update(outs)
 307 
 308     if self.imgkeys:
 309       factor = 2 ** (len(self.depths) - int(bool(self.outer)))
 310       minres = [int(x // factor) for x in self.imgres]
 311       assert 3 <= minres[0] <= 16, minres
 312       assert 3 <= minres[1] <= 16, minres
 313       shape = (*minres, self.depths[-1])
 314       if self.bspace:
 315         u, g = math.prod(shape), self.bspace
 316         x0, x1 = nn.cast((feat['deter'], feat['stoch']))
 317         x1 = x1.reshape((*x1.shape[:-2], -1))
 318         x0 = x0.reshape((-1, x0.shape[-1]))
 319         x1 = x1.reshape((-1, x1.shape[-1]))
 320         x0 = self.sub('sp0', nn.BlockLinear, u, g, **self.kw)(x0)
 321         x0 = einops.rearrange(
 322             x0, '... (g h w c) -> ... h w (g c)',
 323             h=minres[0], w=minres[1], g=g)
 324         x1 = self.sub('sp1', nn.Linear, 2 * self.units, **self.kw)(x1)
 325         x1 = nn.act(self.act)(self.sub('sp1norm', nn.Norm, self.norm)(x1))
 326         x1 = self.sub('sp2', nn.Linear, shape, **self.kw)(x1)
 327         x = nn.act(self.act)(self.sub('spnorm', nn.Norm, self.norm)(x0 + x1))
 328       else:
 329         x = self.sub('space', nn.Linear, shape, **kw)(inp)
 330         x = nn.act(self.act)(self.sub('spacenorm', nn.Norm, self.norm)(x))
 331       for i, depth in reversed(list(enumerate(self.depths[:-1]))):
 332         if self.strided:
 333           kw = dict(**self.kw, transp=True)
 334           x = self.sub(f'conv{i}', nn.Conv2D, depth, K, 2, **kw)(x)
 335         else:
 336           x = x.repeat(2, -2).repeat(2, -3)
 337           x = self.sub(f'conv{i}', nn.Conv2D, depth, K, **self.kw)(x)
 338         x = nn.act(self.act)(self.sub(f'conv{i}norm', nn.Norm, self.norm)(x))
 339       if self.outer:
 340         kw = dict(**self.kw, outscale=self.outscale)
 341         x = self.sub('imgout', nn.Conv2D, self.imgdep, K, **kw)(x)
 342       elif self.strided:
 343         kw = dict(**self.kw, outscale=self.outscale, transp=True)
 344         x = self.sub('imgout', nn.Conv2D, self.imgdep, K, 2, **kw)(x)
 345       else:
 346         x = x.repeat(2, -2).repeat(2, -3)
 347         kw = dict(**self.kw, outscale=self.outscale)
 348         x = self.sub('imgout', nn.Conv2D, self.imgdep, K, **kw)(x)
 349       x = jax.nn.sigmoid(x)
 350       x = x.reshape((*bshape, *x.shape[1:]))
 351       split = np.cumsum(
 352           [self.obs_space[k].shape[-1] for k in self.imgkeys][:-1])
 353       for k, out in zip(self.imgkeys, jnp.split(x, split, -1)):
 354         out = embodied.jax.outs.MSE(out)
 355         out = embodied.jax.outs.Agg(out, 3, jnp.sum)
 356         recons[k] = out
 357 
 358     entries = {}
 359     return carry, entries, recons
