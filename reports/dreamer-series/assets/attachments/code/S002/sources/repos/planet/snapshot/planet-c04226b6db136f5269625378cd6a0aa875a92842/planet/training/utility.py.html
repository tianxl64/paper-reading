  44 def save_config(config, logdir=None):
  45   """Save a new configuration by name.
  46 
  47   If a logging directory is specified, is will be created and the configuration
  48   will be stored there. Otherwise, a log message will be printed.
  49 
  50   Args:
  51     config: Configuration object.
  52     logdir: Location for writing summaries and checkpoints if specified.
  53 
  54   Returns:
  55     Configuration object.
  56   """
  57   if logdir:
  58     with config.unlocked:
  59       config.logdir = logdir
  60     message = 'Start a new run and write summaries and checkpoints to {}.'
  61     tf.logging.info(message.format(config.logdir))
  62     tf.gfile.MakeDirs(config.logdir)
  63     config_path = os.path.join(config.logdir, 'config.yaml')
  64     with tf.gfile.GFile(config_path, 'w') as file_:
  65       yaml.dump(
  66           config, file_, yaml.Dumper,
  67           allow_unicode=True,
  68           default_flow_style=False)
  69   else:
  70     message = (
  71         'Start a new run without storing summaries and checkpoints since no '
  72         'logging directory was specified.')
  73     tf.logging.info(message)
  74   return config

  77 def load_config(logdir):
  78   """Load a configuration from the log directory.
  79 
  80   Args:
  81     logdir: The logging directory containing the configuration file.
  82 
  83   Raises:
  84     IOError: The logging directory does not contain a configuration file.
  85 
  86   Returns:
  87     Configuration object.
  88   """
  89   config_path = logdir and os.path.join(logdir, 'config.yaml')
  90   if not config_path or not tf.gfile.Exists(config_path):
  91     message = (
  92         'Cannot resume an existing run since the logging directory does not '
  93         'contain a configuration file.')
  94     raise IOError(message)
  95   with tf.gfile.GFile(config_path, 'r') as file_:
  96     config = yaml.load(file_, yaml.Loader)
  97     message = 'Resume run and write summaries and checkpoints to {}.'
  98     tf.logging.info(message.format(config.logdir))
  99   return config

 102 def get_batch(datasets, phase, reset):
 103   """Read batches from multiple datasets based on the training phase.
 104 
 105   The test dataset is reset at the beginning of every test phase. The training
 106   dataset is repeated infinitely and doesn't need a reset.
 107 
 108   Args:
 109     datasets: Dictionary of datasets with training phases as keys.
 110     phase: Tensor of the training phase name.
 111     reset: Whether to reset the datasets.
 112 
 113   Returns:
 114     data: a batch of data from either the train or test set.
 115   """
 116   with datasets.unlocked:
 117     datasets.train = datasets.train.make_one_shot_iterator()
 118     datasets.test = datasets.test.make_one_shot_iterator()
 119   data = tf.cond(
 120       tf.equal(phase, 'train'),
 121       datasets.train.get_next,
 122       datasets.test.get_next)
 123   if not isinstance(data, dict):
 124     data = {'data': data}
 125   if 'length' not in data:
 126     example = data[list(data.keys())[0]]
 127     data['length'] = (
 128         tf.zeros((tf.shape(example)[0],), tf.int32) + tf.shape(example)[1])
 129   return data

 132 def train(model_fn, datasets, logdir, config):
 133   """Train a model on a datasets.
 134 
 135   The model function receives the following arguments: data batch, trainer
 136   phase, whether it should log, and the config. The configuration object should
 137   contain the attributes `batch_shape`, `train_steps`, `test_steps`,
 138   `max_steps`, in addition to the attributes expected by the model function.
 139 
 140   Args:
 141     model_fn: Function greating the model graph.
 142     datasets: Dictionary with keys `train` and `test` and datasets as values.
 143     logdir: Optional logging directory for summaries and checkpoints.
 144     config: Configuration object.
 145 
 146   Yields:
 147     Test score of every epoch.
 148 
 149   Raises:
 150     KeyError: if config is falsey.
 151   """
 152   if not config:
 153     raise KeyError('You must specify a configuration.')
 154   logdir = logdir and os.path.expanduser(logdir)
 155   try:
 156     config = load_config(logdir)
 157   except RuntimeError:
 158     print('Failed to load existing config.')
 159   except IOError:
 160     config = save_config(config, logdir)
 161   trainer = trainer_.Trainer(logdir, config=config)
 162   cleanups = []
 163   try:
 164     with tf.variable_scope('graph', use_resource=True):
 165       data = get_batch(datasets, trainer.phase, trainer.reset)
 166       score, summary, cleanups = model_fn(data, trainer, config)
 167       message = 'Graph contains {} trainable variables.'
 168       tf.logging.info(message.format(tools.count_weights()))
 169       if config.train_steps:
 170         trainer.add_phase(
 171             'train', config.train_steps, score, summary,
 172             batch_size=config.batch_shape[0],
 173             report_every=None,
 174             log_every=config.train_log_every,
 175             checkpoint_every=config.train_checkpoint_every)
 176       if config.test_steps:
 177         trainer.add_phase(
 178             'test', config.test_steps, score, summary,
 179             batch_size=config.batch_shape[0],
 180             report_every=config.test_steps,
 181             log_every=config.test_steps,
 182             checkpoint_every=config.test_checkpoint_every)
 183     for saver in config.savers:
 184       trainer.add_saver(**saver)
 185     for score in trainer.iterate(config.max_steps):
 186       yield score
 187   finally:
 188     for cleanup in cleanups:
 189       cleanup()

 192 def compute_objectives(posterior, prior, target, graph, config):
 193   raw_features = graph.cell.features_from_state(posterior)
 194   heads = graph.heads
 195   objectives = []
 196   for name, scale in config.loss_scales.items():
 197     if config.loss_scales[name] == 0.0:
 198       continue
 199     if name in config.heads and name not in config.gradient_heads:
 200       features = tf.stop_gradient(raw_features)
 201       include = r'.*/head_{}/.*'.format(name)
 202       exclude = None
 203     else:
 204       features = raw_features
 205       include = r'.*'
 206       exclude = None
 207 
 208     if name == 'divergence':
 209       loss = graph.cell.divergence_from_states(posterior, prior)
 210       if config.free_nats is not None:
 211         loss = tf.maximum(0.0, loss - float(config.free_nats))
 212       objectives.append(Objective('divergence', loss, min, include, exclude))
 213 
 214     elif name == 'overshooting':
 215       shape = tools.shape(graph.data['action'])
 216       length = tf.tile(tf.constant(shape[1])[None], [shape[0]])
 217       _, priors, posteriors, mask = tools.overshooting(
 218           graph.cell, {}, graph.embedded, graph.data['action'], length,
 219           config.overshooting_distance, posterior)
 220       posteriors, priors, mask = tools.nested.map(
 221           lambda x: x[:, :, 1:-1], (posteriors, priors, mask))
 222       if config.os_stop_posterior_grad:
 223         posteriors = tools.nested.map(tf.stop_gradient, posteriors)
 224       loss = graph.cell.divergence_from_states(posteriors, priors)
 225       if config.free_nats is not None:
 226         loss = tf.maximum(0.0, loss - float(config.free_nats))
 227       objectives.append(Objective('overshooting', loss, min, include, exclude))
 228 
 229     else:
 230       logprob = heads[name](features).log_prob(target[name])
 231       objectives.append(Objective(name, logprob, max, include, exclude))
 232 
 233   objectives = [o._replace(value=tf.reduce_mean(o.value)) for o in objectives]
 234   return objectives

 237 def apply_optimizers(objectives, trainer, config):
 238   # Make sure all losses are computed and apply loss scales.
 239   processed = []
 240   values = [ob.value for ob in objectives]
 241   for ob in objectives:
 242     loss = {min: ob.value, max: -ob.value}[ob.goal]
 243     loss *= config.loss_scales[ob.name]
 244     with tf.control_dependencies(values):
 245       loss = tf.identity(loss)
 246     processed.append(ob._replace(value=loss, goal=min))
 247   # Merge objectives that operate on the whole model to compute only one
 248   # backward pass and to share optimizer statistics.
 249   objectives = []
 250   losses = []
 251   for ob in processed:
 252     if ob.include == r'.*' and ob.exclude is None:
 253       assert ob.goal == min
 254       losses.append(ob.value)
 255     else:
 256       objectives.append(ob)
 257   objectives.append(Objective('main', tf.reduce_sum(losses), min, r'.*', None))
 258   # Apply optimizers and collect loss summaries.
 259   summaries = []
 260   grad_norms = {}
 261   for ob in objectives:
 262     assert ob.name in list(config.loss_scales.keys()) + ['main'], ob
 263     assert ob.goal == min, ob
 264     assert ob.name in config.optimizers, ob
 265     optimizer = config.optimizers[ob.name](
 266         include=ob.include,
 267         exclude=ob.exclude,
 268         step=trainer.step,
 269         log=trainer.log,
 270         debug=config.debug,
 271         name=ob.name)
 272     condition = tf.equal(trainer.phase, 'train')
 273     summary, grad_norm = optimizer.maybe_minimize(condition, ob.value)
 274     summaries.append(summary)
 275     grad_norms[ob.name] = grad_norm
 276   return summaries, grad_norms

 279 def simulate_episodes(
 280     config, params, graph, cleanups, expensive_summaries, gif_summary, name):
 281   def env_ctor():
 282     env = params.task.env_ctor()
 283     if params.save_episode_dir:
 284       env = control.wrappers.CollectGymDataset(env, params.save_episode_dir)
 285     env = control.wrappers.ConcatObservation(env, ['image'])
 286     return env
 287   bind_or_none = lambda x, **kw: x and functools.partial(x, **kw)
 288   cell = graph.cell
 289   agent_config = tools.AttrDict(
 290       cell=cell,
 291       encoder=graph.encoder,
 292       planner=functools.partial(params.planner, graph=graph),
 293       objective=bind_or_none(params.objective, graph=graph),
 294       exploration=params.exploration,
 295       preprocess_fn=config.preprocess_fn,
 296       postprocess_fn=config.postprocess_fn)
 297   params = params.copy()
 298   with params.unlocked:
 299     params.update(agent_config)
 300   with agent_config.unlocked:
 301     agent_config.update(params)
 302   summary, return_, cleanup = control.simulate(
 303       graph.step, env_ctor, params.task.max_length,
 304       params.num_agents, agent_config, config.isolate_envs,
 305       expensive_summaries, gif_summary, name=name)
 306   cleanups.append(cleanup)  # Work around tf.cond() tensor return type.
 307   return summary, return_

 310 def print_metrics(metrics, step, every, name='metrics'):
 311   means, updates = [], []
 312   for key, value in metrics.items():
 313     key = 'metrics_{}_{}'.format(name, key)
 314     mean = tools.StreamingMean((), tf.float32, key)
 315     means.append(mean)
 316     updates.append(mean.submit(value))
 317   with tf.control_dependencies(updates):
 318     # message = 'step/' + '/'.join(metrics.keys()) + ' = '
 319     message = '{}: step/{} ='.format(name, '/'.join(metrics.keys()))
 320     gs = tf.train.get_or_create_global_step()
 321     print_metrics = tf.cond(
 322         tf.equal(step % every, 0),
 323         lambda: tf.print(message, [gs] + [mean.clear() for mean in means]),
 324         tf.no_op)
 325   return print_metrics

 328 def collect_initial_episodes(config):
 329   items = config.random_collects.items()
 330   items = sorted(items, key=lambda x: x[0])
 331   existing = {}
 332   for name, params in items:
 333     outdir = params.save_episode_dir
 334     tf.gfile.MakeDirs(outdir)
 335     if outdir not in existing:
 336       existing[outdir] = len(tf.gfile.Glob(os.path.join(outdir, '*.npz')))
 337     if params.num_episodes <= existing[outdir]:
 338       existing[outdir] -= params.num_episodes
 339     else:
 340       remaining = params.num_episodes - existing[outdir]
 341       existing[outdir] = 0
 342       env_ctor = params.task.env_ctor
 343       print('Collecting {} initial episodes ({}).'.format(remaining, name))
 344       control.random_episodes(env_ctor, remaining, outdir)
