  38 def LOG(x, m, *a):
  39   xf = tf.cast(x, tf.float32)
  40   print_op = tf.print(
  41       m, '(',
  42       tf.shape(xf),
  43       tf.reduce_min(xf),
  44       tf.reduce_mean(xf),
  45       tf.reduce_max(xf),
  46       ')', *a)
  47   with tf.control_dependencies([print_op]):
  48     return tf.identity(x)

  51 def save_config(config, logdir=None):
  52   if logdir:
  53     with config.unlocked:
  54       config.logdir = logdir
  55     message = 'Start a new run and write summaries and checkpoints to {}.'
  56     print(message.format(config.logdir))
  57     tf.gfile.MakeDirs(config.logdir)
  58     config_path = os.path.join(config.logdir, 'config.yaml')
  59     with tf.gfile.GFile(config_path, 'w') as file_:
  60       yaml.dump(
  61           config, file_, yaml.Dumper,
  62           allow_unicode=True,
  63           default_flow_style=False)
  64   else:
  65     message = (
  66         'Start a new run without storing summaries and checkpoints since no '
  67         'logging directory was specified.')
  68     print(message)
  69   return config

  72 def load_config(logdir):
  73   config_path = logdir and os.path.join(logdir, 'config.yaml')
  74   if not config_path or not tf.gfile.Exists(config_path):
  75     message = (
  76         'Cannot resume an existing run since the logging directory does not '
  77         'contain a configuration file.')
  78     raise IOError(message)
  79   with tf.gfile.GFile(config_path, 'r') as file_:
  80     config = yaml.load(file_, yaml.Loader)
  81     message = 'Resume run and write summaries and checkpoints to {}.'
  82     print(message.format(config.logdir))
  83   return config

  86 def train(model_fn, datasets, logdir, config):
  87   if not config:
  88     raise KeyError('You must specify a configuration.')
  89   logdir = logdir and os.path.expanduser(logdir)
  90   try:
  91     config = load_config(logdir)
  92   except RuntimeError:
  93     print('Failed to load existing config.')
  94   except IOError:
  95     config = save_config(config, logdir)
  96   with datasets.unlocked:
  97     datasets.train = datasets.train.make_one_shot_iterator()
  98     datasets.test = datasets.test.make_one_shot_iterator()
  99   trainer = trainer_.Trainer(logdir, config=config)
 100   cleanups = []
 101   try:
 102     with tf.variable_scope('graph', use_resource=True):
 103       data = tf.cond(
 104           tf.equal(trainer.phase, 'train'),
 105           datasets.train.get_next, datasets.test.get_next)
 106       score, summary, cleanups = model_fn(data, trainer, config)
 107       message = 'Graph contains {} trainable variables.'
 108       print(message.format(tools.count_weights()))
 109       if config.train_steps:
 110         trainer.add_phase(
 111             'train', config.train_steps, score, summary,
 112             batch_size=config.batch_shape[0],
 113             report_every=None,
 114             log_every=config.train_log_every,
 115             checkpoint_every=config.train_checkpoint_every)
 116       if config.test_steps:
 117         trainer.add_phase(
 118             'test', config.test_steps, score, summary,
 119             batch_size=config.batch_shape[0],
 120             report_every=config.test_steps,
 121             log_every=config.test_steps,
 122             checkpoint_every=config.test_checkpoint_every)
 123     for saver in config.savers:
 124       trainer.add_saver(**saver)
 125     for score in trainer.iterate(config.max_steps):
 126       yield score
 127   finally:
 128     for cleanup in cleanups:
 129       cleanup()

 132 def compute_objectives(posterior, prior, target, graph, config):
 133   raw_features = graph.cell.features_from_state(posterior)
 134   heads = graph.heads
 135   if config.imagination_horizon:
 136     imagination_start = posterior
 137     if config.imagination_skip_last:
 138       imagination_start = tools.nested.map(
 139           lambda x: x[:, :-config.imagination_skip_last], imagination_start)
 140     raw_states = imagine_forward(
 141         imagination_start, config.imagination_horizon, graph, config,
 142         graph.heads.action, stop_grad_post_action=False,
 143         stop_grad_pre_action=config.stop_grad_pre_action)
 144   else:
 145     raw_states = None
 146   objectives = []
 147   for name, scale in sorted(config.loss_scales.items(), key=lambda x: x[0]):
 148     if config.loss_scales[name] == 0.0:
 149       continue
 150     if name in config.heads and name not in config.gradient_heads:
 151       features = tf.stop_gradient(raw_features)
 152       include = r'.*/head_{}/.*'.format(name)
 153       exclude = None
 154     else:
 155       features = raw_features
 156       include = None
 157       exclude = None
 158 
 159     if name == 'divergence':
 160       loss = graph.cell.divergence_from_states(posterior, prior)
 161       if config.free_nats is not None:
 162         loss = tf.maximum(0.0, loss - float(config.free_nats))
 163       objectives.append(Objective('divergence', loss, min, include, exclude))
 164 
 165     elif name == 'cpc':
 166       pred = heads.cpc(graph.embedded)
 167       objective = compute_cpc_loss(pred, features, config)
 168       objectives.append(Objective('cpc', objective, max, include, exclude))
 169 
 170     elif name == 'overshooting':
 171       shape = tools.shape(graph.data['action'])
 172       length = tf.tile(tf.constant(shape[1])[None], [shape[0]])
 173       _, priors, posteriors, mask = tools.overshooting(
 174           graph.cell, {}, graph.embedded, graph.data['action'], length,
 175           config.overshooting_distance, posterior)
 176       posteriors, priors, mask = tools.nested.map(
 177           lambda x: x[:, :, 1:-1], (posteriors, priors, mask))
 178       if config.os_stop_posterior_grad:
 179         posteriors = tools.nested.map(tf.stop_gradient, posteriors)
 180       loss = graph.cell.divergence_from_states(posteriors, priors)
 181       if config.free_nats is not None:
 182         loss = tf.maximum(0.0, loss - float(config.free_nats))
 183       objectives.append(Objective('overshooting', loss, min, include, exclude))
 184 
 185     elif name == 'value':
 186       if config.value_source == 'dataset':
 187         loss = compute_value_loss(
 188             config, graph, priors, features, target['reward'])
 189       elif config.value_source == 'model':
 190         if 'action_target' in graph.heads or not config.imagination_horizon:
 191           if 'action_target' in graph.heads:
 192             policy = graph.heads.action_target
 193           else:
 194             policy = graph.heads.action
 195           states = imagine_forward(
 196               posterior, config.value_model_horizon, graph, config, policy)
 197         else:
 198           states = raw_states
 199         feat = graph.cell.features_from_state(states)
 200         loss = compute_value_loss(config, graph, states, feat, None)
 201       else:
 202         raise NotImplementedError(config.value_source)
 203       objectives.append(Objective('value', loss, min, include, exclude))
 204 
 205     elif name == 'action':
 206       if config.action_source == 'model':
 207         if not config.imagination_horizon:
 208           states = imagine_forward(
 209               posterior, config.action_model_horizon, graph, config,
 210               policy=graph.heads.action, stop_grad_post_action=False)
 211         else:
 212           states = raw_states
 213         feat = graph.cell.features_from_state(states)
 214         objective = compute_action_values(config, graph, states, feat)
 215         objectives.append(Objective(
 216             'action', objective, max, include, exclude))
 217       elif config.action_source == 'dataset':
 218         objective = heads.action(features).log_prob(target[name])
 219         objective -= compute_action_divergence(features, graph, config)
 220         objectives.append(Objective(
 221             'action', objective, max, include, exclude))
 222       else:
 223         raise NotImplementedError(config.action_source)
 224 
 225     elif name == 'reward':
 226       reward_mask = tf.squeeze(target['reward_mask'], [-1])
 227       logprob = heads.reward(features).log_prob(target[name]) * reward_mask
 228       objectives.append(Objective('reward', logprob, max, include, exclude))
 229 
 230     elif name == 'pcont' and config.pcont_label_weight:
 231       terminal = tf.cast(tf.less(target[name], 0.5), tf.float32)
 232       logprob = heads[name](features).log_prob(target[name])
 233       logprob *= 1 + terminal * (config.pcont_label_weight - 1)
 234       objectives.append(Objective(name, logprob, max, include, exclude))
 235 
 236     else:
 237       logprob = heads[name](features).log_prob(target[name])
 238       objectives.append(Objective(name, logprob, max, include, exclude))
 239 
 240   objectives = [o._replace(value=tf.reduce_mean(o.value)) for o in objectives]
 241   return objectives

 244 def imagine_forward(
 245     initial_state, distance, graph, config, policy,
 246     stop_grad_post_action=True, stop_grad_pre_action=True):
 247   extended_batch = np.prod(tools.shape(
 248       tools.nested.flatten(initial_state)[0])[:2])
 249   obs = tf.zeros([extended_batch] + list(graph.embedded.shape[2:]))
 250   use_obs = tf.zeros([extended_batch, 1], tf.bool)
 251   new_shape = lambda t: [
 252       tf.reduce_prod(tools.shape(t)[:2])] + tools.shape(t)[2:]
 253   initial_state = tools.nested.map(
 254       lambda tensor: tf.reshape(tensor, new_shape(tensor)),
 255       initial_state)
 256   def step_fn(prev, index):
 257     feature = graph.cell.features_from_state(prev)
 258     if stop_grad_pre_action:
 259       feature = tf.stop_gradient(feature)
 260     action = policy(feature).sample()
 261     if stop_grad_post_action:
 262       action = tf.stop_gradient(action)
 263     (_, state), _ = graph.cell((obs, action, use_obs), prev)
 264     return state
 265   states = tf.scan(step_fn, tf.range(distance), initial_state, back_prop=True)
 266   states = tools.nested.map(lambda x: tf.transpose(x, [1, 0, 2]), states)
 267   return states

 270 def compute_value_loss(config, graph, states, features, reward):
 271   if reward is None:
 272     reward = graph.heads.reward(features).mode()
 273   if config.value_maxent:
 274     reward -= compute_action_divergence(features, graph, config)
 275     reward -= compute_state_divergence(states, graph, config)
 276   if config.value_loss_pcont:
 277     pcont = tf.stop_gradient(graph.heads.pcont(features).mean())
 278   else:
 279     pcont = tf.ones_like(reward)
 280   pcont *= config.value_discount
 281   pred = graph.heads.value(features)
 282   if 'value_target' in graph.heads:
 283     value = graph.heads.value_target(features).mode()
 284   else:
 285     value = pred.mode()
 286   bootstrap = None
 287   if config.value_bootstrap:
 288     reward = reward[:, :-1]
 289     value = value[:, :-1]
 290     pcont = pcont[:, :-1]
 291     bootstrap = value[:, -1]
 292   return_ = control.lambda_return(
 293       reward, value, bootstrap, pcont,
 294       config.value_lambda, axis=1, stop_gradient=True)
 295   return_ = tf.concat([return_, tf.zeros_like(return_[:, -1:])], 1)
 296   loss = -pred.log_prob(return_)[:, :-1]
 297   if config.value_pcont_weight:
 298     loss *= tf.stop_gradient(tf.cumprod(tf.concat([
 299         tf.ones_like(pcont[:, :1]), pcont[:, :-1]], 1), 1))
 300   return loss

 303 def compute_action_values(config, graph, states, features):
 304   reward = graph.heads.reward(features).mode()
 305   reward -= compute_action_divergence(features, graph, config)
 306   reward -= compute_state_divergence(states, graph, config)
 307   if config.action_loss_pcont:
 308     pcont = graph.heads.pcont(features).mean()
 309     if config.action_pcont_stop_grad:
 310       pcont = tf.stop_gradient(pcont)
 311   else:
 312     pcont = tf.ones_like(reward)
 313   pcont *= config.action_discount
 314   if 'value' not in graph.heads:
 315     return control.discounted_return(
 316         reward, pcont, bootstrap=None, axis=1, stop_gradient=False)
 317   value = graph.heads.value(features).mode()
 318   bootstrap = None
 319   if config.action_bootstrap:
 320     reward = reward[:, :-1]
 321     value = value[:, :-1]
 322     pcont = pcont[:, :-1]
 323     bootstrap = value[:, -1]
 324   return_ = control.lambda_return(
 325       reward, value, bootstrap, pcont,
 326       config.action_lambda,
 327       axis=1, stop_gradient=False)
 328   if config.action_pcont_weight:
 329     return_ *= tf.stop_gradient(tf.cumprod(tf.concat([
 330         tf.ones_like(pcont[:, :1]), pcont[:, :-1]], 1), 1))
 331   return return_

 334 def compute_action_divergence(features, graph, config):
 335   features = tf.stop_gradient(features)
 336   pred = graph.heads.action(features)
 337   if not config.action_beta:
 338     return 0.0
 339   try:
 340     amount = -pred.entropy()
 341   except NotImplementedError:
 342     samples = pred.sample(100)
 343     amount = tf.reduce_mean(pred.log_prob(samples), 0)
 344   amount *= config.action_beta
 345   value = config.action_beta_dims_value
 346   if value and value < 0:
 347     amount /= value * float(pred.event_shape[-1].value)
 348   if value and value > 0:
 349     amount *= value * float(pred.event_shape[-1].value)
 350   return amount

 353 def compute_state_divergence(states, graph, config):
 354   pred = graph.cell.dist_from_state(states)
 355   if not config.state_beta:
 356     return 0.0
 357   try:
 358     amount = -pred.entropy()
 359   except NotImplementedError:
 360     samples = pred.sample(100)
 361     amount = tf.reduce_mean(pred.log_prob(samples), 0)
 362   amount *= config.state_beta
 363   return amount

 366 def compute_cpc_loss(pred, features, config):
 367   if config.cpc_contrast == 'batch':
 368     ta = tf.TensorArray(tf.float32, 0, True, element_shape=[None, None])
 369     _, _, ta = tf.while_loop(
 370         lambda i, f, ta: tf.less(i, tf.shape(f)[0]),
 371         lambda i, f, ta: (
 372             i + 1, f, ta.write(ta.size(), pred.log_prob(tf.roll(f, i, 0)))),
 373         (0, features, ta), back_prop=True, swap_memory=True)
 374     positive = pred.log_prob(features)
 375     negative = tf.reduce_logsumexp(ta.stack(), 0)
 376     return positive - negative
 377   elif config.cpc_contrast == 'time':
 378     ta = tf.TensorArray(tf.float32, 0, True, element_shape=[None, None])
 379     _, _, ta = tf.while_loop(
 380         lambda i, f, ta: tf.less(i, tf.shape(f)[1]),
 381         lambda i, f, ta: (
 382             i + 1, f, ta.write(ta.size(), pred.log_prob(tf.roll(f, i, 1)))),
 383         (0, features, ta), back_prop=True, swap_memory=True)
 384     positive = pred.log_prob(features)
 385     negative = tf.reduce_logsumexp(ta.stack(), 0)
 386     return positive - negative
 387   elif config.cpc_contrast == 'window':
 388     assert config.cpc_batch_amount <= config.batch_shape[0]
 389     assert config.cpc_time_amount <= config.batch_shape[1]
 390     total_amount = config.cpc_batch_amount * config.cpc_time_amount
 391     ta = tf.TensorArray(tf.float32, 0, True, element_shape=[None, None])
 392     def compute_negatives(index, ta):
 393       batch_shift = tf.math.floordiv(index, config.cpc_time_amount)
 394       time_shift = tf.mod(index, config.cpc_time_amount)
 395       batch_shift -= config.cpc_batch_amount // 2
 396       time_shift -= config.cpc_time_amount // 2
 397       rolled = tf.roll(tf.roll(features, batch_shift, 0), time_shift, 1)
 398       return ta.write(ta.size(), pred.log_prob(rolled))
 399     _, ta = tf.while_loop(
 400         lambda index, ta: tf.less(index, total_amount),
 401         lambda index, ta: (index + 1, compute_negatives(index, ta)),
 402         (0, ta), back_prop=True, swap_memory=True)
 403     positive = pred.log_prob(features)
 404     negative = tf.reduce_logsumexp(ta.stack(), 0)
 405     return positive - negative
 406   else:
 407     raise NotImplementedError(config.cpc_contrast)

 410 def apply_optimizers(objectives, trainer, config):
 411   # Make sure all losses are computed and apply loss scales.
 412   processed = []
 413   values = [ob.value for ob in objectives]
 414   for ob in objectives:
 415     loss = {min: ob.value, max: -ob.value}[ob.goal]
 416     loss *= config.loss_scales[ob.name]
 417     with tf.control_dependencies(values):
 418       loss = tf.identity(loss)
 419     processed.append(ob._replace(value=loss, goal=min))
 420   # Merge objectives that operate on the whole model to compute only one
 421   # backward pass and to share optimizer statistics.
 422   objectives = []
 423   losses = []
 424   for ob in processed:
 425     if ob.include is None and ob.exclude is None:
 426       assert ob.goal == min
 427       losses.append(ob.value)
 428     else:
 429       objectives.append(ob)
 430   loss = tf.reduce_sum(losses)
 431   # modules = ['encoder', 'rnn']
 432   # modules += ['head_{}'.format(name) for name in config.gradient_heads]
 433   # include = r'graph/({})/.*'.format('|'.join(modules))
 434   include = r'.*'
 435   objectives.append(Objective('model', loss, min, include, None))
 436   # Apply optimizers and collect loss summaries.
 437   summaries = []
 438   grad_norms = {}
 439   for ob in objectives:
 440     assert ob.name in list(config.loss_scales.keys()) + ['model'], ob
 441     assert ob.goal == min, ob
 442     if ob.name not in config.optimizers:
 443       with config.optimizers.unlocked:
 444         config.optimizers[ob.name] = config.optimizers.default.copy()
 445     optimizer = config.optimizers[ob.name](
 446         include=ob.include,
 447         exclude=ob.exclude,
 448         step=trainer.step,
 449         log=trainer.log,
 450         debug=config.debug,
 451         name=ob.name)
 452     condition = tf.equal(trainer.phase, 'train')
 453     summary, grad_norm = optimizer.maybe_minimize(condition, ob.value)
 454     summaries.append(summary)
 455     grad_norms[ob.name] = grad_norm
 456   return summaries, grad_norms

 459 def simulate(metrics, config, params, graph, cleanups, gif_summary, name):
 460   def env_ctor():
 461     env = params.task.env_ctor()
 462     if params.save_episode_dir:
 463       env = control.wrappers.CollectDataset(env, params.save_episode_dir)
 464     return env
 465   bind_or_none = lambda x, **kw: x and functools.partial(x, **kw)
 466   cell = graph.cell
 467   agent_config = tools.AttrDict(
 468       cell=cell,
 469       encoder=graph.encoder,
 470       planner=functools.partial(params.planner, graph=graph),
 471       objective=bind_or_none(params.objective, graph=graph),
 472       exploration=params.exploration,
 473       preprocess_fn=config.preprocess_fn,
 474       postprocess_fn=config.postprocess_fn)
 475   params = params.copy()
 476   with params.unlocked:
 477     params.update(agent_config)
 478   with agent_config.unlocked:
 479     agent_config.update(params)
 480   with tf.variable_scope(name):
 481     summaries = []
 482     env = control.create_batch_env(
 483         env_ctor, params.num_envs, config.isolate_envs)
 484     agent = control.MPCAgent(env, graph.step, False, False, agent_config)
 485     cleanup = lambda: env.close()
 486     scores, lengths, data = control.simulate(
 487         agent, env, params.num_episodes, params.num_steps)
 488     summaries.append(tf.summary.scalar('return', scores[0]))
 489     summaries.append(tf.summary.scalar('length', lengths[0]))
 490     if gif_summary:
 491       summaries.append(tools.gif_summary(
 492           'gif', data['image'], max_outputs=1, fps=20))
 493     write_metrics = [
 494         metrics.add_scalars(name + '/return', scores),
 495         metrics.add_scalars(name + '/length', lengths),
 496         # metrics.add_tensor(name + '/frames', data['image']),
 497     ]
 498     with tf.control_dependencies(write_metrics):
 499       summary = tf.summary.merge(summaries)
 500   cleanups.append(cleanup)  # Work around tf.cond() tensor return type.
 501   return summary, tf.reduce_mean(scores)

 504 def print_metrics(metrics, step, every, decimals=2, name='metrics'):
 505   factor = 10 ** decimals
 506   means, updates = [], []
 507   for key, value in metrics.items():
 508     key = 'metrics_{}_{}'.format(name, key)
 509     mean = tools.StreamingMean((), tf.float32, key)
 510     means.append(mean)
 511     updates.append(mean.submit(value))
 512   with tf.control_dependencies(updates):
 513     message = '{}: step/{} ='.format(name, '/'.join(metrics.keys()))
 514     print_metrics = tf.cond(
 515         tf.equal(step % every, 0),
 516         lambda: tf.print(message, [step] + [
 517             tf.round(mean.clear() * factor) / factor for mean in means]),
 518         tf.no_op)
 519   return print_metrics

 522 def collect_initial_episodes(metrics, config):
 523   items = config.random_collects.items()
 524   items = sorted(items, key=lambda x: x[0])
 525   for name, params in items:
 526     metrics.set_tags(
 527         global_step=0,
 528         step=0,
 529         phase=name.split('-')[0],
 530         time=datetime.datetime.utcnow().strftime('%Y%m%dT%H%M%S'))
 531     tf.gfile.MakeDirs(params.save_episode_dir)
 532     # Compute number of missing episodes and steps.
 533     filenames = tf.gfile.Glob(os.path.join(params.save_episode_dir, '*.npz'))
 534     num_episodes = len(filenames)
 535     # num_steps = sum([
 536     #     int(f.rsplit('.', 1)[0].split('-')[2]) for f in filenames])
 537     num_steps = sum([
 538         int(re.search(r'-([0-9]+)\.npz', f).group(1)) for f in filenames])
 539     remaining_episodes = params.num_episodes - num_episodes
 540     remaining_steps = params.num_steps - num_steps
 541     if remaining_episodes <= 0 and remaining_steps <= 0:
 542       continue
 543     # User message.
 544     if params.give_rewards:
 545       env_ctor = params.task.env_ctor
 546       word = 'with'
 547     else:
 548       env_ctor = functools.partial(
 549           lambda ctor: control.wrappers.NoRewardHint(ctor()),
 550           params.task.env_ctor)
 551       word = 'without'
 552     message = 'Collecting initial {} episodes or {} steps ({}) {} rewards.'
 553     print(message.format(remaining_episodes, remaining_steps, name, word))
 554     episodes = control.random_episodes(
 555         env_ctor, remaining_episodes, remaining_steps,
 556         params.save_episode_dir, config.isolate_envs)
 557     scores = [episode['reward'].sum() for episode in episodes]
 558     lengths = [len(episode['reward']) for episode in episodes]
