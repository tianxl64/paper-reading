 154 def _emax_from_kmax(k_max: int) -> int:
 155     emax = int(round(math.log2(k_max)))
 156     assert (1 << emax) == k_max, "k_max must be power of two"
 157     return emax

 160 def _sample_step_excluding_dmin(device: torch.device, B: int, T: int, k_max: int) -> tuple[torch.Tensor, torch.Tensor]:
 161     emax = _emax_from_kmax(k_max)
 162     # step_idx in [0, emax) i.e. excludes emax (d_min)
 163     step_idx = torch.randint(low=0, high=max(1, emax), size=(B, T), device=device, dtype=torch.long)
 164     d = 1.0 / (1 << step_idx).to(torch.float32)
 165     return d, step_idx

 168 def _sample_tau_for_step(device: torch.device, B: int, T: int, k_max: int, step_idx: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
 169     # K = 2^step_idx
 170     K = (1 << step_idx).to(torch.long)  # (B,T)
 171     u = torch.rand((B, T), device=device, dtype=torch.float32)
 172     j_idx = torch.floor(u * K.to(torch.float32)).to(torch.long)  # (B,T) in [0,K)
 173     tau = j_idx.to(torch.float32) / K.to(torch.float32)          # (B,T)
 174     scale = torch.div(torch.tensor(k_max, device=device), K, rounding_mode="floor")  # (B,T)
 175     tau_idx = j_idx * scale                                      # (B,T) <= k_max-1
 176     return tau, tau_idx

 179 def dynamics_pretrain_loss(
 180     dynamics: torch.nn.Module,
 181     *,
 182     z1: torch.Tensor,                    # (B,T,Sz,Dz) packed clean targets
 183     actions: Optional[torch.Tensor],     # (B,T) or None
 184     act_mask: Optional[torch.Tensor],    # (A,) or None
 185     k_max: int,
 186     B_self: int,
 187     step: int,
 188     bootstrap_start: int,
 189     agent_tokens: Optional[torch.Tensor] = None,
 190 ) -> tuple[torch.Tensor, Dict[str, torch.Tensor]]:
 191     device = z1.device
 192     B, T = z1.shape[:2]
 193     assert 0 <= B_self < B
 194     B_emp = B - B_self
 195     emax = _emax_from_kmax(k_max)
 196 
 197     # action mask slices
 198     act_mask_full = act_mask
 199     act_mask_self = None if act_mask_full is None else act_mask_full[B_emp:]
 200 
 201     # step idx: empirical rows are finest (d_min), self rows sample coarser
 202     step_idx_emp = torch.full((B_emp, T), emax, device=device, dtype=torch.long)
 203     if B_self > 0:
 204         d_self, step_idx_self = _sample_step_excluding_dmin(device, B_self, T, k_max)
 205         step_idx_full = torch.cat([step_idx_emp, step_idx_self], dim=0)
 206     else:
 207         d_self = torch.zeros((0, T), device=device, dtype=torch.float32)
 208         step_idx_self = torch.zeros((0, T), device=device, dtype=torch.long)
 209         step_idx_full = step_idx_emp
 210 
 211     # sigma/tau per row/time
 212     sigma_full, sigma_idx_full = _sample_tau_for_step(device, B, T, k_max, step_idx_full)
 213     sigma_emp = sigma_full[:B_emp]
 214     sigma_self = sigma_full[B_emp:]
 215     sigma_idx_self = sigma_idx_full[B_emp:]
 216 
 217     # Corrupt inputs
 218     z0_full = torch.randn_like(z1)
 219     z_tilde_full = (1.0 - sigma_full)[..., None, None] * z0_full + sigma_full[..., None, None] * z1
 220     z_tilde_self = z_tilde_full[B_emp:]
 221 
 222     # Weights
 223     w_emp = 0.9 * sigma_emp + 0.1
 224     w_self = 0.9 * sigma_self + 0.1
 225 
 226     # Main forward
 227     z1_hat_full, _ = dynamics(actions, step_idx_full, sigma_idx_full, z_tilde_full, act_mask=act_mask_full, agent_tokens=agent_tokens)
 228     z1_hat_emp = z1_hat_full[:B_emp]
 229     z1_hat_self = z1_hat_full[B_emp:]
 230 
 231     flow_per = (z1_hat_emp.float() - z1[:B_emp].float()).pow(2).mean(dim=(2, 3))  # (B_emp,T)
 232     loss_emp = (flow_per * w_emp).mean()
 233 
 234     boot_mse = torch.zeros((), device=device, dtype=torch.float32)
 235     loss_self = torch.zeros((), device=device, dtype=torch.float32)
 236 
 237     do_boot = (B_self > 0) and (step >= bootstrap_start)
 238     if do_boot:
 239         d_half = d_self / 2.0
 240         step_idx_half = step_idx_self + 1
 241         sigma_plus = sigma_self + d_half
 242         sigma_idx_plus = sigma_idx_self + (torch.tensor(k_max, device=device, dtype=torch.float32) * d_half).to(torch.long)
 243 
 244         z1_hat_half1, _ = dynamics(actions[B_emp:] if actions is not None else None, step_idx_half, sigma_idx_self, z_tilde_self, act_mask=act_mask_self, agent_tokens=agent_tokens[B_emp:] if agent_tokens is not None else None)
 245         b_prime = (z1_hat_half1.float() - z_tilde_self.float()) / (1.0 - sigma_self).clamp_min(1e-6)[..., None, None]
 246         z_prime = z_tilde_self.float() + b_prime * d_half[..., None, None]
 247 
 248         z1_hat_half2, _ = dynamics(actions[B_emp:] if actions is not None else None, step_idx_half, sigma_idx_plus, z_prime.to(z_tilde_self.dtype), act_mask=act_mask_self, agent_tokens=agent_tokens[B_emp:] if agent_tokens is not None else None)
 249         b_doubleprime = (z1_hat_half2.float() - z_prime.float()) / (1.0 - sigma_plus).clamp_min(1e-6)[..., None, None]
 250 
 251         vhat_sigma = (z1_hat_self.float() - z_tilde_self.float()) / (1.0 - sigma_self).clamp_min(1e-6)[..., None, None]
 252         vbar_target = ((b_prime + b_doubleprime) / 2.0).detach()
 253 
 254         boot_per = (1.0 - sigma_self).pow(2) * (vhat_sigma - vbar_target).pow(2).mean(dim=(2, 3))  # (B_self,T)
 255         loss_self = (boot_per * w_self).mean()
 256         boot_mse = boot_per.mean()
 257 
 258     # Combine losses
 259     loss = ((loss_emp * (B - B_self)) + (loss_self * B_self)) / B
 260 
 261     aux = {
 262         "flow_mse": flow_per.mean().detach(),
 263         "bootstrap_mse": boot_mse.detach(),
 264         "loss_emp": loss_emp.detach(),
 265         "loss_self": loss_self.detach(),
 266         "sigma_mean": sigma_full.mean().detach(),
 267     }
 268     return loss, aux

 275 def make_tau_schedule(*, k_max: int, schedule: str, d: Optional[float] = None) -> Dict[str, Any]:
 276     """
 277     Returns a schedule dict:
 278       K = number of integration steps (also grid size)
 279       e = log2(K)  (step_idx)
 280       scale = k_max // K
 281       tau_idx[i] = discrete signal index at step i
 282       tau[i] = i/K
 283       dt = 1/K
 284     """
 285     assert _is_pow2(k_max), "k_max must be power of two"
 286     if schedule == "finest":
 287         K = k_max
 288     elif schedule == "shortcut":
 289         assert d is not None, "shortcut schedule requires --eval_d"
 290         inv = int(round(1.0 / float(d)))
 291         assert _is_pow2(inv), "eval_d must be 1/(power of two)"
 292         assert inv <= k_max, "eval_d must be >= 1/k_max"
 293         assert (k_max % inv) == 0, "k_max must be divisible by 1/eval_d"
 294         K = inv
 295     else:
 296         raise ValueError(f"unknown schedule: {schedule}")
 297 
 298     e = int(round(math.log2(K)))
 299     scale = k_max // K
 300     tau = [i / K for i in range(K)] + [1.0]
 301     tau_idx = [i * scale for i in range(K)] + [k_max]  # allow final clean index
 302     return dict(K=K, e=e, scale=scale, tau=tau, tau_idx=tau_idx, dt=1.0 / K, schedule=schedule, d=1.0 / K)

 305 @torch.no_grad()
 306 def sample_one_timestep_packed(
 307     dyn: Dynamics,
 308     *,
 309     past_packed: torch.Tensor,          # (B,t,n_spatial,d_spatial)
 310     k_max: int,
 311     sched: Dict[str, Any],
 312     actions: Optional[torch.Tensor] = None,     # (B,T,A) aligned to frames or None
 313     act_mask: Optional[torch.Tensor] = None,    # (B,T,A) or (A,) or None
 314 ) -> torch.Tensor:
 315     device = past_packed.device
 316     dtype = past_packed.dtype
 317     B, t = past_packed.shape[:2]
 318     n_spatial, d_spatial = past_packed.shape[2], past_packed.shape[3]
 319 
 320     K = int(sched["K"])
 321     e = int(sched["e"])
 322     tau = sched["tau"]
 323     tau_idx = sched["tau_idx"]
 324     dt = float(sched["dt"])
 325 
 326     # start from noise at tau=0
 327     z = torch.randn((B, 1, n_spatial, d_spatial), device=device, dtype=dtype)
 328 
 329     emax = int(round(math.log2(k_max)))
 330 
 331     step_idxs_full = torch.full((B, t + 1), emax, device=device, dtype=torch.long)
 332     step_idxs_full[:, -1] = e  # only the sampled timestep uses the shortcut step
 333 
 334     signal_idxs_full = torch.full((B, t + 1), k_max - 1, device=device, dtype=torch.long)
 335 
 336     # broadcast (A,) -> (B,T,A) if needed (only if actions are present)
 337     if act_mask is not None and act_mask.dim() == 1:
 338         act_mask = act_mask.view(1, 1, -1)
 339 
 340     for i in range(K):
 341         tau_i = float(tau[i])
 342         sig_i = int(tau_idx[i])
 343 
 344         signal_idxs_full[:, -1] = sig_i
 345         packed_seq = torch.cat([past_packed, z], dim=1)  # (B,t+1,...)
 346 
 347         actions_in = None if actions is None else actions[:, : t + 1]
 348         actmask_in = None if act_mask is None else act_mask[:, : t + 1]
 349 
 350         x1_hat_full, _ = dyn(
 351             actions_in,
 352             step_idxs_full,
 353             signal_idxs_full,
 354             packed_seq,
 355             act_mask=actmask_in,
 356             agent_tokens=None,
 357         )
 358         x1_hat = x1_hat_full[:, -1:, :, :]  # (B,1,n_spatial,d_spatial)
 359 
 360         denom = max(1e-4, 1.0 - tau_i)
 361         b = (x1_hat.float() - z.float()) / denom
 362         z = (z.float() + b * dt).to(dtype)
 363 
 364     return z[:, 0]  # (B,n_spatial,d_spatial)

 367 @torch.no_grad()
 368 def sample_autoregressive_packed_sequence(
 369     dyn: Dynamics,
 370     *,
 371     z_gt_packed: torch.Tensor,                  # (B,T,n_spatial,d_spatial)
 372     ctx_length: int,
 373     horizon: int,
 374     k_max: int,
 375     sched: Dict[str, Any],
 376     actions: Optional[torch.Tensor] = None,     # (B,T,A) or None
 377     act_mask: Optional[torch.Tensor] = None,    # (B,T,A) or (A,) or None
 378 ) -> torch.Tensor:
 379     B, T = z_gt_packed.shape[:2]
 380     L = min(T, ctx_length + horizon)
 381     ctx_length = min(ctx_length, L - 1)
 382     horizon = min(horizon, L - ctx_length)
 383 
 384     outs = [z_gt_packed[:, t] for t in range(ctx_length)]
 385 
 386     for t in range(ctx_length, ctx_length + horizon):
 387         past = torch.stack(outs, dim=1)  # (B,t,...)
 388         z_next = sample_one_timestep_packed(
 389             dyn,
 390             past_packed=past,
 391             k_max=k_max,
 392             sched=sched,
 393             actions=actions,
 394             act_mask=act_mask,
 395         )
 396         outs.append(z_next)
 397 
 398     return torch.stack(outs, dim=1)

 401 @torch.no_grad()
 402 def decode_packed_to_frames(
 403     decoder: Decoder,
 404     *,

 415 @torch.no_grad()
 416 def log_dynamics_eval_wandb(
 417     *,
 418     gt: torch.Tensor,          # (B,T,C,H,W) float [0,1]
 419     pred: torch.Tensor,        # (B,T,C,H,W) float [0,1]
 420     ctx_length: int,
 421     step: int,
 422     tag: str,
 423     max_items: int = 4,
 424     gap_px: int = 16,
 425 ):
 426     B, T, C, H, W = gt.shape
 427     Bv = min(B, max_items)
 428 
 429     def tile_time(x: torch.Tensor) -> torch.Tensor:
 430         x = x[:Bv]
 431         B_, T_, C_, H_, W_ = x.shape
 432         ctx = int(max(0, min(ctx_length, T_)))
 433 
 434         y = x.permute(0, 2, 3, 1, 4).contiguous().view(B_, C_, H_, T_ * W_)
 435 
 436         if gap_px > 0 and 0 < ctx < T_:
 437             split = ctx * W_
 438             left = y[..., :split]
 439             right = y[..., split:]
 440             gap = torch.zeros((B_, C_, H_, gap_px), device=y.device, dtype=y.dtype)
 441             y = torch.cat([left, gap, right], dim=-1)
 442         return y
 443 
 444     gt_t = tile_time(gt)
 445     pr_t = tile_time(pred)
 446 
 447     # Stack rows: GT / Pred
 448     panel = torch.cat([gt_t, pr_t], dim=2)   # (Bv,C,2H,TW+gap)
 449     big = torch.cat([panel[i] for i in range(Bv)], dim=1)  # (C,Bv*2H,TW+gap)
 450 
 451     big = (big.clamp(0, 1) * 255.0).to(torch.uint8)
 452     big_hwc = big.permute(1, 2, 0).cpu().numpy()
 453 
 454     wandb.log(
 455         {f"{tag}/viz": wandb.Image(big_hwc, caption=f"rows=GT/Pred | ctx={ctx_length} | T={T}")},
 456         step=step,
 457     )

 460 @torch.no_grad()
 461 def run_dynamics_eval(
 462     *,
 463     encoder: Encoder,
 464     decoder: Decoder,
 465     dyn: Dynamics,
 466     frames: torch.Tensor,            # (B,T,C,H,W) float [0,1]
 467     actions: Optional[torch.Tensor],    # (B,T,A) or None
 468     act_mask: Optional[torch.Tensor], # (A,) or None
 469     H: int, W: int, C: int, patch: int,
 470     packing_factor: int,
 471     k_max: int,
 472     ctx_length: int,
 473     horizon: int,
 474     sched: Dict[str, Any],
 475     max_items: int,
 476     step: int,

 479     dyn.eval()

 484     horizon = min(horizon, T_eval - ctx_length)
 485 
 486     frames_eval = frames[:, :T_eval]
 487 
 488     patches = temporal_patchify(frames_eval, patch)
 489     z_btLd, _ = encoder(patches)  # (B,T_eval,L,D_b)
 490     assert z_btLd.shape[2] % packing_factor == 0
 491     n_spatial = z_btLd.shape[2] // packing_factor
 492     z_gt_packed = pack_bottleneck_to_spatial(z_btLd, n_spatial=n_spatial, k=packing_factor)  # (B,T_eval,Sz,Dz)
 493 
 494     actions_eval = None if actions is None else actions[:, :T_eval]
 495     act_mask_eval = None if act_mask is None else act_mask[:, :T_eval] if act_mask.dim() == 3 else act_mask
 496 
 497     z_pred_packed = sample_autoregressive_packed_sequence(
 498         dyn,
 499         z_gt_packed=z_gt_packed,
 500         ctx_length=ctx_length,
 501         horizon=horizon,
 502         k_max=k_max,
 503         sched=sched,
 504         actions=actions_eval,
 505         act_mask=act_mask_eval,
 506     )
 507 
 508     pred_frames = decode_packed_to_frames(
 509         decoder,
 510         z_packed=z_pred_packed,
 511         H=H, W=W, C=C, patch=patch,
 512         packing_factor=packing_factor,
 513     )
 514 
 515     # floor baseline: repeat last context frame over horizon
 516     floor = frames_eval.clone()
 517     if horizon > 0:
 518         floor[:, ctx_length:ctx_length + horizon] = frames_eval[:, ctx_length - 1:ctx_length].expand(-1, horizon, -1, -1, -1)
 519 
 520     # metric on horizon only
 521     gt_h    = frames_eval[:, ctx_length:ctx_length + horizon]         # (B,Hz,C,H,W)
 522     pred_h  = pred_frames[:, ctx_length:ctx_length + horizon]
 523     floor_h = floor[:, ctx_length:ctx_length + horizon]
 524 
 525     mse_pred  = (pred_h.float()  - gt_h.float()).pow(2).mean()
 526     mse_floor = (floor_h.float() - gt_h.float()).pow(2).mean()
 527 
 528     psnr_pred  = 10.0 * torch.log10(1.0 / mse_pred.clamp_min(1e-12))
 529     psnr_floor = 10.0 * torch.log10(1.0 / mse_floor.clamp_min(1e-12))
 530 
 531     mse_ratio = mse_pred / mse_floor.clamp_min(1e-12)     # <1 is better
 532     psnr_gain = psnr_pred - psnr_floor                    # >0 is better
 533 
 534     # per-timestep horizon MSE: log first/mid/last
 535     # per_t: (Hz,)
 536     per_t_pred  = (pred_h.float()  - gt_h.float()).pow(2).mean(dim=(0,2,3,4))
 537     per_t_floor = (floor_h.float() - gt_h.float()).pow(2).mean(dim=(0,2,3,4))
 538 
 539     if horizon > 0:
 540         i0 = 0
 541         im = (horizon - 1) // 2
 542         i1 = horizon - 1
 543 
 544         wandb.log(
 545             {
 546                 "eval/mse_pred": float(mse_pred.item()),
 547                 "eval/mse_floor": float(mse_floor.item()),
 548                 "eval/mse_ratio_pred_over_floor": float(mse_ratio.item()),
 549 
 550                 "eval/psnr_pred": float(psnr_pred.item()),
 551                 "eval/psnr_floor": float(psnr_floor.item()),
 552                 "eval/psnr_gain_over_floor_db": float(psnr_gain.item()),
 553 
 554                 # 1-indexed step labels in the horizon
 555                 "eval/mse_pred_t1": float(per_t_pred[i0].item()),
 556                 "eval/mse_pred_tmid": float(per_t_pred[im].item()),
 557                 "eval/mse_pred_tend": float(per_t_pred[i1].item()),
 558 
 559                 "eval/mse_floor_t1": float(per_t_floor[i0].item()),
 560                 "eval/mse_floor_tmid": float(per_t_floor[im].item()),
 561                 "eval/mse_floor_tend": float(per_t_floor[i1].item()),
 562             },
 563             step=step,
 564         )
 565 
 566     log_dynamics_eval_wandb(
 567         gt=frames_eval,
 568         pred=pred_frames,
 569         ctx_length=ctx_length,
 570         step=step,
 571         tag="eval",
 572         max_items=max_items,
 573     )
 574 
 575     if dyn_was_training:
 576         dyn.train()
 577

 579 def train(args):
 580     ddp, rank, world_size, local_rank = init_distributed()
 581     device = torch.device(f"cuda:{local_rank}" if torch.cuda.is_available() else "cpu")
 582 
 583     seed_everything(args.seed + rank)
 584 
 585     # Dataset and DataLoader
 586     if args.use_actions:
 587         from wm_dataset import WMDataset, collate_batch
 588         dataset = WMDataset(
 589             data_dir=args.data_dirs,
 590             frames_dir=args.frame_dirs,
 591             seq_len=args.seq_len,
 592             img_size=128,
 593             action_dim=16,
 594             tasks_json=args.tasks_json,
 595             tasks=TASK_SET,
 596             verbose=is_rank0(),
 597         )
 598         sampler = DistributedSampler(dataset, num_replicas=world_size, rank=rank, shuffle=True) if ddp else None
 599         loader = DataLoader(
 600             dataset,
 601             batch_size=args.batch_size,
 602             sampler=sampler,
 603             shuffle=(sampler is None),
 604             num_workers=args.num_workers,
 605             pin_memory=True,
 606             drop_last=True,
 607             persistent_workers=(args.num_workers > 0),
 608             worker_init_fn=worker_init_fn,
 609             collate_fn=collate_batch,
 610         )
 611     else:
 612         dataset = ShardedFrameDataset(
 613             outdirs=args.frame_dirs,
 614             tasks=TASK_SET,
 615             seq_len=args.seq_len,
 616         )
 617         sampler = DistributedSampler(dataset, num_replicas=world_size, rank=rank, shuffle=True) if ddp else None
 618         loader = DataLoader(
 619             dataset,
 620             batch_size=args.batch_size,
 621             sampler=sampler,
 622             shuffle=(sampler is None),
 623             num_workers=args.num_workers,
 624             pin_memory=True,
 625             drop_last=True,
 626             persistent_workers=(args.num_workers > 0),
 627             worker_init_fn=worker_init_fn,
 628         )
 629 
 630     # Load frozen tokenizer
 631     tok_override = {}
 632     if args.H is not None: tok_override["H"] = args.H
 633     if args.W is not None: tok_override["W"] = args.W
 634     if args.C is not None: tok_override["C"] = args.C
 635     if args.patch is not None: tok_override["patch"] = args.patch
 636 
 637     encoder, decoder, tok_args = load_frozen_tokenizer_from_pt_ckpt(
 638         args.tokenizer_ckpt, device=device, override=tok_override
 639     )
 640 
 641     H = int(tok_args.get("H", 128))
 642     W = int(tok_args.get("W", 128))
 643     C = int(tok_args.get("C", 3))
 644     patch = int(tok_args.get("patch", 4))
 645     n_latents = int(tok_args.get("n_latents", 16))
 646     d_bottleneck = int(tok_args.get("d_bottleneck", 32))
 647 
 648     assert H % patch == 0 and W % patch == 0
 649     assert n_latents % args.packing_factor == 0
 650     n_spatial = n_latents // args.packing_factor
 651     d_spatial = d_bottleneck * args.packing_factor
 652 
 653     # Build dynamics model
 654     dyn = Dynamics(
 655         d_model=args.d_model_dyn,
 656         d_bottleneck=d_bottleneck,
 657         d_spatial=d_spatial,
 658         n_spatial=n_spatial,
 659         n_register=args.n_register,
 660         n_agent=args.n_agent,
 661         n_heads=args.n_heads,
 662         depth=args.dyn_depth,
 663         k_max=args.k_max,
 664         dropout=args.dropout,
 665         mlp_ratio=args.mlp_ratio,
 666         time_every=args.time_every,
 667         space_mode=args.space_mode,
 668         scale_pos_embeds=args.scale_pos_embeds,
 669     ).to(device)
 670 
 671     if is_rank0():
 672         print(dyn)
 673         param_count = sum(p.numel() for p in dyn.parameters() if p.requires_grad)
 674         print(f"Learnable parameters (dynamics): {param_count:,}")
 675         print(f"[tokenizer] H={H} W={W} C={C} patch={patch} n_lat={n_latents} d_b={d_bottleneck} packing={args.packing_factor}")
 676 
 677     if args.compile:
 678         dyn = torch.compile(dyn)
 679 
 680     if ddp:
 681         dyn = torch.nn.parallel.DistributedDataParallel(
 682             dyn, device_ids=[local_rank], output_device=local_rank, broadcast_buffers=False
 683         )
 684 
 685     # Optimizer and scaler
 686     opt = torch.optim.AdamW(
 687         dyn.parameters(), lr=args.lr, weight_decay=args.weight_decay, betas=(0.9, 0.999)
 688     )
 689     use_amp = torch.cuda.is_available()
 690     scaler = GradScaler(device="cuda", enabled=use_amp)
 691 
 692     # Initialize wandb
 693     if is_rank0():
 694         wandb.init(
 695             project=args.wandb_project,
 696             name=args.wandb_run_name,
 697             entity=args.wandb_entity,
 698             mode="online",
 699             config=vars(args),
 700         )
 701 
 702     # Resume from checkpoint
 703     step = 0
 704     start_epoch = 0
 705     ckpt_dir = Path(args.ckpt_dir)
 706     if args.resume is not None:
 707         step, start_epoch = load_ckpt(Path(args.resume), dyn_model=dyn, opt=opt, scaler=scaler)
 708         if is_rank0():
 709             print(f"[rank0] Resumed from {args.resume} (step={step}, epoch={start_epoch})")
 710 
 711     # Training loop
 712     dyn.train()
 713     t0 = time.time()
 714     grad_accum = max(1, int(args.grad_accum))
 715 
 716     while step < args.max_steps:
 717         for epoch in range(start_epoch, 10_000_000):
 718             if sampler is not None:
 719                 sampler.set_epoch(epoch)
 720 
 721             for batch in loader:
 722                 if step >= args.max_steps:
 723                     break
 724 
 725                 if args.use_actions:
 726                     obs_u8 = batch["obs"].to(device, non_blocking=True)          # (B,T+1,3,H,W) uint8
 727                     act    = batch["act"].to(device, non_blocking=True)          # (B,T,16) float
 728                     mask   = batch["act_mask"].to(device, non_blocking=True)     # (B,T,16) float (optional but good)
 729 
 730                     act = act.clamp(-1, 1) * mask
 731 
 732                     # Keep obs[0..T-1], align action[t] as action that produced obs[t]
 733                     frames = obs_u8[:, :-1].float() / 255.0                      # (B,T,3,H,W)
 734                     actions = torch.zeros_like(act)
 735                     actions[:, 1:] = act[:, :-1]
 736                     act_mask = torch.zeros_like(mask)
 737                     act_mask[:, 1:] = mask[:, :-1]
 738                 else:
 739                     frames = batch.to(device, non_blocking=True)                 # (B,T,3,H,W)
 740                     actions = None
 741                     act_mask = None
 742 
 743                 # Safeguard: convert to [0, 1] if dataset returns uint8
 744                 if frames.dtype == torch.uint8:
 745                     frames = frames.float() / 255.0
 746 
 747                 # Frozen encoder -> packed spatial tokens z1
 748                 with torch.no_grad():
 749                     patches = temporal_patchify(frames, patch)  # (B,T,Np,Dp)
 750                     z_btLd, _ = encoder(patches)                # (B,T,n_latents,d_b)
 751                     z1 = pack_bottleneck_to_spatial(z_btLd, n_spatial=n_spatial, k=args.packing_factor)  # (B,T,Sz,Dz)
 752 
 753                 if actions is not None:
 754                     actions = actions.to(device, non_blocking=True)
 755 
 756                 B = z1.shape[0]
 757                 B_self = int(round(args.self_fraction * B))
 758                 B_self = max(0, min(B - 1, B_self))
 759 
 760                 with autocast(device_type="cuda", enabled=use_amp):
 761                     loss, aux = dynamics_pretrain_loss(
 762                         dyn.module if hasattr(dyn, "module") else dyn,
 763                         z1=z1,
 764                         actions=actions,
 765                         act_mask=act_mask,
 766                         k_max=args.k_max,
 767                         B_self=B_self,
 768                         step=step,
 769                         bootstrap_start=args.bootstrap_start,
 770                         agent_tokens=None,
 771                     )
 772 
 773                 if not torch.isfinite(loss):
 774                     raise RuntimeError(f"Non-finite loss at step {step}: loss={loss}")
 775 
 776                 loss_to_backprop = loss / grad_accum
 777                 scaler.scale(loss_to_backprop).backward()
 778 
 779                 do_step = ((step + 1) % grad_accum == 0)
 780                 if do_step:
 781                     if args.grad_clip > 0:
 782                         scaler.unscale_(opt)
 783                         torch.nn.utils.clip_grad_norm_(
 784                             (dyn.module if hasattr(dyn, "module") else dyn).parameters(),
 785                             max_norm=args.grad_clip,
 786                         )
 787 
 788                     if use_amp:
 789                         scaler.step(opt)
 790                         scaler.update()
 791                     else:
 792                         opt.step()
 793                     opt.zero_grad(set_to_none=True)
 794 
 795                 # Evaluation / visualization
 796                 if is_rank0() and args.eval_every > 0 and (step % args.eval_every == 0):
 797 
 798                     # Evaluate on a small slice of the current batch
 799                     B_eval = min(frames.shape[0], args.eval_batch_size)
 800                     frames_eval = frames[:B_eval]
 801 
 802                     if args.use_actions:
 803                         actions_eval = actions[:B_eval]
 804                         act_mask_eval = act_mask[:B_eval]
 805                     else:
 806                         actions_eval = None
 807                         act_mask_eval = None
 808 
 809                     sched = make_tau_schedule(k_max=args.k_max, schedule=args.eval_schedule, d=args.eval_d)
 810 
 811                     run_dynamics_eval(
 812                         encoder=encoder,
 813                         decoder=decoder,
 814                         dyn=(dyn.module if hasattr(dyn, "module") else dyn),
 815                         frames=frames_eval,
 816                         actions=actions_eval,
 817                         act_mask=act_mask_eval,
 818                         H=H, W=W, C=C, patch=patch,
 819                         packing_factor=args.packing_factor,
 820                         k_max=args.k_max,
 821                         ctx_length=args.eval_ctx,
 822                         horizon=args.eval_horizon,
 823                         sched=sched,
 824                         max_items=args.eval_max_items,
 825                         step=step,
 826                     )
 827 
 828                 # Logging
 829                 if is_rank0() and (step % args.log_every == 0):
 830                     
 831                     # Action shuffle loss ratio
 832                     if actions is not None:
 833                         with torch.no_grad():
 834                             loss_real, _ = dynamics_pretrain_loss(
 835                                 dyn.module if hasattr(dyn, "module") else dyn,
 836                                 z1=z1,
 837                                 actions=actions,
 838                                 act_mask=act_mask,
 839                                 k_max=args.k_max,
 840                                 B_self=B_self,
 841                                 step=step,
 842                                 bootstrap_start=args.bootstrap_start,
 843                                 agent_tokens=None,
 844                             )
 845                             perm = torch.randperm(actions.shape[0], device=actions.device)
 846                             loss_shuffled, _ = dynamics_pretrain_loss(
 847                                 dyn.module if hasattr(dyn, "module") else dyn,
 848                                 z1=z1,
 849                                 actions=actions[perm],
 850                                 act_mask=act_mask,
 851                                 k_max=args.k_max,
 852                                 B_self=B_self,
 853                                 step=step,
 854                                 bootstrap_start=args.bootstrap_start,
 855                                 agent_tokens=None,
 856                             )
 857                         action_shuffle_loss_ratio = loss_shuffled / loss
 858                     else:
 859                         action_shuffle_loss_ratio = torch.tensor(0., device=device)
 860 
 861                     # Log to wandb
 862                     wandb.log(
 863                         {
 864                             "loss/total": float(loss.item()),
 865                             "loss/flow_mse": float(aux["flow_mse"].item()),
 866                             "loss/bootstrap_mse": float(aux["bootstrap_mse"].item()),
 867                             "loss/loss_emp": float(aux["loss_emp"].item()),
 868                             "loss/loss_self": float(aux["loss_self"].item()),
 869                             "stats/action_shuffle_loss_ratio": float(action_shuffle_loss_ratio.item()),
 870                             "stats/sigma_mean": float(aux["sigma_mean"].item()),
 871                             "stats/B_self": float(B_self),
 872                             "lr": float(opt.param_groups[0]["lr"]),
 873                             "time/hrs": (time.time() - t0) / 3600.0,
 874                         },
 875                         step=step,
 876                     )
 877 
 878                     # Log to console
 879                     print(
 880                         f"step {step:07d} | loss={loss.item():.6f} "
 881                         f"| flow_mse={aux['flow_mse'].item():.6f} "
 882                         f"| boot_mse={aux['bootstrap_mse'].item():.6f} "
 883                         f"| sigma={aux['sigma_mean'].item():.3f} | B_self={B_self}"
 884                     )
 885 
 886                 # Checkpointing
 887                 if is_rank0() and args.save_every > 0 and (step % args.save_every == 0) and do_step:
 888                     ckpt_path = ckpt_dir / f"step_{step:07d}.pt"
 889                     save_ckpt(ckpt_path, step=step, epoch=epoch, dyn_model=dyn, opt=opt, scaler=scaler, args=args)
 890                     latest = ckpt_dir / "latest.pt"
 891                     save_ckpt(latest, step=step, epoch=epoch, dyn_model=dyn, opt=opt, scaler=scaler, args=args)
 892 
 893                 step += 1
 894 
 895             start_epoch = epoch + 1
 896 
 897     if ddp:
 898         dist.barrier()
 899         dist.destroy_process_group()
 900
