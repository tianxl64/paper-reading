  25 class TokenLayout:
  26     n_latents: int
  27     segments: Tuple[Tuple[Modality, int], ...]
  28 
  29     def S(self) -> int:
  30         return self.n_latents + sum(n for _, n in self.segments)
  31 
  32     def modality_ids(self) -> torch.Tensor:
  33         parts = []
  34         if self.n_latents > 0:
  35             parts.append(torch.full((self.n_latents,), int(Modality.LATENT), dtype=torch.int32))
  36         for m, n in self.segments:
  37             if n > 0:
  38                 parts.append(torch.full((n,), int(m), dtype=torch.int32))
  39         return torch.cat(parts, dim=0) if parts else torch.zeros((0,), dtype=torch.int32)
  40 
  41     def slices(self) -> Dict[Modality, slice]:
  42         idx = 0
  43         out: Dict[Modality, slice] = {}
  44         if self.n_latents > 0:
  45             out[Modality.LATENT] = slice(idx, idx + self.n_latents)
  46             idx += self.n_latents
  47         for m, n in self.segments:
  48             if n > 0 and m not in out:
  49                 out[m] = slice(idx, idx + n)
  50             idx += n
  51         return out

  54 def temporal_patchify(videos_btchw: torch.Tensor, patch: int) -> torch.Tensor:
  55     """
  56     videos: (B,T,C,H,W) float in [0,1]
  57     returns: (B,T,Np,Dp) where Dp = patch*patch*C and Np = (H/patch)*(W/patch)
  58     """
  59     assert videos_btchw.dim() == 5
  60     B, T, C, H, W = videos_btchw.shape
  61     assert H % patch == 0 and W % patch == 0
  62     x = videos_btchw.reshape(B * T, C, H, W)
  63     cols = F.unfold(x, kernel_size=patch, stride=patch)          # (BT, C*pp, Np)
  64     cols = cols.transpose(1, 2).contiguous()                     # (BT, Np, Dp)
  65     Np, Dp = cols.shape[1], cols.shape[2]
  66     return cols.reshape(B, T, Np, Dp)

  81 def sinusoid_table(n: int, d: int, base: float = 10000.0, device=None) -> torch.Tensor:
  82     # fp32 by construction
  83     pos = torch.arange(n, device=device, dtype=torch.float32).unsqueeze(1)  # (n,1)
  84     i   = torch.arange(d, device=device, dtype=torch.float32).unsqueeze(0)  # (1,d)
  85     k   = torch.floor(i / 2.0)
  86     # stable: exp(log(base) * exponent)
  87     div = torch.exp(-(2.0 * k) / max(1.0, float(d)) * math.log(base))
  88     ang = pos * div
  89     return torch.where((i % 2) == 0, torch.sin(ang), torch.cos(ang))  # (n,d) fp32

  92 def add_sinusoidal_positions(tokens_btSd: torch.Tensor, scale_pos_embeds) -> torch.Tensor:
  93     B, T, S, D = tokens_btSd.shape
  94     device = tokens_btSd.device
  95     pos_t = sinusoid_table(T, D, device=device)  # fp32
  96     pos_s = sinusoid_table(S, D, device=device)  # fp32
  97     if scale_pos_embeds:
  98         pos = (pos_t[None, :, None, :] + pos_s[None, None, :, :]) * (1.0 / math.sqrt(D))
  99     else:
 100         pos = (pos_t[None, :, None, :] + pos_s[None, None, :, :])
 101 
 102     return tokens_btSd + pos.to(dtype=tokens_btSd.dtype)

 105 class MAEReplacer(nn.Module):
 106     def __init__(self, d_model: int, p_min: float = 0.0, p_max: float = 0.9):
 107         super().__init__()
 108         self.p_min = float(p_min)
 109         self.p_max = float(p_max)
 110         self.mask_token = nn.Parameter(torch.empty(d_model))
 111         nn.init.normal_(self.mask_token, std=0.02)
 112 
 113     def forward(self, patches_btnd: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
 114         """
 115         patches: (B,T,Np,D)
 116         returns:
 117           replaced: (B,T,Np,D)
 118           mae_mask: (B,T,Np,1) bool, True where masked (must reconstruct)
 119           keep_prob:(B,T,1) float
 120         """
 121         B, T, Np, D = patches_btnd.shape
 122         device = patches_btnd.device
 123 
 124         # fast path: deterministic "no MAE"
 125         if self.p_min == 0.0 and self.p_max == 0.0:
 126             keep_prob = torch.ones((B, T, 1), device=device, dtype=patches_btnd.dtype)
 127             mae_mask = torch.zeros((B, T, Np, 1), device=device, dtype=torch.bool)
 128             return patches_btnd, mae_mask, keep_prob
 129 
 130         p_bt = torch.empty((B, T), device=device).uniform_(self.p_min, self.p_max)
 131         keep_prob = (1.0 - p_bt).unsqueeze(-1)                          # (B,T,1)
 132         keep = (torch.rand((B, T, Np), device=device) < keep_prob)      # (B,T,Np)
 133         keep_ = keep.unsqueeze(-1)
 134         mask_tok = self.mask_token.to(dtype=patches_btnd.dtype)
 135         replaced = torch.where(keep_, patches_btnd, mask_tok.view(1, 1, 1, D))
 136         mae_mask = (~keep_).to(torch.bool)
 137         return replaced, mae_mask, keep_prob

 140 class RMSNorm(nn.Module):
 141     def __init__(self, d: int, eps: float = 1e-6):
 142         super().__init__()
 143         self.eps = eps
 144         self.scale = nn.Parameter(torch.ones(d))
 145 
 146     def forward(self, x: torch.Tensor) -> torch.Tensor:
 147         var = x.pow(2).mean(dim=-1, keepdim=True)
 148         return x * (self.scale / torch.sqrt(var + self.eps))

 151 class MLP(nn.Module):
 152     def __init__(self, d_model: int, mlp_ratio: float = 4.0, dropout: float = 0.0):
 153         super().__init__()
 154         hidden = int(d_model * mlp_ratio)
 155         self.fc_in = nn.Linear(d_model, 2 * hidden)
 156         self.fc_out = nn.Linear(hidden, d_model)
 157         self.drop = nn.Dropout(dropout)
 158 
 159     def forward(self, x: torch.Tensor) -> torch.Tensor:
 160         u, v = self.fc_in(x).chunk(2, dim=-1)
 161         h = u * F.silu(v)
 162         h = self.drop(h)
 163         y = self.fc_out(h)
 164         y = self.drop(y)
 165         return y

 168 class MultiheadSelfAttention(nn.Module):
 169     def __init__(self, d_model: int, n_heads: int, dropout: float = 0.0):
 170         super().__init__()
 171         assert d_model % n_heads == 0
 172         self.d_model = d_model
 173         self.n_heads = n_heads
 174         self.head_dim = d_model // n_heads
 175         self.dropout_p = float(dropout)
 176 
 177         self.qkv = nn.Linear(d_model, 3 * d_model, bias=True)
 178         self.out = nn.Linear(d_model, d_model, bias=True)
 179 
 180     def forward(self, x_nld: torch.Tensor, *, attn_mask: Optional[torch.Tensor] = None, is_causal: bool = False):
 181         """
 182         x: (N,L,D)
 183         attn_mask: bool, True means "allowed to attend" (for torch SDPA), broadcastable to (N,1,L,L) or (N,H,L,L)
 184         """
 185         N, L, D = x_nld.shape
 186         q, k, v = self.qkv(x_nld).chunk(3, dim=-1)
 187 
 188         q = q.view(N, L, self.n_heads, self.head_dim).transpose(1, 2)
 189         k = k.view(N, L, self.n_heads, self.head_dim).transpose(1, 2)
 190         v = v.view(N, L, self.n_heads, self.head_dim).transpose(1, 2)
 191 
 192         drop = self.dropout_p if self.training else 0.0
 193         y = F.scaled_dot_product_attention(q, k, v, attn_mask=attn_mask, dropout_p=drop, is_causal=is_causal)
 194         y = y.transpose(1, 2).contiguous().view(N, L, D)
 195         return self.out(y)

 198 class SpaceSelfAttentionModality(nn.Module):
 199     def __init__(self, d_model: int, n_heads: int, modality_ids: torch.Tensor, n_latents: int, mode: str, dropout: float):
 200         super().__init__()
 201         self.n_latents = int(n_latents)
 202         self.mode = mode
 203         self.register_buffer("modality_ids", modality_ids.to(torch.int32), persistent=False)
 204 
 205         S = int(self.modality_ids.numel())
 206         allow = self._build_allow(S)                               # (S,S) True=allowed
 207         attn_mask = allow.unsqueeze(0).unsqueeze(0)                # (1,1,S,S) True=allowed (PyTorch SDPA bool mask)
 208         self.register_buffer("attn_mask", attn_mask, persistent=False)
 209 
 210         self.attn = MultiheadSelfAttention(d_model, n_heads, dropout=dropout)
 211 
 212     def _build_allow(self, S: int) -> torch.Tensor:
 213         device = self.modality_ids.device
 214         q_idx = torch.arange(S, device=device).unsqueeze(1)  # (S,1)
 215         k_idx = torch.arange(S, device=device).unsqueeze(0)  # (1,S)
 216 
 217         is_q_lat = q_idx < self.n_latents
 218         is_k_lat = k_idx < self.n_latents
 219 
 220         q_mod = self.modality_ids[q_idx]
 221         k_mod = self.modality_ids[k_idx]
 222         same_mod = (q_mod == k_mod)
 223 
 224         if self.mode == "encoder":
 225             allow_lat_q = torch.ones((S, S), dtype=torch.bool, device=device)
 226             allow_nonlat_q = same_mod
 227             return torch.where(is_q_lat, allow_lat_q, allow_nonlat_q)
 228 
 229         if self.mode == "decoder":
 230             allow_lat_q = is_k_lat
 231             allow_nonlat_q = same_mod | is_k_lat
 232             return torch.where(is_q_lat, allow_lat_q, allow_nonlat_q)
 233 
 234         if self.mode == "wm_agent":
 235             # full mixing across modalities
 236             return torch.ones((S, S), dtype=torch.bool, device=device)
 237 
 238         if self.mode == "wm_agent_isolated":
 239             # non-agent tokens: can attend to everything EXCEPT agent tokens
 240             # agent tokens: attend only to agent tokens (keeps them inert in pretrain)
 241             is_q_agent = (q_mod == int(Modality.AGENT))
 242             is_k_agent = (k_mod == int(Modality.AGENT))
 243 
 244             allow = torch.ones((S, S), dtype=torch.bool, device=device)
 245 
 246             # non-agent queries cannot see agent keys
 247             allow_non_agent_q = ~is_q_agent
 248             allow = torch.where(allow_non_agent_q, ~is_k_agent, allow)
 249 
 250             # agent queries only see agent keys
 251             allow = torch.where(is_q_agent, is_k_agent, allow)
 252             return allow
 253 
 254         raise ValueError(f"Unsupported mode for tokenizer/wm: {self.mode}")
 255 
 256     def forward(self, x_btSd: torch.Tensor) -> torch.Tensor:
 257         B, T, S, D = x_btSd.shape
 258         x = x_btSd.reshape(B * T, S, D)
 259         mask = self.attn_mask.expand(B * T, 1, S, S)
 260         y = self.attn(x, attn_mask=mask, is_causal=False)
 261         return y.reshape(B, T, S, D)

 264 class TimeSelfAttention(nn.Module):
 265     def __init__(self, d_model: int, n_heads: int, dropout: float, latents_only: bool, n_latents: int):
 266         super().__init__()
 267         self.latents_only = bool(latents_only)
 268         self.n_latents = int(n_latents)
 269         self.attn = MultiheadSelfAttention(d_model, n_heads, dropout=dropout)
 270 
 271     def forward(self, x_btSd: torch.Tensor) -> torch.Tensor:
 272         B, T, S, D = x_btSd.shape
 273         if self.latents_only:
 274             L = self.n_latents
 275             lat = x_btSd[:, :, :L, :]  # (B,T,L,D)
 276             lat_nld = lat.permute(0, 2, 1, 3).contiguous().view(B * L, T, D)
 277             out = self.attn(lat_nld, is_causal=True)
 278             out = out.view(B, L, T, D).permute(0, 2, 1, 3).contiguous()
 279             x = x_btSd.clone()
 280             x[:, :, :L, :] = out
 281             return x
 282         else:
 283             x_nld = x_btSd.permute(0, 2, 1, 3).contiguous().view(B * S, T, D)
 284             out = self.attn(x_nld, is_causal=True)
 285             return out.view(B, S, T, D).permute(0, 2, 1, 3).contiguous()

 288 class BlockCausalLayer(nn.Module):
 289     def __init__(
 290         self,
 291         d_model: int,
 292         n_heads: int,
 293         n_latents: int,
 294         modality_ids: torch.Tensor,
 295         space_mode: str,
 296         dropout: float,
 297         mlp_ratio: float,
 298         layer_index: int,
 299         time_every: int,
 300         latents_only_time: bool,
 301     ):
 302         super().__init__()
 303         self.do_time = ((layer_index + 1) % time_every == 0)
 304 
 305         self.norm1 = RMSNorm(d_model)
 306         self.space = SpaceSelfAttentionModality(d_model, n_heads, modality_ids, n_latents, space_mode, dropout)
 307         self.drop1 = nn.Dropout(dropout)
 308 
 309         if self.do_time:
 310             self.norm2 = RMSNorm(d_model)
 311             self.time = TimeSelfAttention(d_model, n_heads, dropout, latents_only_time, n_latents)
 312             self.drop2 = nn.Dropout(dropout)
 313 
 314         self.norm3 = RMSNorm(d_model)
 315         self.mlp = MLP(d_model, mlp_ratio=mlp_ratio, dropout=dropout)
 316 
 317     def forward(self, x: torch.Tensor) -> torch.Tensor:
 318         x = x + self.drop1(self.space(self.norm1(x)))
 319         if self.do_time:
 320             x = x + self.drop2(self.time(self.norm2(x)))
 321         x = x + self.mlp(self.norm3(x))
 322         return x

 325 class BlockCausalTransformer(nn.Module):
 326     def __init__(
 327         self,
 328         d_model: int,
 329         n_heads: int,
 330         depth: int,
 331         n_latents: int,
 332         modality_ids: torch.Tensor,
 333         space_mode: str,
 334         dropout: float,
 335         mlp_ratio: float,
 336         time_every: int,
 337         latents_only_time: bool,
 338     ):
 339         super().__init__()
 340         self.layers = nn.ModuleList([
 341             BlockCausalLayer(
 342                 d_model=d_model, n_heads=n_heads, n_latents=n_latents,
 343                 modality_ids=modality_ids, space_mode=space_mode,
 344                 dropout=dropout, mlp_ratio=mlp_ratio,
 345                 layer_index=i, time_every=time_every,
 346                 latents_only_time=latents_only_time,
 347             )
 348             for i in range(depth)
 349         ])
 350 
 351     def forward(self, x: torch.Tensor) -> torch.Tensor:
 352         for layer in self.layers:
 353             x = layer(x)
 354         return x

 357 class Encoder(nn.Module):
 358     def __init__(
 359         self,
 360         *,
 361         patch_dim: int,
 362         d_model: int,
 363         n_latents: int,
 364         n_patches: int,
 365         n_heads: int,
 366         depth: int,
 367         d_bottleneck: int,
 368         dropout: float = 0.0,
 369         mlp_ratio: float = 4.0,
 370         time_every: int = 4,
 371         latents_only_time: bool = True,
 372         mae_p_min: float = 0.0,
 373         mae_p_max: float = 0.9,
 374         scale_pos_embeds: bool = True,
 375     ):
 376         super().__init__()
 377         self.d_model = d_model
 378         self.n_latents = n_latents
 379         self.n_patches = n_patches
 380         self.scale_pos_embeds = scale_pos_embeds
 381 
 382         self.patch_proj = nn.Linear(patch_dim, d_model)
 383         self.bottleneck_proj = nn.Linear(d_model, d_bottleneck)
 384 
 385         self.layout = TokenLayout(n_latents=n_latents, segments=((Modality.IMAGE, n_patches),))
 386         modality_ids = self.layout.modality_ids()  # CPU buffer, moves with .to(device)
 387 
 388         self.transformer = BlockCausalTransformer(
 389             d_model=d_model, n_heads=n_heads, depth=depth,
 390             n_latents=n_latents, modality_ids=modality_ids,
 391             space_mode="encoder",
 392             dropout=dropout, mlp_ratio=mlp_ratio,
 393             time_every=time_every, latents_only_time=latents_only_time,
 394         )
 395         self.mae = MAEReplacer(d_model=d_model, p_min=mae_p_min, p_max=mae_p_max)
 396 
 397         self.latents = nn.Parameter(torch.empty(n_latents, d_model))
 398         nn.init.normal_(self.latents, std=0.02)
 399 
 400     def forward(self, patch_tokens_btnd: torch.Tensor):
 401         B, T, Np, Dp = patch_tokens_btnd.shape
 402         assert Np == self.n_patches
 403 
 404         proj = self.patch_proj(patch_tokens_btnd)            # (B,T,Np,D)
 405         proj_masked, mae_mask, keep_prob = self.mae(proj)    # (B,T,Np,D), (B,T,Np,1), (B,T,1)
 406 
 407         lat = self.latents.view(1, 1, self.n_latents, -1).expand(B, T, -1, -1)
 408         tokens = torch.cat([lat, proj_masked], dim=2)        # (B,T,S,D)
 409         tokens = add_sinusoidal_positions(tokens, self.scale_pos_embeds)
 410 
 411         enc = self.transformer(tokens)
 412         z = torch.tanh(self.bottleneck_proj(enc[:, :, :self.n_latents, :]))
 413         return z, (mae_mask, keep_prob)

 416 class Decoder(nn.Module):
 417     def __init__(
 418         self,
 419         *,
 420         d_bottleneck: int,
 421         d_model: int,
 422         n_heads: int,
 423         depth: int,
 424         n_latents: int,
 425         n_patches: int,
 426         d_patch: int,
 427         dropout: float = 0.0,
 428         mlp_ratio: float = 4.0,
 429         time_every: int = 4,
 430         latents_only_time: bool = True,
 431         scale_pos_embeds: bool = True,
 432     ):
 433         super().__init__()
 434         self.n_latents = n_latents
 435         self.n_patches = n_patches
 436         self.scale_pos_embeds = scale_pos_embeds
 437 
 438         self.up_proj = nn.Linear(d_bottleneck, d_model)
 439         self.patch_queries = nn.Parameter(torch.empty(n_patches, d_model))
 440         nn.init.normal_(self.patch_queries, std=0.02)
 441 
 442         self.patch_head = nn.Linear(d_model, d_patch)
 443 
 444         self.layout = TokenLayout(n_latents=n_latents, segments=((Modality.IMAGE, n_patches),))
 445         modality_ids = self.layout.modality_ids()
 446 
 447         self.transformer = BlockCausalTransformer(
 448             d_model=d_model, n_heads=n_heads, depth=depth,
 449             n_latents=n_latents, modality_ids=modality_ids,
 450             space_mode="decoder",
 451             dropout=dropout, mlp_ratio=mlp_ratio,
 452             time_every=time_every, latents_only_time=latents_only_time,
 453         )
 454 
 455     def forward(self, z_btLd: torch.Tensor) -> torch.Tensor:
 456         B, T, L, _ = z_btLd.shape
 457         assert L == self.n_latents
 458 
 459         lat = torch.tanh(self.up_proj(z_btLd))                                 # (B,T,L,D)
 460         qry = self.patch_queries.view(1, 1, self.n_patches, -1).expand(B, T, -1, -1)
 461         tokens = torch.cat([lat, qry], dim=2)                                  # (B,T,S,D)
 462         tokens = add_sinusoidal_positions(tokens, self.scale_pos_embeds)
 463 
 464         x = self.transformer(tokens)
 465         x_p = x[:, :, L:, :]
 466         return torch.sigmoid(self.patch_head(x_p))                             # (B,T,Np,Dp)

 481 def pack_bottleneck_to_spatial(z_btLd: torch.Tensor, *, n_spatial: int, k: int) -> torch.Tensor:
 482     """
 483     z: (B,T,L,D_b) where L == n_spatial * k
 484     -> (B,T,n_spatial,D_b*k)
 485     """
 486     B, T, L, D = z_btLd.shape
 487     assert L == n_spatial * k, f"L={L} must equal n_spatial*k={n_spatial*k}"
 488     return z_btLd.view(B, T, n_spatial, k * D)

 491 def unpack_spatial_to_bottleneck(z_btSd: torch.Tensor, *, k: int) -> torch.Tensor:
 492     """
 493     z: (B,T,n_spatial,D_b*k) -> (B,T,n_spatial*k,D_b)
 494     """
 495     B, T, S, DK = z_btSd.shape
 496     assert DK % k == 0, f"D={DK} must be divisible by k={k}"
 497     D = DK // k
 498     return z_btSd.view(B, T, S * k, D)

 501 class ActionEncoder(nn.Module):
 502     """
 503     Continuous actions in [-1,1], shape (B,T,A) -> token (B,T,1,D).
 504     If actions is None (unlabeled pretrain), emits a learned base token.
 505     """
 506     def __init__(self, d_model: int, action_dim: int = 16, hidden_mult: float = 2.0):
 507         super().__init__()
 508         self.d_model = int(d_model)
 509         self.action_dim = int(action_dim)
 510 
 511         hidden = int(self.d_model * hidden_mult)
 512         self.base = nn.Parameter(torch.empty(self.d_model))
 513         nn.init.normal_(self.base, std=0.02)
 514 
 515         self.fc1 = nn.Linear(self.action_dim, hidden)
 516         self.fc2 = nn.Linear(hidden, self.d_model)
 517 
 518         nn.init.normal_(self.fc2.weight, std=1e-3)
 519         nn.init.zeros_(self.fc2.bias)
 520 
 521     def forward(
 522         self,
 523         actions: Optional[torch.Tensor],                 # (B,T,A) or None
 524         *,
 525         batch_time_shape: Optional[Tuple[int,int]] = None,
 526         act_mask: Optional[torch.Tensor] = None,         # (B,T,A) or (A,)
 527         as_tokens: bool = True,
 528     ) -> torch.Tensor:
 529         if actions is None:
 530             assert batch_time_shape is not None
 531             B, T = batch_time_shape
 532             out = self.base.view(1, 1, -1).expand(B, T, -1)
 533         else:
 534             x = actions
 535             if act_mask is not None:
 536                 x = x * act_mask
 537             x = x.clamp(-1, 1)
 538             out = self.fc2(F.silu(self.fc1(x))) + self.base.view(1, 1, -1)
 539 
 540         return out[:, :, None, :] if as_tokens else out

 543 class TaskEmbedder(nn.Module):
 544     def __init__(
 545         self,
 546         d_model: int,
 547         n_agent: int = 1,
 548         use_ids: bool = True,
 549         n_tasks: int = 128,
 550         d_task: int = 64,
 551     ):
 552         super().__init__()
 553         self.d_model = int(d_model)
 554         self.n_agent = int(n_agent)
 555         self.use_ids = bool(use_ids)
 556         self.n_tasks = int(n_tasks)
 557         self.d_task = int(d_task)
 558 
 559         if self.use_ids:
 560             self.task_table = nn.Embedding(self.n_tasks, self.d_model)
 561         else:
 562             self.task_proj = nn.Linear(self.d_task, self.d_model)
 563 
 564         self.agent_base = nn.Parameter(torch.empty(self.d_model))
 565         nn.init.normal_(self.agent_base, std=0.02)
 566 
 567     def forward(self, task: torch.Tensor, *, B: int, T: int) -> torch.Tensor:
 568         if self.use_ids:
 569             emb = self.task_table(task.to(torch.long))  # (B,D)
 570         else:
 571             emb = self.task_proj(task)                  # (B,D)
 572 
 573         x = emb + self.agent_base.view(1, -1)          # (B,D)
 574         return x[:, None, None, :].expand(B, T, self.n_agent, self.d_model)

 577 class Dynamics(nn.Module):
 578     def __init__(
 579         self,
 580         *,
 581         d_model: int,
 582         d_bottleneck: int,
 583         d_spatial: int,
 584         n_spatial: int,
 585         n_register: int,
 586         n_agent: int,
 587         n_heads: int,
 588         depth: int,
 589         k_max: int,
 590         dropout: float = 0.0,
 591         mlp_ratio: float = 4.0,
 592         time_every: int = 4,
 593         space_mode: str = "wm_agent_isolated",  # or "wm_agent"
 594         scale_pos_embeds: bool = True,
 595     ):
 596         super().__init__()
 597         assert d_spatial % d_bottleneck == 0, "expected packing: d_spatial = d_bottleneck * packing_factor"
 598         self.d_model = int(d_model)
 599         self.d_spatial = int(d_spatial)
 600         self.n_spatial = int(n_spatial)
 601         self.n_register = int(n_register)
 602         self.n_agent = int(n_agent)
 603         self.k_max = int(k_max)
 604         self.scale_pos_embeds = scale_pos_embeds
 605 
 606         self.spatial_proj = nn.Linear(self.d_spatial, self.d_model)
 607         self.register_tokens = nn.Parameter(torch.empty(self.n_register, self.d_model))
 608         nn.init.normal_(self.register_tokens, std=0.02)
 609 
 610         self.action_encoder = ActionEncoder(d_model=self.d_model, action_dim=16)
 611 
 612         # shortcut conditioning
 613         self.num_step_bins = int(math.log2(self.k_max)) + 1
 614         self.step_embed = nn.Embedding(self.num_step_bins, self.d_model)
 615         self.signal_embed = nn.Embedding(self.k_max + 1, self.d_model)
 616 
 617         segments = [
 618             (Modality.ACTION, 1),
 619             (Modality.SHORTCUT_SIGNAL, 1),
 620             (Modality.SHORTCUT_STEP, 1),
 621             (Modality.SPATIAL, self.n_spatial),
 622             (Modality.REGISTER, self.n_register),
 623         ]
 624         if self.n_agent > 0:
 625             segments.append((Modality.AGENT, self.n_agent))
 626 
 627         self.layout = TokenLayout(n_latents=0, segments=tuple(segments))
 628         sl = self.layout.slices()
 629         self.spatial_slice = sl[Modality.SPATIAL]
 630         self.agent_slice = sl.get(Modality.AGENT, slice(0, 0))
 631         modality_ids = self.layout.modality_ids()
 632 
 633         self.transformer = BlockCausalTransformer(
 634             d_model=self.d_model,
 635             n_heads=int(n_heads),
 636             depth=int(depth),
 637             n_latents=0,
 638             modality_ids=modality_ids,
 639             space_mode=space_mode,
 640             dropout=float(dropout),
 641             mlp_ratio=float(mlp_ratio),
 642             time_every=int(time_every),
 643             latents_only_time=False,
 644         )
 645 
 646         self.flow_x_head = nn.Linear(self.d_model, self.d_spatial)
 647         nn.init.zeros_(self.flow_x_head.weight)
 648         nn.init.zeros_(self.flow_x_head.bias)
 649 
 650     def forward(
 651         self,
 652         actions: Optional[torch.Tensor],          # (B,T,16) or None
 653         step_idxs: torch.Tensor,                  # (B,T)
 654         signal_idxs: torch.Tensor,                # (B,T)
 655         packed_enc_tokens: torch.Tensor,          # (B,T,n_spatial,d_spatial)
 656         *,
 657         act_mask: Optional[torch.Tensor] = None,  # (B,T,16) or (16,) or None
 658         agent_tokens: Optional[torch.Tensor] = None,
 659     ) -> Tuple[torch.Tensor, Optional[torch.Tensor]]:
 660         B, T = packed_enc_tokens.shape[:2]
 661 
 662         spatial_tokens = self.spatial_proj(packed_enc_tokens)  # (B,T,n_spatial,d_model)
 663 
 664         action_tokens = self.action_encoder(
 665             actions,
 666             batch_time_shape=(B, T),
 667             act_mask=act_mask,
 668             as_tokens=True,
 669         )  # (B,T,1,d_model)
 670 
 671         reg = self.register_tokens.view(1, 1, self.n_register, self.d_model).expand(B, T, -1, -1)
 672 
 673         step_tok = self.step_embed(step_idxs.to(torch.long))[:, :, None, :]
 674         sig_tok = self.signal_embed(signal_idxs.to(torch.long))[:, :, None, :]
 675 
 676         if self.n_agent > 0:
 677             if agent_tokens is None:
 678                 agent_tokens = torch.zeros((B, T, self.n_agent, self.d_model), device=spatial_tokens.device, dtype=spatial_tokens.dtype)
 679             toks = [action_tokens, sig_tok, step_tok, spatial_tokens, reg, agent_tokens]
 680         else:
 681             toks = [action_tokens, sig_tok, step_tok, spatial_tokens, reg]
 682 
 683         tokens = torch.cat(toks, dim=2)  # (B,T,S,D)
 684         tokens = add_sinusoidal_positions(tokens, self.scale_pos_embeds)
 685         x = self.transformer(tokens)
 686 
 687         spatial_out = x[:, :, self.spatial_slice, :]
 688         x1_hat = self.flow_x_head(spatial_out)  # (B,T,n_spatial,d_spatial)
 689 
 690         h_t = None
 691         if self.n_agent > 0:
 692             h_t = x[:, :, self.agent_slice, :]   # (B,T,n_agent,d_model)
 693 
 694         return x1_hat, h_t

 697 def recon_loss_from_mae(pred_btnd: torch.Tensor,
 698                         target_btnd: torch.Tensor,
 699                         mae_mask_btNp1: torch.Tensor) -> torch.Tensor:
 700     # mask: (B,T,Np,1) bool, True where masked
 701     mask = mae_mask_btNp1.to(dtype=torch.float32)  # (B,T,Np,1)
 702 
 703     # compute in fp32 to avoid fp16 overflow on reduction
 704     diff = (pred_btnd.float() - target_btnd.float())          # (B,T,Np,Dp)
 705     sq = diff.mul(diff) * mask                                # broadcast mask over Dp
 706     denom = mask.sum().clamp_min(1.0) * diff.shape[-1]        # (#masked patches) * Dp
 707     return sq.sum() / denom

 710 def lpips_on_mae_recon(
 711     lpips_fn,
 712     pred_btnd: torch.Tensor,
 713     target_btnd: torch.Tensor,
 714     mae_mask_btNp1: torch.Tensor,
 715     *,
 716     H: int, W: int, C: int, patch: int,
 717     subsample_frac: float = 1.0,
 718 ) -> torch.Tensor:
 719     recon_masked_btnd = torch.where(mae_mask_btNp1, pred_btnd, target_btnd)
 720     recon = temporal_unpatchify(recon_masked_btnd.float(), H, W, C, patch)
 721     tgt   = temporal_unpatchify(target_btnd.float(),        H, W, C, patch)
 722 
 723     if subsample_frac < 1.0:
 724         B, T = recon.shape[:2]
 725         step = max(1, int(1.0 / subsample_frac))
 726         recon = recon[:, ::step]
 727         tgt   = tgt[:, ::step]
 728 
 729     recon = (recon.clamp(0, 1) * 2.0 - 1.0).float()
 730     tgt   = (tgt.clamp(0, 1)   * 2.0 - 1.0).float()
 731 
 732     B, T = recon.shape[:2]
 733     recon = recon.reshape(B * T, C, H, W)
 734     tgt   = tgt.reshape(B * T, C, H, W)
 735 
 736     with torch.autocast(device_type="cuda", enabled=False):
 737         lp = lpips_fn(recon, tgt)
 738     return lp.mean()
