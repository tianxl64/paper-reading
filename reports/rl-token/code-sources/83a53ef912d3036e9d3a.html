factory.py

1 """Dynamic factories for pluggable environments and intervention managers.
2 
3 Users provide a Python import path to a factory function.  The same
4 pattern is used for both the environment and the intervention manager.
5 
6 Example usage::
7 
8     env = make_env(
9         "rlt_openpi.envs.franka.env_factory.make_franka_env",
10         action_dim=7,
11         chunk_length=10,
12         task_prompt="pick up the cup",
13     )
14 
15     intervention = make_intervention(
16         "rlt_openpi.envs.franka.intervention.make_vr_intervention",
17         env=env,
18     )
19 """
20 
21 from __future__ import annotations
22 
23 import importlib
24 import logging
25 import sys
26 from pathlib import Path
27 from typing import Any
28 
29 from rlt_openpi.rollout.intervention import InterventionManager
30 
31 logger = logging.getLogger(__name__)
32 
33 
34 def _ensure_project_root_on_path() -> None:
35     project_root = str(Path(__file__).resolve().parents[3])
36     if project_root not in sys.path:
37         sys.path.insert(0, project_root)
38 
39 
40 def _import_factory(import_path: str) -> Any:
41     """Dynamically import a factory function from a dotted path."""
42     _ensure_project_root_on_path()
43     module_path, func_name = import_path.rsplit(".", 1)
44     module = importlib.import_module(module_path)
45     return getattr(module, func_name)
46 
47 
48 def make_env(env_factory: str, **kwargs: Any) -> Any:
49     """Import and call a user-provided env factory function.
50 
51     Args:
52         env_factory: Dotted Python import path to a callable, e.g.
53             ``"my_package.envs.make_franka_env"``.
54         **kwargs: Forwarded to the factory function (typically
55             ``action_dim``, ``chunk_length``, ``task_prompt``).
56 
57     Returns:
58         An env object with ``reset()``, ``step(action_chunk)``,
59         ``action_dim``, and ``chunk_length``.
60     """
61     factory_fn = _import_factory(env_factory)
62     logger.info("Creating env via %s", env_factory)
63     return factory_fn(**kwargs)
64 
65 
66 def make_intervention(
67     intervention_factory: str, *, env: Any, **kwargs: Any
68 ) -> InterventionManager:
69     """Import and call a user-provided intervention factory function.
70 
71     Args:
72         intervention_factory: Dotted Python import path to a callable,
73             e.g. ``"rlt_openpi.envs.franka.intervention.make_vr_intervention"``.
74         env: The environment object (passed as the first argument to the
75             factory so it can access robot handles).
76         **kwargs: Forwarded to the factory function.
77 
78     Returns:
79         An ``InterventionManager`` instance.
80     """
81     factory_fn = _import_factory(intervention_factory)
82     logger.info("Creating intervention manager via %s", intervention_factory)
83     return factory_fn(env=env, **kwargs)
