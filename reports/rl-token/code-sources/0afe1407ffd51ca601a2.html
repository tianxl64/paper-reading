actor.py

1 """Actor network with VLA reference action conditioning.
2 
3 The actor takes the RL state x = cat(z_rl, s^p) and a VLA reference action
4 chunk a_tilde, applies reference dropout during training, and outputs an
5 action chunk mu (plus optional Gaussian exploration noise).
6 """
7 
8 import torch
9 from torch import Tensor, nn
10 
11 from rlt_openpi.models.networks import MLP
12 
13 
14 class Actor(nn.Module):
15     """Actor with reference action dropout and exploration noise.
16 
17     Input: cat(x, a_tilde_masked) where a_tilde is zeroed for a fraction
18     of the batch during training (ref_dropout probability).
19 
20     Output: mu + N(0, sigma^2) during training, mu during eval.
21 
22     Args:
23         state_dim: Dimension of RL state (z_rl + s^p).
24         action_chunk_dim: Dimension of flattened action chunk (C * d).
25         hidden_dim: MLP hidden layer width.
26         num_hidden_layers: Number of MLP hidden layers.
27         sigma: Exploration noise std (applied during training only).
28         ref_dropout: Probability of zeroing reference actions during training.
29     """
30 
31     def __init__(
32         self,
33         state_dim: int,
34         action_chunk_dim: int,
35         hidden_dim: int = 256,
36         num_hidden_layers: int = 2,
37         sigma: float = 0.1,
38         ref_dropout: float = 0.5,
39     ) -> None:
40         super().__init__()
41         self.action_chunk_dim = action_chunk_dim
42         self.sigma = sigma
43         self.ref_dropout = ref_dropout
44 
45         self.mlp = MLP(
46             input_dim=state_dim + action_chunk_dim,
47             output_dim=action_chunk_dim,
48             hidden_dim=hidden_dim,
49             num_hidden_layers=num_hidden_layers,
50         )
51 
52         # Zero-init the last linear layer so the residual starts at zero,
53         # meaning the actor initially reproduces the VLA reference exactly.
54         last_linear = [m for m in self.mlp.net if isinstance(m, nn.Linear)][-1]
55         nn.init.zeros_(last_linear.weight)
56         nn.init.zeros_(last_linear.bias)
57 
58     def forward(self, x: Tensor, a_tilde: Tensor) -> Tensor:
59         """Compute action chunk as VLA reference + learned residual.
60 
61         Args:
62             x: RL state [B, state_dim].
63             a_tilde: Flattened VLA reference action chunk [B, action_chunk_dim].
64 
65         Returns:
66             Action chunk [B, action_chunk_dim], clamped to [-1, 1].
67             During training: a_tilde + residual + noise, with ref dropout.
68             During eval: a_tilde + residual, with full a_tilde.
69         """
70         a_tilde_input = self._apply_ref_dropout(a_tilde)
71         residual = self.mlp(torch.cat([x, a_tilde_input], dim=-1))
72         mu = a_tilde + residual
73 
74         if self.training:
75             noise = torch.randn_like(mu) * self.sigma
76             return (mu + noise).clamp(-1.0, 1.0)
77         return mu.clamp(-1.0, 1.0)
78 
79     def _apply_ref_dropout(self, a_tilde: Tensor) -> Tensor:
80         """Zero out reference actions for a fraction of the batch during training."""
81         if not self.training or self.ref_dropout == 0.0:
82             return a_tilde
83 
84         B = a_tilde.shape[0]
85         # Per-sample dropout mask: 1 = keep, 0 = drop
86         keep_mask = torch.rand(B, 1, device=a_tilde.device) >= self.ref_dropout
87         return a_tilde * keep_mask
