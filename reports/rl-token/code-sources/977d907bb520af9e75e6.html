train_online_rl.py

1 """Stage 2: Online RL training with frozen VLA + RL token (Algorithm 1).
2 
3 Usage:
4     uv run python scripts/train_online_rl.py --help
5     uv run python scripts/train_online_rl.py --vla-config-name pi0_aloha_sim \
6         --vla-checkpoint-dir /path/to/vla.safetensors \
7         --rl-token-checkpoint /path/to/rl_token.pt
8 """
9 
10 from __future__ import annotations
11 
12 import logging
13 
14 import torch
15 import tyro
16 
17 from rlt_openpi.rollout.factory import make_env, make_intervention
18 from rlt_openpi.rollout.intervention import InterventionManager
19 from rlt_openpi.training.config import OnlineRLTrainConfig
20 from rlt_openpi.training.online_rl_trainer import OnlineRLTrainer
21 from rlt_openpi.utils.checkpoint import load_rl_token_model
22 from rlt_openpi.utils.logging import Logger
23 from rlt_openpi.vla.vla_wrapper import VLAWrapper
24 
25 logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(name)s] %(message)s")
26 log = logging.getLogger(__name__)
27 
28 
29 def main(config: OnlineRLTrainConfig) -> None:
30     """Run online RL training (Stage 2, Algorithm 1)."""
31     log.info("Stage 2 config: %s", config)
32 
33     # Set up logger
34     rl_logger = Logger.from_train_config(config)
35 
36     # Load frozen VLA
37     log.info("Loading VLA: config=%s, checkpoint=%s", config.vla_config_name, config.vla_checkpoint_dir)
38     vla = VLAWrapper(
39         checkpoint_path=config.vla_checkpoint_dir,
40         config_name=config.vla_config_name,
41         device="cuda",
42     )
43 
44     # Load frozen RL token model from Stage 1
45     log.info("Loading RL token model from %s", config.rl_token_checkpoint)
46     rl_token_model = load_rl_token_model(config.rl_token_checkpoint, device="cuda")
47 
48     # Restore fine-tuned VLA weights from Stage 1 checkpoint (if available).
49     # Load to CPU first to avoid OOM — the VLA + RL token already occupy most VRAM.
50     stage1_ckpt = torch.load(config.rl_token_checkpoint, map_location="cpu", weights_only=False)
51     if "vla_model" in stage1_ckpt:
52         vla.extractor.pi0.load_state_dict(stage1_ckpt["vla_model"])
53         log.info("Restored fine-tuned VLA weights from Stage 1 checkpoint")
54     else:
55         log.warning("No fine-tuned VLA weights found in Stage 1 checkpoint; using base VLA")
56     del stage1_ckpt
57     torch.cuda.empty_cache()
58 
59     # Create trainer
60     trainer = OnlineRLTrainer(
61         config=config,
62         vla=vla,
63         rl_token_model=rl_token_model,
64         device="cuda",
65     )
66 
67     # Resume from checkpoint if provided
68     if config.resume_checkpoint:
69         log.info("Resuming from checkpoint: %s", config.resume_checkpoint)
70         trainer.load(config.resume_checkpoint)
71 
72     # Create environment via pluggable factory.
73     # Pass --env-factory to specify a Python import path, e.g.:
74     #   --env-factory rlt_openpi.envs.franka.env_factory.make_franka_env
75     #   --env-factory rlt_openpi.rollout.sim_env.make_sim_env
76     if not config.env_factory:
77         log.error("--env-factory is required. Provide a Python import path to an env factory function.")
78         raise SystemExit(1)
79 
80     env = make_env(
81         config.env_factory,
82         action_dim=config.action_dim,
83         chunk_length=config.chunk_length,
84         task_prompt=config.task_prompt,
85         max_episode_chunks=config.max_episode_chunks,
86     )
87     log.info("Environment created: action_dim=%d, chunk_length=%d", env.action_dim, env.chunk_length)
88 
89     # Create intervention manager (VR teleoperation, etc.) if specified.
90     intervention_mgr: InterventionManager | None = None
91     if config.intervention_factory:
92         intervention_mgr = make_intervention(config.intervention_factory, env=env)
93         log.info("Intervention manager created via %s", config.intervention_factory)
94 
95     trainer.train(env=env, intervention_mgr=intervention_mgr, log_fn=rl_logger.log)
96 
97     rl_logger.finish()
98 
99 
100 if __name__ == "__main__":
101     main(tyro.cli(OnlineRLTrainConfig))
