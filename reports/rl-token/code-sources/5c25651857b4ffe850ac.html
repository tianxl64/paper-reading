evaluate.py

1 """Evaluate a trained model on an environment.
2 
3 Supports two modes:
4   - **stage1**: VLA-only evaluation (fine-tuned VLA from Stage 1, no actor).
5   - **stage2**: Full pipeline (VLA + RL token + actor from Stage 2).
6 
7 The mode is auto-detected: if --checkpoint (Stage 2) is provided, runs
8 stage2 eval; otherwise runs stage1 VLA-only eval.
9 
10 Usage:
11     # Stage 1: evaluate fine-tuned VLA
12     uv run python scripts/evaluate.py \
13         --vla-checkpoint-dir checkpoints/pi05_droid_pytorch/model.safetensors \
14         --stage1-checkpoint checkpoints/rl_token/rl_token_step5000.pt \
15         --env-factory rlt_openpi.envs.franka.env_factory.make_franka_env \
16         --task-prompt "stack the three blocks on the tray" \
17         --num-episodes 50
18 
19     # Stage 2: evaluate full trained model
20     uv run python scripts/evaluate.py \
21         --checkpoint checkpoints/online_rl/run_latest/online_rl_ep100.pt \
22         --vla-checkpoint-dir checkpoints/pi05_droid_pytorch/model.safetensors \
23         --rl-token-checkpoint checkpoints/rl_token/rl_token_step5000.pt \
24         --env-factory rlt_openpi.envs.franka.env_factory.make_franka_env \
25         --task-prompt "stack the three blocks on the tray" \
26         --num-episodes 50
27 """

53 @dataclass
54 class EvalConfig:
55     """Evaluation configuration."""
56 
57     # Stage 2 checkpoint (if provided, runs full pipeline eval)
58     checkpoint: str = ""
59     # Stage 1 checkpoint with fine-tuned VLA weights (if provided without
60     # --checkpoint, runs VLA-only eval)
61     stage1_checkpoint: str = ""
62 
63     vla_config_name: str = "pi05_droid_finetune"
64     vla_checkpoint_dir: str = ""
65     rl_token_checkpoint: str = ""
66     env_factory: str = ""
67     task_prompt: str = ""
68     action_dim: int = 8
69     chunk_length: int = 10
70     num_episodes: int = 50
71     save_dir: str = ""
72     device: str = "cuda"
73 
74 
75 def _run(config: EvalConfig) -> None:
76     """Evaluate full pipeline: VLA + RL token + actor."""
77     ckpt = torch.load(config.checkpoint, map_location=config.device, weights_only=False)
78     train_config: OnlineRLTrainConfig = ckpt["config"]
79 
80     vla = VLAWrapper(
81         checkpoint_path=config.vla_checkpoint_dir,
82         config_name=config.vla_config_name,
83         device=config.device,
84     )
85 
86     rl_token_model = load_rl_token_model(config.rl_token_checkpoint, device=config.device)
87     rl_token_model.eval()
88 
89     actor = Actor(
90         state_dim=train_config.state_dim,
91         action_chunk_dim=train_config.action_chunk_dim,
92         hidden_dim=train_config.mlp_hidden_dim,
93         num_hidden_layers=train_config.mlp_num_hidden_layers,
94         sigma=train_config.actor_noise_sigma,
95         ref_dropout=0.0,
96     ).to(config.device)
97     actor.load_state_dict(ckpt["actor"])
98     actor.eval()
99 
100     log.info(
101         "Actor loaded from %s (episode %d, %d env steps)",
102         config.checkpoint,
103         ckpt["total_episodes"],
104         ckpt["total_env_steps"],
105     )
106 
107     env = make_env(
108         config.env_factory,
109         action_dim=train_config.action_dim,
110         chunk_length=train_config.chunk_length,
111         task_prompt=config.task_prompt,
112     )
113     log.info("Environment created: action_dim=%d, chunk_length=%d", env.action_dim, env.chunk_length)
114 
115     buf = ReplayBuffer(1, train_config.state_dim, train_config.action_chunk_dim, train_config.chunk_length)
116     worker = RolloutWorker(
117         env, vla, rl_token_model, actor, buf,
118         InterventionManager(), train_config.chunk_length,
119         train_config.action_dim, config.device,
120     )
121 
122     episodes = []
123     for ep in range(config.num_episodes):
124         stats = worker.collect_episode(store_transitions=False)
125         success = stats.extra.get("success", False)
126         episodes.append({
127             "episode": ep,
128             "reward": stats.total_reward,
129             "success": success,
130             "num_chunks": stats.num_chunks,
131             "num_steps": stats.num_steps,
132         })
133         log.info("Episode %d: reward=%.3f, success=%s", ep, stats.total_reward, success)
134 
135     return episodes, {
136         "mode": "stage2",
137         "checkpoint": config.checkpoint,
138         "train_episodes": ckpt["total_episodes"],
139         "train_env_steps": ckpt["total_env_steps"],
140     }
141 
142 
143 def _run_vla(config: EvalConfig) -> None:
144     """Evaluate VLA-only (with optional Stage 1 fine-tuned weights)."""
145     vla = VLAWrapper(
146         checkpoint_path=config.vla_checkpoint_dir,
147         config_name=config.vla_config_name,
148         device=config.device,
149     )
150 
151     if config.stage1_checkpoint:
152         ckpt = torch.load(config.stage1_checkpoint, map_location="cpu", weights_only=False)
153         if "vla_model" in ckpt:
154             vla.extractor.pi0.load_state_dict(ckpt["vla_model"])
155             log.info("Loaded fine-tuned VLA weights from %s", config.stage1_checkpoint)
156         else:
157             log.warning("No vla_model key in %s; using base VLA weights", config.stage1_checkpoint)
158         del ckpt
159 
160     env = make_env(
161         config.env_factory,
162         action_dim=config.action_dim,
163         chunk_length=config.chunk_length,
164         task_prompt=config.task_prompt,
165     )
166     log.info("Environment created: action_dim=%d, chunk_length=%d", env.action_dim, env.chunk_length)
167 
168     episodes = []
169     for ep in range(config.num_episodes):
170         obs = env.reset()
171         episode_reward = 0.0
172         episode_chunks = 0
173 
174         while True:
175             with torch.no_grad():
176                 vla_input = vla.preprocess_obs(obs)
177                 action_chunk = vla.get_rl_chunk_reference(vla_input, config.chunk_length)
178                 action_chunk = action_chunk.squeeze(0).cpu().numpy()
179 
180             next_obs, chunk_rewards, done, info = env.step(action_chunk)
181             episode_reward += float(chunk_rewards.sum())
182             episode_chunks += 1
183 
184             if done:
185                 success = info.get("success", False)
186                 episodes.append({
187                     "episode": ep,
188                     "reward": episode_reward,
189                     "success": success,
190                     "num_chunks": episode_chunks,
191                     "num_steps": info.get("steps_executed", episode_chunks * config.chunk_length),
192                 })
193                 log.info(
194                     "Episode %d/%d: chunks=%d, reward=%.3f, success=%s",
195                     ep + 1, config.num_episodes, episode_chunks, episode_reward, success,
196                 )
197                 break
198 
199             obs = next_obs
200 
201     return episodes, {
202         "mode": "stage1",
203         "stage1_checkpoint": config.stage1_checkpoint,
204     }
205 
206 
207 def main(config: EvalConfig) -> None:
208     """Run evaluation (auto-detects stage1 vs stage2)."""
209     log.info("Eval config: %s", config)
210 
211     if not config.env_factory:
212         log.error("--env-factory is required.")
213         raise SystemExit(1)
214 
215     if config.checkpoint:
216         log.info("Full eval: VLA + RL token + actor")
217         episodes, meta = _run(config)
218     else:
219         log.info("VLA-only eval")
220         episodes, meta = _run_vla(config)
221 
222     num_success = sum(e["success"] for e in episodes)
223     success_rate = num_success / len(episodes) if episodes else 0.0
224     mean_reward = sum(e["reward"] for e in episodes) / len(episodes) if episodes else 0.0
225 
226     log.info("Success rate: %.1f%% (%d/%d)", 100 * success_rate, num_success, len(episodes))
227     log.info("Mean reward: %.3f", mean_reward)
228 
229     results = {
230         **meta,
231         "vla_checkpoint_dir": config.vla_checkpoint_dir,
232         "vla_config_name": config.vla_config_name,
233         "eval_timestamp": datetime.now().isoformat(),
234         "num_episodes": len(episodes),
235         "success_rate": success_rate,
236         "mean_reward": mean_reward,
237         "episodes": episodes,
238     }
239 
240     if config.save_dir:
241         save_dir = Path(config.save_dir)
242     elif config.checkpoint:
243         save_dir = Path(config.checkpoint).parent
244     else:
245         save_dir = Path("results") / meta["mode"]
246     save_dir.mkdir(parents=True, exist_ok=True)
247 
248     results_path = save_dir / f"eval_{meta['mode']}_{len(episodes)}ep_{datetime.now().strftime('%Y%m%d_%H%M%S')}.json"
249     results_path.write_text(json.dumps(results, indent=2))
250     log.info("Results saved to %s", results_path)
251 
252 
253 if __name__ == "__main__":
254     main(tyro.cli(EvalConfig))
