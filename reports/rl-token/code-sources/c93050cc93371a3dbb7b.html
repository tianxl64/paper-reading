online_rl_trainer.py

1 """Stage 2 trainer: Online RL with frozen VLA + RL token (Algorithm 1).
2 
3 Implements the full online RL loop from the paper:
4 1. Warmup: fill replay buffer with VLA-only rollouts
5 2. Main loop: collect episode → update critic/actor (UTD ratio G)
6 3. TD3-style updates: twin Q-critics, delayed actor, Polyak targets
7 """
8

32 class OnlineRLTrainer:
33     """Stage 2: Online RL training with Algorithm 1.
34 
35     Loads frozen VLA + frozen RL token model, creates Actor + TwinQCritic +
36     optimizers + ReplayBuffer + RolloutWorker, and runs the online RL loop.
37 
38     Args:
39         config: Stage 2 training hyperparameters.
40         vla: Frozen VLA wrapper (pre-loaded).
41         rl_token_model: Frozen RL token model (loaded from Stage 1 checkpoint).
42         device: Torch device for training.
43     """
44 
45     def __init__(
46         self,
47         config: OnlineRLTrainConfig,
48         vla: VLAWrapper,
49         rl_token_model: RLTokenModel,
50         device: torch.device | str = "cuda",
51     ) -> None:
52         self.config = config
53         self.device = torch.device(device)
54 
55         # Frozen components
56         self.vla = vla
57         self.rl_token_model = rl_token_model
58         self.rl_token_model.eval()
59         for param in self.rl_token_model.parameters():
60             param.requires_grad_(False)
61 
62         # Trainable actor
63         self.actor = Actor(
64             state_dim=config.state_dim,
65             action_chunk_dim=config.action_chunk_dim,
66             hidden_dim=config.mlp_hidden_dim,
67             num_hidden_layers=config.mlp_num_hidden_layers,
68             sigma=config.actor_noise_sigma,
69             ref_dropout=config.ref_action_dropout,
70         ).to(self.device)
71 
72         # Trainable twin Q-critic
73         self.critic = TwinQCritic(
74             state_dim=config.state_dim,
75             action_chunk_dim=config.action_chunk_dim,
76             hidden_dim=config.mlp_hidden_dim,
77             num_hidden_layers=config.mlp_num_hidden_layers,
78         ).to(self.device)
79 
80         # Optimizers
81         self.actor_optimizer = torch.optim.Adam(self.actor.parameters(), lr=config.actor_lr)
82         self.critic_optimizer = torch.optim.Adam(self.critic.parameters(), lr=config.critic_lr)
83 
84         # Replay buffer
85         self.replay_buffer = ReplayBuffer(
86             capacity=config.buffer_capacity,
87             state_dim=config.state_dim,
88             action_chunk_dim=config.action_chunk_dim,
89             chunk_length=config.chunk_length,
90         )
91 
92         # Counters
93         self._total_env_steps = 0
94         self._total_updates = 0
95         self._total_episodes = 0
96 
97     def _create_rollout_worker(
98         self,
99         env: Any,
100         intervention_mgr: InterventionManager | None = None,
101     ) -> RolloutWorker:
102         """Create a rollout worker wired to this trainer's components."""
103         return RolloutWorker(
104             env=env,
105             vla=self.vla,
106             rl_token_model=self.rl_token_model,
107             actor=self.actor,
108             replay_buffer=self.replay_buffer,
109             intervention_mgr=intervention_mgr or InterventionManager(),
110             chunk_length=self.config.chunk_length,
111             action_dim=self.config.action_dim,
112             device=self.device,
113         )
114 
115     def _update_step(self, update_idx: int) -> dict[str, float]:
116         """Run one TD3 update step.
117 
118         Critic is updated every call.  Actor is updated every
119         ``critic_updates_per_actor`` calls.
120 
121         Args:
122             update_idx: 0-based index within the current UTD batch.
123 
124         Returns:
125             Dict of logged metrics.
126         """
127         cfg = self.config
128         batch = self.replay_buffer.sample(batch_size=cfg.batch_size, device=str(self.device))
129 
130         x = batch["x"]
131         a = batch["a"]
132         a_tilde = batch["a_tilde"]
133         rewards = batch["rewards"]
134         next_x = batch["next_x"]
135         dones = batch["dones"]
136 
137         # --- Critic update (every step) ---
138         # Compute TD target
139         # For next_a_tilde, we use a_tilde from the batch as an approximation
140         # (the true next reference would require a VLA call, which is expensive)
141         td_target = compute_td_target(
142             rewards=rewards,
143             dones=dones,
144             next_x=next_x,
145             next_a_tilde=a_tilde,  # approximate next reference
146             actor=self.actor,
147             critic=self.critic,
148             gamma=cfg.gamma,
149             chunk_length=cfg.chunk_length,
150             target_noise_sigma=cfg.target_noise_sigma,
151             target_noise_clip=cfg.target_noise_clip,
152         )
153 
154         q1, q2 = self.critic(x, a)
155         c_loss = critic_loss(q1, q2, td_target)
156 
157         self.critic_optimizer.zero_grad()
158         c_loss.backward()
159         self.critic_optimizer.step()
160 
161         metrics: dict[str, float] = {
162             "critic_loss": c_loss.item(),
163             "q1_mean": q1.mean().item(),
164             "q2_mean": q2.mean().item(),
165         }
166 
167         # --- Actor update (delayed) ---
168         if update_idx % cfg.critic_updates_per_actor == 0:
169             self.actor.train()
170             a_actor = self.actor(x, a_tilde)
171             q_value = self.critic.q_min(x, a_actor)
172             a_loss = actor_loss(q_value, a_actor, a_tilde, cfg.bc_regularizer_beta)
173 
174             self.actor_optimizer.zero_grad()
175             a_loss.backward()
176             self.actor_optimizer.step()
177 
178             metrics["actor_loss"] = a_loss.item()
179 
180         # --- Polyak target update ---
181         self.critic.update_targets(cfg.tau)
182 
183         self._total_updates += 1
184         return metrics
185 
186     def train(
187         self,
188         env: Any,
189         intervention_mgr: InterventionManager | None = None,
190         log_fn: Any | None = None,
191     ) -> None:
192         """Run the full online RL training loop (Algorithm 1).
193 
194         Args:
195             env: Chunk-level environment wrapper.
196             intervention_mgr: Optional human intervention manager.
197             log_fn: Optional callable ``log_fn(metrics_dict)`` for logging.
198         """
199         cfg = self.config
200         worker = self._create_rollout_worker(env, intervention_mgr)
201         train_display = display.TrainingDisplay(window_size=20)
202         train_start = time.time()
203 
204         display.training_start({
205             "Task": cfg.task_prompt or "(not set)",
206             "Max env steps": f"{cfg.max_env_steps:,}",
207             "UTD ratio": str(cfg.utd_ratio),
208             "Chunk length": str(cfg.chunk_length),
209             "Action dim": str(cfg.action_dim),
210             "Run name": cfg.run_name,
211         })
212 
213         # Phase 1: Warmup with VLA-only policy (skip if buffer already has data)
214         if self.replay_buffer.size > 0:
215             logger.info(
216                 "Skipping warmup — replay buffer already has %d transitions (resumed from checkpoint)",
217                 self.replay_buffer.size,
218             )
219         elif cfg.warmup_buffer:
220             self._load_warmup_buffer(cfg.warmup_buffer)
221         else:
222             display.warmup_start(cfg.warmup_steps)
223             stored = 0
224             obs = env.reset()
225             for i in range(cfg.warmup_steps):
226                 action_chunk = worker._get_warmup_action(obs)
227                 x, a_tilde_flat = worker._extract_rl_state(obs)
228                 a_flat = action_chunk.reshape(-1)
229                 next_obs, rewards, done, _info = env.step(action_chunk)
230                 next_x, _ = worker._extract_rl_state(next_obs)
231                 self.replay_buffer.add(
232                     x=x, a=a_flat, a_tilde=a_tilde_flat,
233                     rewards=rewards, next_x=next_x, done=float(done),
234                 )
235                 stored += 1
236                 self._total_env_steps += cfg.chunk_length
237                 display.warmup_progress(i + 1, cfg.warmup_steps)
238                 if done:
239                     obs = env.reset()
240                 else:
241                     obs = next_obs
242             display.warmup_done(stored, self.replay_buffer.size)
243             self._save_warmup_buffer()
244 
245         # Phase 2: Online RL loop
246         # Inject episode counter into env so RobotEnv.reset() can display it
247         if hasattr(env, '_display_episode_num'):
248             env._display_episode_num = self._total_episodes + 1
249 
250         while self._total_env_steps < cfg.max_env_steps:
251             if hasattr(env, '_display_episode_num'):
252                 env._display_episode_num = self._total_episodes + 1
253 
254             self.actor.eval()
255             stats = worker.collect_episode()
256             self._total_episodes += 1
257             self._total_env_steps += stats.num_steps
258 
259             success = stats.extra.get("success", False)
260             train_display.record_episode(success, stats.total_reward)
261 
262             display.episode_result(
263                 episode_num=self._total_episodes,
264                 total_reward=stats.total_reward,
265                 success=success,
266                 num_chunks=stats.num_chunks,
267                 num_steps=stats.num_steps,
268                 interventions=stats.interventions,
269             )
270 
271             episode_metrics = {
272                 "episode_reward": stats.total_reward,
273                 "episode_success": int(success),
274                 "episode_chunks": stats.num_chunks,
275                 "episode_steps": stats.num_steps,
276                 "episode_interventions": stats.interventions,
277                 "total_env_steps": self._total_env_steps,
278                 "total_episodes": self._total_episodes,
279                 "buffer_size": self.replay_buffer.size,
280             }
281 
282             # Update actor and critic (UTD ratio G)
283             update_metrics: dict[str, float] = {}
284             for g in range(cfg.utd_ratio):
285                 step_metrics = self._update_step(g)
286                 update_metrics = step_metrics
287 
288             all_metrics = {**episode_metrics, **update_metrics}
289 
290             train_display.print_summary(
291                 total_episodes=self._total_episodes,
292                 total_env_steps=self._total_env_steps,
293                 max_env_steps=cfg.max_env_steps,
294                 buffer_size=self.replay_buffer.size,
295                 critic_loss=update_metrics.get("critic_loss", 0.0),
296                 actor_loss=update_metrics.get("actor_loss"),
297                 q_mean=update_metrics.get("q1_mean", 0.0),
298             )
299 
300             if log_fn is not None:
301                 log_fn(all_metrics)
302 
303             if self._total_episodes % cfg.save_every == 0:
304                 ckpt_path = self.save()
305                 display.checkpoint_saved(str(ckpt_path))
306 
307         # Final save
308         ckpt_path = self.save()
309         display.checkpoint_saved(str(ckpt_path))
310         display.training_done(
311             self._total_episodes,
312             self._total_env_steps,
313             self._total_updates,
314             time.time() - train_start,
315         )
316 
317     def _save_warmup_buffer(self) -> Path:
318         """Save the replay buffer as a standalone file after warmup."""
319         save_dir = Path(self.config.save_dir) / self.config.run_name
320         save_dir.mkdir(parents=True, exist_ok=True)
321         buf_path = save_dir / "warmup_buffer.pt"
322         torch.save(self.replay_buffer.state_dict(), buf_path)
323         logger.info("Saved warmup buffer (%d transitions) to %s", self.replay_buffer.size, buf_path)
324         return buf_path
325 
326     def _load_warmup_buffer(self, path: str) -> None:
327         """Load a standalone warmup buffer file into the replay buffer."""
328         buf_state = torch.load(path, map_location="cpu", weights_only=False)
329         self.replay_buffer.load_state_dict(buf_state)
330         logger.info("Loaded warmup buffer (%d transitions) from %s", self.replay_buffer.size, path)
331 
332     def save(self, path: str | None = None, save_buffer: bool = True) -> Path:
333         """Save actor, critic, optimizer, and replay buffer states.
334 
335         Args:
336             path: Override save path. Defaults to config.save_dir.
337             save_buffer: Whether to include replay buffer in the checkpoint.
338                 Disable for eval-only checkpoints to save disk space.
339 
340         Returns:
341             Path to the saved checkpoint.
342         """
343         save_dir = Path(path or self.config.save_dir) / self.config.run_name
344         save_dir.mkdir(parents=True, exist_ok=True)
345         ckpt_path = save_dir / f"online_rl_ep{self._total_episodes}.pt"
346         payload: dict[str, Any] = {
347             "actor": self.actor.state_dict(),
348             "critic": self.critic.state_dict(),
349             "actor_optimizer": self.actor_optimizer.state_dict(),
350             "critic_optimizer": self.critic_optimizer.state_dict(),
351             "total_env_steps": self._total_env_steps,
352             "total_updates": self._total_updates,
353             "total_episodes": self._total_episodes,
354             "config": self.config,
355         }
356         if save_buffer:
357             payload["replay_buffer"] = self.replay_buffer.state_dict()
358         torch.save(payload, ckpt_path)
359         logger.info("Saved checkpoint to %s (buffer=%s)", ckpt_path, save_buffer)
360         return ckpt_path
361 
362     def load(self, ckpt_path: str) -> None:
363         """Load actor, critic, optimizer, and replay buffer from checkpoint.
364 
365         Args:
366             ckpt_path: Path to a saved checkpoint file.
367         """
368         ckpt = torch.load(ckpt_path, map_location=self.device, weights_only=False)
369         self.actor.load_state_dict(ckpt["actor"])
370         self.critic.load_state_dict(ckpt["critic"])
371         self.actor_optimizer.load_state_dict(ckpt["actor_optimizer"])
372         self.critic_optimizer.load_state_dict(ckpt["critic_optimizer"])
373         self._total_env_steps = ckpt["total_env_steps"]
374         self._total_updates = ckpt["total_updates"]
375         self._total_episodes = ckpt["total_episodes"]
376 
377         if "replay_buffer" in ckpt:
378             self.replay_buffer.load_state_dict(ckpt["replay_buffer"])
379             logger.info(
380                 "Restored replay buffer (%d transitions)", self.replay_buffer.size
381             )
382 
383         logger.info(
384             "Loaded checkpoint from %s (episode %d, step %d)",
385             ckpt_path,
386             self._total_episodes,
387             self._total_env_steps,
388         )
