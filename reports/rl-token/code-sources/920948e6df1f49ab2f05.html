robot_env.py

1 """Real-robot environment for online RL training.
2 
3 Provides ``RobotEnv``, a chunk-level environment that connects to any
4 robot through three user-supplied callables (``step_fn``, ``reset_fn``,
5 ``get_obs_fn``).  No dependency on any specific robot stack (DROID,
6 polymetis, ROS, etc.) — the wiring happens in the user's launch script.
7 
8 Human reward (success/failure) is collected via
9 :class:`~rlt_openpi.rollout.human_reward.HumanReward` using instant
10 keypress detection (no Enter needed) during episodes.
11 
12 Usage (with DROID)::
13 
14     from droid.robot_env import RobotEnv as DroidEnv
15 
16     droid = DroidEnv(action_space="cartesian_velocity", control_hz=15)
17 
18     def get_obs():
19         obs = droid.get_observation()
20         return {
21             "observation/joint_position": np.array(
22                 obs["robot_state"]["joint_positions"], dtype=np.float32,
23             ),
24             "observation/gripper_position": np.array(
25                 [obs["robot_state"]["gripper_position"]], dtype=np.float32,
26             ),
27             "observation/exterior_image_1_left": obs["image"]["39790647_left"],
28             "observation/wrist_image_left": obs["image"]["15850436_left"],
29             "observation/exterior_image_2_left": obs["image"]["35840217_left"],
30             "prompt": "stack the three blocks on the tray",
31         }
32 
33     env = RobotEnv(
34         step_fn=droid.step,
35         reset_fn=droid.reset,
36         get_obs_fn=get_obs,
37         action_dim=7,
38         chunk_length=10,
39         control_hz=15,
40     )
41 """
42 
43 from __future__ import annotations
44 
45 import logging
46 import time
47 from typing import Any, Callable
48 
49 import numpy as np
50 from numpy.typing import NDArray
51 
52 from rlt_openpi.rollout.reward import HumanReward
53 from rlt_openpi.utils import display
54 
55 logger = logging.getLogger(__name__)
56 
57 
58 class RobotEnv:
59     """Chunk-level environment for real robot online RL.
60 
61     Robot-agnostic: connects to any robot through three callables.
62     Provides the same interface as ``SimEnv`` (``reset``, ``step``,
63     ``action_dim``, ``chunk_length``) so it works with ``RolloutWorker``
64     and ``OnlineRLTrainer``.
65 
66     Args:
67         step_fn: Callable that sends a single action ``[action_dim]`` to
68             the robot.  Signature: ``step_fn(action: np.ndarray) -> Any``.
69         reset_fn: Callable that resets the robot to a home pose.
70             Signature: ``reset_fn() -> Any``.
71         get_obs_fn: Callable that returns an observation dict with at
72             least a ``"state"`` key (proprioceptive, ``np.float32``).
73             Camera images and ``"prompt"`` should also be included for
74             VLA embedding extraction.
75             Signature: ``get_obs_fn() -> dict[str, Any]``.
76         action_dim: Dimension of a single-step action (e.g. 7 for
77             cartesian velocity + gripper).
78         chunk_length: C, number of single-step actions per chunk.
79         control_hz: Robot control frequency in Hz.
80         max_episode_chunks: Maximum chunks per episode before forced
81             termination.
82     """
83 
84     def __init__(
85         self,
86         step_fn: Callable[[NDArray], Any],
87         reset_fn: Callable[[], Any],
88         get_obs_fn: Callable[[], dict[str, Any]],
89         action_dim: int = 7,
90         chunk_length: int = 10,
91         control_hz: int = 15,
92         max_episode_chunks: int = 50,
93     ) -> None:
94         self._step_fn = step_fn
95         self._reset_fn = reset_fn
96         self._get_obs_fn = get_obs_fn
97         self._action_dim = action_dim
98         self._chunk_length = chunk_length
99         self._control_period = 1.0 / control_hz
100         self._max_episode_chunks = max_episode_chunks
101 
102         self._chunk_count = 0
103         self._feedback = HumanReward()
104         self._display_episode_num: int = 0
105 
106     @property
107     def action_dim(self) -> int:
108         return self._action_dim
109 
110     @property
111     def chunk_length(self) -> int:
112         return self._chunk_length
113 
114     def reset(self, **kwargs: Any) -> dict[str, Any]:
115         """Reset robot to home position and wait for scene setup.
116 
117         Returns:
118             Observation dict with ``"state"``, camera images, and ``"prompt"``.
119         """
120         self._feedback.stop()
121         self._reset_fn()
122         self._chunk_count = 0
123 
124         display.episode_reset(self._display_episode_num)
125         input("")
126 
127         self._feedback.start()
128         display.episode_running(self._display_episode_num)
129 
130         return self._get_obs_fn()
131 
132     def step(
133         self, action_chunk: NDArray
134     ) -> tuple[dict[str, Any], NDArray, bool, dict[str, Any]]:
135         """Execute C single-step actions on the robot.
136 
137         Args:
138             action_chunk: Actions to execute, shape ``[C, action_dim]``.
139 
140         Returns:
141             next_obs: Observation dict after the last step.
142             rewards: Per-step rewards ``[C]``.  Terminal success gives +1;
143                 progress keypresses (``p``) give +0.5 without ending
144                 the episode.
145             done: Whether the episode ended.
146             info: Contains ``"success"`` key on termination.
147         """
148         C = self._chunk_length
149         rewards = np.zeros(C, dtype=np.float32)
150         done = False
151         info: dict[str, Any] = {}
152 
153         for k in range(C):
154             t_start = time.time()
155 
156             self._step_fn(action_chunk[k])
157 
158             # Check for human signal between steps
159             signal = self._feedback.check()
160             if signal is not None:
161                 if signal == "s":
162                     rewards[k] = 1.0
163                     done = True
164                     info["success"] = True
165                     logger.info("Human signal: SUCCESS")
166                 elif signal == "f":
167                     done = True
168                     info["success"] = False
169                     logger.info("Human signal: FAILURE")
170                 elif signal == "p":
171                     rewards[k] = self._feedback.progress_reward
172                     logger.info("Human signal: PROGRESS (+%.2f)", self._feedback.progress_reward)
173                 if done:
174                     break
175 
176             # Enforce control frequency
177             elapsed = time.time() - t_start
178             sleep_time = self._control_period - elapsed
179             if sleep_time > 0:
180                 time.sleep(sleep_time)
181 
182         info["steps_executed"] = k + 1
183         self._chunk_count += 1
184 
185         # Timeout: force episode end after max chunks
186         if not done and self._chunk_count >= self._max_episode_chunks:
187             done = True
188             info["success"] = False
189             info["timeout"] = True
190             logger.info("Episode timed out after %d chunks", self._chunk_count)
191 
192         # Restore terminal when episode ends
193         if done:
194             self._feedback.stop()
195 
196         obs = self._get_obs_fn()
197         return obs, rewards, done, info
