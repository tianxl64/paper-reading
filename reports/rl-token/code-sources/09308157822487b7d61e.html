1 """Rollout worker for online RL data collection.
2 
3 Orchestrates environment interaction, VLA embedding extraction, RL token
4 encoding, actor inference, and replay buffer storage.  Supports both
5 VLA-only warmup rollouts and full RL episode collection with optional
6 human intervention.
7 """

25 @dataclass
26 class EpisodeStats:
27     """Statistics for a single collected episode."""
28 
29     total_reward: float = 0.0
30     num_chunks: int = 0

36 
37 class RolloutWorker:
38     """Collects environment rollouts for online RL training.
39 
40     During warmup, runs the VLA-only policy and stores transitions.
41     During RL training, uses the actor conditioned on z_rl and VLA
42     reference actions, with optional human intervention override.
43 
44     Args:
45         env: Chunk-level environment wrapper.
46         vla: Frozen VLA wrapper for embeddings and reference actions.
47         rl_token_model: Frozen RL token encoder (Stage 1 output).
48         actor: RL actor network.
49         replay_buffer: Buffer to store transitions.
50         intervention_mgr: Human intervention manager.
51         chunk_length: C, number of steps per action chunk.
52         action_dim: Dimension of a single-step action.
53         device: Torch device for model inference.
54     """
55 
56     def __init__(
57         self,
58         env: Any,
59         vla: VLAWrapper,
60         rl_token_model: RLTokenModel,
61         actor: Actor,
62         replay_buffer: ReplayBuffer,
63         intervention_mgr: InterventionManager,
64         chunk_length: int,
65         action_dim: int,
66         device: torch.device | str = "cuda",
67     ) -> None:
68         self.env = env
69         self.vla = vla
70         self.rl_token_model = rl_token_model
71         self.actor = actor
72         self.replay_buffer = replay_buffer
73         self.intervention_mgr = intervention_mgr
74         self.chunk_length = chunk_length
75         self.action_dim = action_dim
76         self.device = torch.device(device)
77 
78         self._action_chunk_dim = chunk_length * action_dim
79 
80     def _obs_to_vla_input(self, obs: dict[str, Any]) -> Any:
81         """Prepare observation dict for VLA inference.
82 
83         Uses ``VLAWrapper.preprocess_obs`` if available (real VLA), which
84         applies the full OpenPI transform chain and returns an
85         ``Observation``.  Falls back to simple batch-wrapping for tests
86         with a mock VLA.
87         """
88         if hasattr(self.vla, "preprocess_obs"):
89             return self.vla.preprocess_obs(obs)
90 
91         # Fallback: simple batch-wrap (for tests with mock VLA)
92         batched: dict[str, Any] = {}
93         for key, val in obs.items():
94             arr = np.asarray(val)
95             batched[key] = arr[np.newaxis]  # add batch dim
96         return batched
97 
98     @torch.no_grad()
99     def _extract_rl_state(self, obs: dict[str, Any]) -> tuple[NDArray, NDArray]:
100         """Extract RL state x = cat(z_rl, s^p) and VLA reference chunk.
101 
102         Returns:
103             x: RL state [state_dim] as numpy array.
104             a_tilde_flat: Flattened VLA reference chunk [action_chunk_dim] as numpy.
105         """
106         vla_input = self._obs_to_vla_input(obs)
107 
108         # Extract VLA embeddings and encode into z_rl
109         z, pad_mask = self.vla.extract_embeddings(vla_input)
110         z_rl = self.rl_token_model.encode(z, pad_mask)  # [1, D]
111 
112         # Get VLA reference action chunk (first C steps)
113         a_tilde = self.vla.get_rl_chunk_reference(vla_input, self.chunk_length)  # [1, C, action_dim]
114         a_tilde_flat = a_tilde.reshape(1, -1)  # [1, C*d]
115 
116         # Proprioceptive state s^p from the preprocessed VLA observation.
117         # DroidInputs merges joint_pos + gripper into state, then
118         # PadStatesAndActions zero-pads to the VLA's internal width.
119         # Slice to action_dim to drop the padding.
120         s_p = vla_input.state[:, :self.action_dim].to(dtype=torch.float32, device=self.device)  # [1, d]
121 
122         # RL state: x = cat(z_rl, s^p)
123         x = torch.cat([z_rl, s_p], dim=-1)  # [1, state_dim]
124 
125         return (
126             x.squeeze(0).cpu().numpy(),
127             a_tilde_flat.squeeze(0).cpu().numpy(),
128         )
129 
130     @torch.no_grad()
131     def _get_warmup_action(self, obs: dict[str, Any]) -> NDArray:
132         """Get action from VLA-only policy for warmup.
133 
134         Returns:
135             action_chunk: [C, action_dim] numpy array.
136         """
137         vla_input = self._obs_to_vla_input(obs)
138         a_tilde = self.vla.get_rl_chunk_reference(vla_input, self.chunk_length)  # [1, C, action_dim]
139         return a_tilde.squeeze(0).cpu().numpy()  # [C, action_dim]
140 
141     @torch.no_grad()
142     def _get_actor_action(self, x: NDArray, a_tilde_flat: NDArray) -> NDArray:
143         """Get action from the RL actor.
144 
145         Args:
146             x: RL state [state_dim].
147             a_tilde_flat: Flattened VLA reference chunk [action_chunk_dim].
148 
149         Returns:
150             action_chunk: [C, action_dim] numpy array.
151         """
152         x_t = torch.as_tensor(x, dtype=torch.float32, device=self.device).unsqueeze(0)
153         a_tilde_t = torch.as_tensor(a_tilde_flat, dtype=torch.float32, device=self.device).unsqueeze(0)
154 
155         a_flat = self.actor(x_t, a_tilde_t)  # [1, C*d]
156         return a_flat.squeeze(0).cpu().numpy().reshape(self.chunk_length, self.action_dim)
157 
158     def collect_warmup(self, num_chunks: int) -> int:
159         """Run VLA-only policy and store transitions in the replay buffer.
160 
161         Collects ``num_chunks`` chunk-level transitions across potentially
162         multiple episodes (auto-resets on termination).
163 
164         Args:
165             num_chunks: Number of chunk-level transitions to collect.
166 
167         Returns:
168             Total number of transitions stored.
169         """
170         stored = 0
171         obs = self.env.reset()
172 
173         for _ in range(num_chunks):
174             # Get VLA reference action (used as both executed and reference)
175             action_chunk = self._get_warmup_action(obs)  # [C, action_dim]
176 
177             # Build RL state for this observation
178             x, a_tilde_flat = self._extract_rl_state(obs)
179             a_flat = action_chunk.reshape(-1)  # [C*d]
180 
181             # Step environment
182             next_obs, rewards, done, _info = self.env.step(action_chunk)
183 
184             # Build next RL state
185             next_x, _ = self._extract_rl_state(next_obs)
186 
187             # Store transition
188             self.replay_buffer.add(
189                 x=x,
190                 a=a_flat,
191                 a_tilde=a_tilde_flat,
192                 rewards=rewards,
193                 next_x=next_x,
194                 done=float(done),
195             )
196             stored += 1
197 
198             if done:
199                 obs = self.env.reset()
200             else:
201                 obs = next_obs
202 
203         return stored
204 
205     def collect_episode(self, store_transitions: bool = True) -> EpisodeStats:
206         """Collect a single RL episode using the actor policy.
207 
208         At each chunk boundary:
209         1. Extract z_rl and VLA reference actions
210         2. Form RL state x = cat(z_rl, s^p)
211         3. Check for human intervention
212         4. Run actor (or use human action) to get action chunk
213         5. Step environment for C steps
214         6. Store transition in replay buffer (if ``store_transitions``)
215 
216         Args:
217             store_transitions: Whether to add transitions to the replay
218                 buffer.  Set to ``False`` during evaluation to avoid
219                 unnecessary buffer writes.
220 
221         Returns:
222             Episode statistics.
223         """
224         stats = EpisodeStats()
225         obs = self.env.reset()
226 
227         while True:
228             # Extract RL state and VLA reference
229             x, a_tilde_flat = self._extract_rl_state(obs)
230 
231             # Check for human intervention.
232             # If the intervention manager stepped the robot internally
233             # (InterventionResult), we use its outputs directly and skip
234             # env.step().  Otherwise fall through to the actor.
235             intervention: InterventionResult | None = None
236             if self.intervention_mgr.check_intervention():
237                 intervention = self.intervention_mgr.get_human_action(
238                     self.action_dim, self.chunk_length
239                 )
240 
241             if intervention is not None:
242                 action_chunk = intervention.action_chunk
243                 next_obs = intervention.next_obs
244                 rewards = intervention.rewards
245                 done = intervention.done
246                 info = intervention.info
247                 stats.interventions += 1
248             else:
249                 action_chunk = self._get_actor_action(x, a_tilde_flat)
250                 next_obs, rewards, done, info = self.env.step(action_chunk)
251 
252             a_flat = action_chunk.reshape(-1)  # [C*d]
253 
254             if store_transitions:
255                 # Build next RL state (requires VLA forward pass)
256                 next_x, _ = self._extract_rl_state(next_obs)
257 
258                 self.replay_buffer.add(
259                     x=x,
260                     a=a_flat,
261                     a_tilde=a_tilde_flat,
262                     rewards=rewards,
263                     next_x=next_x,
264                     done=float(done),
265                 )
266 
267             # Update stats
268             stats.total_reward += float(rewards.sum())
269             stats.num_chunks += 1
270             # Use env-reported steps if available, else fall back to chunk_length
271             stats.num_steps += info.get("steps_executed", self.chunk_length)
272 
273             if done:
274                 stats.done = True
275                 stats.extra = info
276                 break
277 
278             obs = next_obs
279 
280         return stats
