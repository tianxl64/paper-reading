0025 def create_batch_env(env_ctor, num_envs, isolate_envs):
0026   envs, blockings = zip(*[
0027       create_env(env_ctor, isolate_envs)
0028       for _ in range(num_envs)])
0029   assert all(blocking == blockings[0] for blocking in blockings)
0030   return batch_env.TFBatchEnv(envs, blockings[0])

0033 def create_env(env_ctor, isolate_envs):
0034   if isolate_envs == 'none':
0035     env = env_ctor()
0036     blocking = True
0037   elif isolate_envs == 'thread':
0038     env = wrappers.Async(env_ctor, 'thread')
0039     blocking = False
0040   elif isolate_envs == 'process':
0041     env = wrappers.Async(env_ctor, 'process')
0042     blocking = False
0043   else:
0044     raise NotImplementedError(isolate_envs)
0045   return env, blocking

0048 def simulate(agent, env, episodes=None, steps=None):
0049 
0050   def pred(step, episode, *args, **kwargs):
0051     if episodes is None:
0052       return tf.less(step, steps)
0053     if steps is None:
0054       return tf.less(episode, episodes)
0055     return tf.logical_or(tf.less(episode, episodes), tf.less(step, steps))
0056 
0057   def reset(mask, scores, lengths, previous):
0058     indices = tf.where(mask)[:, 0]
0059     reset_agent = agent.reset(indices)
0060     values = env.reset(indices)
0061     # This would be shorter but gives an internal TensorFlow error.
0062     # previous = {
0063     #     key: tf.tensor_scatter_update(
0064     #         previous[key], indices[:, None], values[key])
0065     #     for key in previous}
0066     idx = tf.cast(indices[:, None], tf.int32)
0067     previous = {
0068         key: tf.where(
0069             mask,
0070             tf.scatter_nd(idx, values[key], tf.shape(previous[key])),
0071             previous[key])
0072         for key in previous}
0073     scores = tf.where(mask, tf.zeros_like(scores), scores)
0074     lengths = tf.where(mask, tf.zeros_like(lengths), lengths)
0075     with tf.control_dependencies([reset_agent]):
0076       return tf.identity(scores), tf.identity(lengths), previous
0077 
0078   def body(step, episode, scores, lengths, previous, ta_out, ta_sco, ta_len):
0079     # Reset episodes and agents if necessary.
0080     reset_mask = tf.cond(
0081         tf.equal(step, 0),
0082         lambda: tf.ones(len(env), tf.bool),
0083         lambda: previous['done'])
0084     reset_mask.set_shape([len(env)])
0085     scores, lengths, previous = tf.cond(
0086         tf.reduce_any(reset_mask),
0087         lambda: reset(reset_mask, scores, lengths, previous),
0088         lambda: (scores, lengths, previous))
0089     step_indices = tf.range(len(env))
0090     action = agent.step(step_indices, previous)
0091     values = env.step(action)
0092     # Update book keeping variables.
0093     done_indices = tf.cast(tf.where(values['done']), tf.int32)
0094     step += tf.shape(step_indices)[0]
0095     episode += tf.shape(done_indices)[0]
0096     scores += values['reward']
0097     lengths += tf.shape(step_indices)[0]
0098     # Write transitions, scores, and lengths to tensor arrays.
0099     ta_out = {
0100         key: array.write(array.size(), values[key])
0101         for key, array in ta_out.items()}
0102     ta_sco = tf.cond(
0103         tf.greater(tf.shape(done_indices)[0], 0),
0104         lambda: ta_sco.write(
0105             ta_sco.size(), tf.gather(scores, done_indices)[:, 0]),
0106         lambda: ta_sco)
0107     ta_len = tf.cond(
0108         tf.greater(tf.shape(done_indices)[0], 0),
0109         lambda: ta_len.write(
0110             ta_len.size(), tf.gather(lengths, done_indices)[:, 0]),
0111         lambda: ta_len)
0112     return step, episode, scores, lengths, values, ta_out, ta_sco, ta_len
0113 
0114   initial = env.reset()
0115   ta_out = {
0116       key: tf.TensorArray(
0117           value.dtype, 0, True, element_shape=value.shape)
0118       for key, value in initial.items()}
0119   ta_sco = tf.TensorArray(
0120       tf.float32, 0, True, element_shape=[None], infer_shape=False)
0121   ta_len = tf.TensorArray(
0122       tf.int32, 0, True, element_shape=[None], infer_shape=False)
0123   zero_scores = tf.zeros(len(env), tf.float32, name='scores')
0124   zero_lengths = tf.zeros(len(env), tf.int32, name='lengths')
0125   ta_out, ta_sco, ta_len = tf.while_loop(
0126       pred, body,
0127       (0, 0, zero_scores, zero_lengths, initial, ta_out, ta_sco, ta_len),
0128       parallel_iterations=1, back_prop=False)[-3:]
0129   transitions = {key: array.stack() for key, array in ta_out.items()}
0130   transitions = {
0131       key: tf.transpose(value, [1, 0] + list(range(2, value.shape.ndims)))
0132       for key, value in transitions.items()}
0133   scores = ta_sco.concat()
0134   lengths = ta_len.concat()
0135   return scores, lengths, transitions
