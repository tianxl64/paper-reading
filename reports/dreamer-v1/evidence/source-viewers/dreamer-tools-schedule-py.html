0023 def binary(step, batch_size, after, every, until):
0024   # https://www.desmos.com/calculator/csbhr5cjzz
0025   offset_step = step - after
0026   phase = tf.less(offset_step % every, batch_size)
0027   active = tf.greater_equal(step, after)
0028   if until > 0:
0029       active = tf.logical_and(active, tf.less(step, until))
0030   result = tf.logical_and(phase, active)
0031   result.set_shape(tf.TensorShape([]))
0032   return result

0035 def linear(step, ramp, min=None, max=None):
0036   # https://www.desmos.com/calculator/nrumhgvxql
0037   if ramp == 0:
0038     result = tf.constant(1, tf.float32)
0039   if ramp > 0:
0040     result = tf.minimum(
0041         tf.cast(step, tf.float32) / tf.cast(ramp, tf.float32), 1)
0042   if ramp < 0:
0043     result = 1 - linear(step, abs(ramp))
0044   if min is not None and max is not None:
0045     assert min <= max
0046   if min is not None:
0047     min = float(min)
0048     assert 0 <= min <= 1
0049     result = tf.maximum(min, result)
0050   if max is not None:
0051     max = float(max)
0052     assert 0 <= min <= 1
0053     result = tf.minimum(result, max)
0054   result.set_shape(tf.TensorShape([]))
0055   return result

0058 def step(step, distance):
0059   # https://www.desmos.com/calculator/sh1zsjtsqg
0060   if distance == 0:
0061     result = tf.constant(1, tf.float32)
0062   if distance > 0:
0063     result = tf.cast(tf.less(step, distance), tf.float32)
0064   if distance < 0:
0065     result = 1 - tf.cast(tf.less(step, abs(distance)), tf.float32)
0066   result.set_shape(tf.TensorShape([]))
0067   return result

0070 def exponential(step, distance, target, min=None, max=None):
0071   # https://www.desmos.com/calculator/egt24luequ
0072   target = tf.constant(target, tf.float32)
0073   step = tf.cast(step, tf.float32)
0074   if distance == 0:
0075     result = tf.constant(1, tf.float32)
0076   elif distance > 0:
0077     distance = tf.constant(abs(distance), tf.float32)
0078     result = tf.exp(tf.log(target) / distance) ** step
0079   elif distance < 0:
0080     distance = tf.constant(abs(distance), tf.float32)
0081     result = 1 - tf.exp(tf.log(1 - target) / distance) ** step
0082   if min is not None and max is not None:
0083     assert min <= max
0084   if min is not None:
0085     min = float(min)
0086     assert 0 <= min <= 1
0087     result = tf.maximum(min, result)
0088   if max is not None:
0089     max = float(max)
0090     assert 0 <= min <= 1
0091     result = tf.minimum(result, max)
0092   result.set_shape(tf.TensorShape([]))
0093   return result

0096 def linear_reset(step, ramp, after, every):
0097   # https://www.desmos.com/calculator/motbnqhacw
0098   assert every > ramp, (every, ramp)  # Would never reach max value.
0099   assert not (every != np.inf and after == np.inf), (every, after)
0100   step, ramp, after, every = [
0101       tf.cast(x, tf.float32) for x in (step, ramp, after, every)]
0102   before = tf.cast(tf.less(step, after), tf.float32) * step
0103   after_mask = tf.cast(tf.greater_equal(step, after), tf.float32)
0104   after = after_mask * ((step - after) % every)
0105   result = tf.minimum((before + after) / ramp, 1)
0106   result.set_shape(tf.TensorShape([]))
0107   return result
