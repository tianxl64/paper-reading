0015 from __future__ import absolute_import
0016 from __future__ import division
0017 from __future__ import print_function
0018 
0019 import tensorflow as tf
0020 
0021 from dreamer.tools import filter_variables_lib
0022 
0023 
0024 class CustomOptimizer(object):
0025 
0026   def __init__(
0027       self, optimizer_cls, step, log, learning_rate,
0028       include=None, exclude=None, clipping=None, schedule=None,
0029       debug=False, name='custom_optimizer'):
0030     if schedule:
0031       learning_rate *= schedule(step)
0032     self._step = step
0033     self._log = log
0034     self._learning_rate = learning_rate
0035     self._variables = filter_variables_lib.filter_variables(include, exclude)
0036     self._clipping = float(clipping)
0037     self._debug = debug
0038     self._name = name
0039     self._optimizer = optimizer_cls(learning_rate=learning_rate, name=name)
0040 
0041   def maybe_minimize(self, condition, loss):
0042     with tf.name_scope('optimizer_{}'.format(self._name)):
0043       # loss = tf.cond(condition, lambda: loss, float)
0044       update_op, grad_norm = tf.cond(
0045           condition,
0046           lambda: self.minimize(loss),
0047           lambda: (tf.no_op(), 0.0))
0048       with tf.control_dependencies([update_op]):
0049         summary = tf.cond(
0050             tf.logical_and(condition, self._log),
0051             lambda: self.summarize(grad_norm), str)
0052       if self._debug:
0053         # print_op = tf.print('{}_grad_norm='.format(self._name), grad_norm)
0054         message = 'Zero gradient norm in {} optimizer.'.format(self._name)
0055         assertion = lambda: tf.assert_greater(grad_norm, 0.0, message=message)
0056         assert_op = tf.cond(condition, assertion, tf.no_op)
0057         with tf.control_dependencies([assert_op]):
0058           summary = tf.identity(summary)
0059       return summary, grad_norm
0060 
0061   def minimize(self, loss):
0062     if self._debug:
0063       loss = tf.check_numerics(loss, '{}_loss'.format(self._name))
0064     if hasattr(self._optimizer, 'compute_gradients'):  # TF 1.0
0065       gradients, variables = zip(*self._optimizer.compute_gradients(
0066           loss, self._variables, colocate_gradients_with_ops=True))
0067     else:  # Keras
0068       gradients = self._optimizer.get_gradients(loss, self._variables)
0069       variables = self._variables
0070     grad_norm = tf.global_norm(gradients)
0071     if self._clipping:
0072       gradients, _ = tf.clip_by_global_norm(
0073           gradients, self._clipping, grad_norm)
0074     optimize = self._optimizer.apply_gradients(zip(gradients, variables))
0075     return optimize, grad_norm
0076 
0077   def summarize(self, grad_norm):
0078     summaries = []
0079     with tf.name_scope('optimizer_{}'.format(self._name)):
0080       summaries.append(tf.summary.scalar('learning_rate', self._learning_rate))
0081       summaries.append(tf.summary.scalar('grad_norm', grad_norm))
0082       if self._clipping:
0083         clipped = tf.minimum(grad_norm, self._clipping)
0084         summaries.append(tf.summary.scalar('clipped_gradient_norm', clipped))
0085       summary = tf.summary.merge(summaries)
0086     return summary
