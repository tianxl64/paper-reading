0477     with self.LOCK:
0478       self._env.reset()
0479     # Use at least one no-op.
0480     noops = self._random.randint(1, self._noops) if self._noops > 1 else 1
0481     for _ in range(noops):

0497       _, reward, done, info = self._env.step(action)
0498       total_reward += reward

0500         lives = self._env.ale.lives()

0502         self._lives = lives

0504         # In principle, the loop could exit before two valid frames have been

0507       elif step >= self._action_repeat - 2:
0508         index = step - (self._action_repeat - 2)
0509         if self._grayscale:
0510           self._env.ale.getScreenGrayscale(self._buffers[index])
0511         else:

0921   def _worker(self, constructor, conn):
