0023 def zip_(*structures, **kwargs):
0024   # Named keyword arguments are not allowed after *args in Python 2.
0025   flatten = kwargs.pop('flatten', False)
0026   assert not kwargs, 'zip() got unexpected keyword arguments.'
0027   return map(
0028       lambda *x: x if len(x) > 1 else x[0],
0029       *structures,
0030       flatten=flatten)

0033 def map_(function, *structures, **kwargs):
0034   # Named keyword arguments are not allowed after *args in Python 2.
0035   flatten = kwargs.pop('flatten', False)
0036   assert not kwargs, 'map() got unexpected keyword arguments.'
0037 
0038   def impl(function, *structures):
0039     if len(structures) == 0:
0040       return structures
0041     if all(isinstance(s, (tuple, list)) for s in structures):
0042       if len(set(len(x) for x in structures)) > 1:
0043         raise ValueError('Cannot merge tuples or lists of different length.')
0044       args = tuple((impl(function, *x) for x in _builtin_zip(*structures)))
0045       if hasattr(structures[0], '_fields'):  # namedtuple
0046         return type(structures[0])(*args)
0047       else:  # tuple, list
0048         return type(structures[0])(args)
0049     if all(isinstance(s, dict) for s in structures):
0050       if len(set(frozenset(x.keys()) for x in structures)) > 1:
0051         raise ValueError('Cannot merge dicts with different keys.')
0052       merged = {
0053           k: impl(function, *(s[k] for s in structures))
0054           for k in structures[0]}
0055       return type(structures[0])(merged)
0056     return function(*structures)
0057 
0058   result = impl(function, *structures)
0059   if flatten:
0060     result = flatten_(result)
0061   return result

0064 def flatten_(structure):
0065   if isinstance(structure, dict):
0066     result = ()
0067     for key in sorted(list(structure.keys())):
0068         result += flatten_(structure[key])
0069     return result
0070   if isinstance(structure, (tuple, list)):
0071     result = ()
0072     for element in structure:
0073       result += flatten_(element)
0074     return result
0075   return (structure,)

0078 def filter_(predicate, *structures, **kwargs):
0079   # Named keyword arguments are not allowed after *args in Python 2.
0080   flatten = kwargs.pop('flatten', False)
0081   assert not kwargs, 'filter() got unexpected keyword arguments.'
0082 
0083   def impl(predicate, *structures):
0084     if len(structures) == 0:
0085       return structures
0086     if all(isinstance(s, (tuple, list)) for s in structures):
0087       if len(set(len(x) for x in structures)) > 1:
0088         raise ValueError('Cannot merge tuples or lists of different length.')
0089       # Only wrap in tuples if more than one structure provided.
0090       if len(structures) > 1:
0091         filtered = (impl(predicate, *x) for x in _builtin_zip(*structures))
0092       else:
0093         filtered = (impl(predicate, x) for x in structures[0])
0094       # Remove empty containers and construct result structure.
0095       if hasattr(structures[0], '_fields'):  # namedtuple
0096         filtered = (x if x != () else None for x in filtered)
0097         return type(structures[0])(*filtered)
0098       else:  # tuple, list
0099         filtered = (
0100             x for x in filtered if not isinstance(x, (tuple, list, dict)) or x)
0101         return type(structures[0])(filtered)
0102     if all(isinstance(s, dict) for s in structures):
0103       if len(set(frozenset(x.keys()) for x in structures)) > 1:
0104         raise ValueError('Cannot merge dicts with different keys.')
0105       # Only wrap in tuples if more than one structure provided.
0106       if len(structures) > 1:
0107         filtered = {
0108             k: impl(predicate, *(s[k] for s in structures))
0109             for k in structures[0]}
0110       else:
0111         filtered = {k: impl(predicate, v) for k, v in structures[0].items()}
0112       # Remove empty containers and construct result structure.
0113       filtered = {
0114           k: v for k, v in filtered.items()
0115           if not isinstance(v, (tuple, list, dict)) or v}
0116       return type(structures[0])(filtered)
0117     if len(structures) > 1:
0118       return structures if predicate(*structures) else ()
0119     else:
0120       return structures[0] if predicate(structures[0]) else ()
0121 
0122   result = impl(predicate, *structures)
0123   if flatten:
0124     result = flatten_(result)
0125   return result
