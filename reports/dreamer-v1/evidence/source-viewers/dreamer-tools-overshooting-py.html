0028 def overshooting(
0029     cell, target, embedded, prev_action, length, amount, posterior=None,
0030     ignore_input=False):
0031   # Closed loop unroll to get posterior states, which are the starting points
0032   # for open loop unrolls. We don't need the last time step, since we have no
0033   # targets for unrolls from it.
0034   if posterior is None:
0035     use_obs = tf.ones(tf.shape(
0036         nested.flatten(embedded)[0][:, :, :1])[:3], tf.bool)
0037     use_obs = tf.cond(
0038         tf.convert_to_tensor(ignore_input),
0039         lambda: tf.zeros_like(use_obs, tf.bool),
0040         lambda: use_obs)
0041     (_, posterior), _ = tf.nn.dynamic_rnn(
0042         cell, (embedded, prev_action, use_obs), length, dtype=tf.float32,
0043         swap_memory=True)
0044 
0045   # Arrange inputs for every iteration in the open loop unroll. Every loop
0046   # iteration below corresponds to one row in the docstring illustration.
0047   max_length = shape.shape(nested.flatten(embedded)[0])[1]
0048   first_output = {
0049       # 'observ': embedded,
0050       'prev_action': prev_action,
0051       'posterior': posterior,
0052       'target': target,
0053       'mask': tf.sequence_mask(length, max_length, tf.int32),
0054   }
0055 
0056   progress_fn = lambda tensor: tf.concat([tensor[:, 1:], 0 * tensor[:, :1]], 1)
0057   other_outputs = tf.scan(
0058       lambda past_output, _: nested.map(progress_fn, past_output),
0059       tf.range(amount), first_output)
0060   sequences = nested.map(
0061       lambda lhs, rhs: tf.concat([lhs[None], rhs], 0),
0062       first_output, other_outputs)
0063 
0064   # Merge batch and time dimensions of steps to compute unrolls from every
0065   # time step as one batch. The time dimension becomes the number of
0066   # overshooting distances.
0067   sequences = nested.map(
0068       lambda tensor: _merge_dims(tensor, [1, 2]),
0069       sequences)
0070   sequences = nested.map(
0071       lambda tensor: tf.transpose(
0072           tensor, [1, 0] + list(range(2, tensor.shape.ndims))),
0073       sequences)
0074   merged_length = tf.reduce_sum(sequences['mask'], 1)
0075 
0076   # Mask out padding frames; unnecessary if the input is already masked.
0077   sequences = nested.map(
0078       lambda tensor: tensor * tf.cast(
0079           _pad_dims(sequences['mask'], tensor.shape.ndims),
0080           tensor.dtype),
0081       sequences)
0082 
0083   # Compute open loop rollouts.
0084   use_obs = tf.zeros(tf.shape(sequences['mask']), tf.bool)[..., None]
0085   embed_size = nested.flatten(embedded)[0].shape[2].value
0086   obs = tf.zeros(shape.shape(sequences['mask']) + [embed_size])
0087   prev_state = nested.map(
0088       lambda tensor: tf.concat([0 * tensor[:, :1], tensor[:, :-1]], 1),
0089       posterior)
0090   prev_state = nested.map(
0091       lambda tensor: _merge_dims(tensor, [0, 1]), prev_state)
0092   (priors, _), _ = tf.nn.dynamic_rnn(
0093       cell, (obs, sequences['prev_action'], use_obs),
0094       merged_length,
0095       prev_state)
0096 
0097   # Restore batch dimension.
0098   target, prior, posterior, mask = nested.map(
0099       functools.partial(_restore_batch_dim, batch_size=shape.shape(length)[0]),
0100       (sequences['target'], priors, sequences['posterior'], sequences['mask']))
0101 
0102   mask = tf.cast(mask, tf.bool)
0103   return target, prior, posterior, mask

0106 def _merge_dims(tensor, dims):
0107   if isinstance(tensor, (list, tuple, dict)):
0108     return nested.map(tensor, lambda x: _merge_dims(x, dims))
0109   tensor = tf.convert_to_tensor(tensor)
0110   if (np.array(dims) - min(dims) != np.arange(len(dims))).all():
0111     raise ValueError('Dimensions to merge must all follow each other.')
0112   start, end = dims[0], dims[-1]
0113   output = tf.reshape(tensor, tf.concat([
0114       tf.shape(tensor)[:start],
0115       [tf.reduce_prod(tf.shape(tensor)[start: end + 1])],
0116       tf.shape(tensor)[end + 1:]], axis=0))
0117   merged = tensor.shape[start: end + 1].as_list()
0118   output.set_shape(
0119       tensor.shape[:start].as_list() +
0120       [None if None in merged else np.prod(merged)] +
0121       tensor.shape[end + 1:].as_list())
0122   return output

0125 def _pad_dims(tensor, rank):
0126   for _ in range(rank - tensor.shape.ndims):
0127     tensor = tensor[..., None]
0128   return tensor

0131 def _restore_batch_dim(tensor, batch_size):
0132   initial = shape.shape(tensor)
0133   desired = [batch_size, initial[0] // batch_size] + initial[1:]
0134   return tf.reshape(tensor, desired)
