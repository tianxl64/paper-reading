0022 def shape(tensor):
0023   static = tensor.get_shape().as_list()
0024   dynamic = tf.unstack(tf.shape(tensor))
0025   assert len(static) == len(dynamic)
0026   combined = [d if s is None else s for s, d in zip(static, dynamic)]
0027   return combined
