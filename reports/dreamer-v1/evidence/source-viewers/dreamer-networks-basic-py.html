0019 import numpy as np
0020 import tensorflow as tf
0021 from tensorflow_probability import distributions as tfd
0022 
0023 from dreamer import distributions
0024 from dreamer import tools
0025 
0026 
0027 def feed_forward(
0028     features, data_shape, num_layers=2, activation=tf.nn.relu,
0029     mean_activation=None, stop_gradient=False, trainable=True, units=100,
0030     std=1.0, low=-1.0, high=1.0, dist='normal', min_std=1e-2, init_std=1.0):
0031   hidden = features
0032   if stop_gradient:
0033     hidden = tf.stop_gradient(hidden)
0034   for _ in range(num_layers):
0035     hidden = tf.layers.dense(hidden, units, activation, trainable=trainable)
0036   mean = tf.layers.dense(
0037       hidden, int(np.prod(data_shape)), mean_activation, trainable=trainable)
0038   mean = tf.reshape(mean, tools.shape(features)[:-1] + data_shape)
0039   if std == 'learned':
0040     std = tf.layers.dense(
0041         hidden, int(np.prod(data_shape)), None, trainable=trainable)
0042     init_std = np.log(np.exp(init_std) - 1)
0043     std = tf.nn.softplus(std + init_std) + min_std
0044     std = tf.reshape(std, tools.shape(features)[:-1] + data_shape)
0045   if dist == 'normal':
0046     dist = tfd.Normal(mean, std)
0047     dist = tfd.Independent(dist, len(data_shape))
0048   elif dist == 'deterministic':
0049     dist = tfd.Deterministic(mean)
0050     dist = tfd.Independent(dist, len(data_shape))
0051   elif dist == 'binary':
0052     dist = tfd.Bernoulli(mean)
0053     dist = tfd.Independent(dist, len(data_shape))
0054   elif dist == 'trunc_normal':
0055     # https://www.desmos.com/calculator/rnksmhtgui
0056     dist = tfd.TruncatedNormal(mean, std, low, high)
0057     dist = tfd.Independent(dist, len(data_shape))
0058   elif dist == 'tanh_normal':
0059     # https://www.desmos.com/calculator/794s8kf0es
0060     dist = distributions.TanhNormal(mean, std)
0061   elif dist == 'tanh_normal_tanh':
0062     # https://www.desmos.com/calculator/794s8kf0es
0063     mean = 5.0 * tf.tanh(mean / 5.0)
0064     dist = distributions.TanhNormal(mean, std)
0065   elif dist == 'onehot_score':
0066     dist = distributions.OneHot(mean, gradient='score')
0067   elif dist == 'onehot_straight':
0068     dist = distributions.OneHot(mean, gradient='straight')
0069   else:
0070     raise NotImplementedError(dist)
0071   return dist
