0029 def define_model(logdir, metrics, data, trainer, config):
0030   print('Build TensorFlow compute graph.')
0031   dependencies = []
0032   cleanups = []
0033   step = trainer.step
0034   global_step = trainer.global_step
0035   phase = trainer.phase
0036   timestamp = tf.py_func(
0037       lambda: datetime.datetime.utcnow().strftime('%Y%m%dT%H%M%S'),
0038       [], tf.string)
0039   dependencies.append(metrics.set_tags(
0040       global_step=global_step, step=step, phase=phase,
0041       time=timestamp))
0042 
0043   # Instantiate network blocks. Note, this initialization would be expensive
0044   # when using tf.function since it would run at every step.
0045   try:
0046     cell = config.cell()
0047   except TypeError:
0048     cell = config.cell(action_size=data['action'].shape[-1].value)
0049   kwargs = dict(create_scope_now_=True)
0050   encoder = tf.make_template('encoder', config.encoder, **kwargs)
0051   heads = tools.AttrDict(_unlocked=True)
0052   raw_dummy_features = cell.features_from_state(
0053       cell.zero_state(1, tf.float32))[:, None]
0054   for key, head in config.heads.items():
0055     name = 'head_{}'.format(key)
0056     kwargs = dict(create_scope_now_=True)
0057     if key in data:
0058       kwargs['data_shape'] = data[key].shape[2:].as_list()
0059     if key == 'action_target':
0060       kwargs['data_shape'] = data['action'].shape[2:].as_list()
0061     if key == 'cpc':
0062       kwargs['data_shape'] = [cell.feature_size]
0063       dummy_features = encoder(data)[:1, :1]
0064     else:
0065       dummy_features = raw_dummy_features
0066     heads[key] = tf.make_template(name, head, **kwargs)
0067     heads[key](dummy_features)  # Initialize weights.
0068 
0069   # Update target networks.
0070   if 'value_target' in heads:
0071     dependencies.append(tools.track_network(
0072         trainer, config.batch_shape[0],
0073         r'.*/head_value/.*', r'.*/head_value_target/.*',
0074         config.value_target_period, config.value_target_update))
0075   if 'value_target_2' in heads:
0076     dependencies.append(tools.track_network(
0077         trainer, config.batch_shape[0],
0078         r'.*/head_value/.*', r'.*/head_value_target_2/.*',
0079         config.value_target_period, config.value_target_update))
0080   if 'action_target' in heads:
0081     dependencies.append(tools.track_network(
0082         trainer, config.batch_shape[0],
0083         r'.*/head_action/.*', r'.*/head_action_target/.*',
0084         config.action_target_period, config.action_target_update))
0085 
0086   # Apply and optimize model.
0087   embedded = encoder(data)
0088   with tf.control_dependencies(dependencies):
0089     embedded = tf.identity(embedded)
0090   graph = tools.AttrDict(locals())
0091   prior, posterior = tools.unroll.closed_loop(
0092       cell, embedded, data['action'], config.debug)
0093   objectives = utility.compute_objectives(
0094       posterior, prior, data, graph, config)
0095   summaries, grad_norms = utility.apply_optimizers(
0096       objectives, trainer, config)
0097   dependencies += summaries
0098 
0099   # Active data collection.
0100   with tf.variable_scope('collection'):
0101     with tf.control_dependencies(dependencies):  # Make sure to train first.
0102       for name, params in config.train_collects.items():
0103         schedule = tools.schedule.binary(
0104             step, config.batch_shape[0],
0105             params.steps_after, params.steps_every, params.steps_until)
0106         summary, _ = tf.cond(
0107             tf.logical_and(tf.equal(trainer.phase, 'train'), schedule),
0108             functools.partial(
0109                 utility.simulate, metrics, config, params, graph, cleanups,
0110                 gif_summary=False, name=name),
0111             lambda: (tf.constant(''), tf.constant(0.0)),
0112             name='should_collect_' + name)
0113         summaries.append(summary)
0114         dependencies.append(summary)
0115 
0116   # Compute summaries.
0117   graph = tools.AttrDict(locals())
0118   summary, score = tf.cond(
0119       trainer.log,
0120       lambda: define_summaries.define_summaries(graph, config, cleanups),
0121       lambda: (tf.constant(''), tf.zeros((0,), tf.float32)),
0122       name='summaries')
0123   summaries = tf.summary.merge([summaries, summary])
0124   dependencies.append(utility.print_metrics(
0125       {ob.name: ob.value for ob in objectives},
0126       step, config.print_metrics_every, 2, 'objectives'))
0127   dependencies.append(utility.print_metrics(
0128       grad_norms, step, config.print_metrics_every, 2, 'grad_norms'))
0129   dependencies.append(tf.cond(trainer.log, metrics.flush, tf.no_op))
0130   with tf.control_dependencies(dependencies):
0131     score = tf.identity(score)
0132   return score, summaries, cleanups
