0026 def encoder(obs):
0027   kwargs = dict(strides=2, activation=tf.nn.relu)
0028   hidden = tf.reshape(obs['image'], [-1] + obs['image'].shape[2:].as_list())
0029   hidden = tf.layers.conv2d(hidden, 32, 4, **kwargs)
0030   hidden = tf.layers.conv2d(hidden, 64, 4, **kwargs)
0031   hidden = tf.layers.conv2d(hidden, 128, 4, **kwargs)
0032   hidden = tf.layers.conv2d(hidden, 256, 4, **kwargs)
0033   hidden = tf.layers.flatten(hidden)
0034   assert hidden.shape[1:].as_list() == [1024], hidden.shape.as_list()
0035   hidden = tf.reshape(hidden, tools.shape(obs['image'])[:2] + [
0036       np.prod(hidden.shape[1:].as_list())])
0037   return hidden
0038

0040 def decoder(features, data_shape, std=1.0):
0041   kwargs = dict(strides=2, activation=tf.nn.relu)
0042   hidden = tf.layers.dense(features, 1024, None)
0043   hidden = tf.reshape(hidden, [-1, 1, 1, hidden.shape[-1].value])
0044   hidden = tf.layers.conv2d_transpose(hidden, 128, 5, **kwargs)
0045   hidden = tf.layers.conv2d_transpose(hidden, 64, 5, **kwargs)
0046   hidden = tf.layers.conv2d_transpose(hidden, 32, 6, **kwargs)
0047   mean = tf.layers.conv2d_transpose(hidden, data_shape[-1], 6, strides=2)
0048   assert mean.shape[1:].as_list() == data_shape, mean.shape
0049   mean = tf.reshape(mean, tools.shape(features)[:-1] + data_shape)
0050   return tfd.Independent(tfd.Normal(mean, std), len(data_shape))
