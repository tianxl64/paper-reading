0015 from __future__ import absolute_import

0019 import tensorflow as tf
0020 from tensorflow_probability import distributions as tfd
0021 
0022 from dreamer import tools
0023 
0024 
0025 class Base(tf.nn.rnn_cell.RNNCell):
0026 
0027   def __init__(self, transition_tpl, posterior_tpl, reuse=None):
0028     super(Base, self).__init__(_reuse=reuse)
0029     self._posterior_tpl = posterior_tpl
0030     self._transition_tpl = transition_tpl
0031     self._debug = False
0032 
0033   @property
0034   def state_size(self):
0035     raise NotImplementedError
0036 
0037   @property
0038   def updates(self):
0039     return []
0040 
0041   @property
0042   def losses(self):
0043     return []
0044 
0045   @property
0046   def output_size(self):
0047     return (self.state_size, self.state_size)
0048 
0049   def zero_state(self, batch_size, dtype):
0050     return tools.nested.map(
0051         lambda size: tf.zeros([batch_size, size], dtype),
0052         self.state_size)
0053 
0054   def features_from_state(self, state):
0055     raise NotImplementedError
0056 
0057   def dist_from_state(self, state, mask=None):
0058     raise NotImplementedError
0059 
0060   def divergence_from_states(self, lhs, rhs, mask=None):
0061     lhs = self.dist_from_state(lhs, mask)
0062     rhs = self.dist_from_state(rhs, mask)
0063     divergence = tfd.kl_divergence(lhs, rhs)
0064     if mask is not None:
0065       divergence = tools.mask(divergence, mask)
0066     return divergence
0067 
0068   def call(self, inputs, prev_state):
0069     obs, prev_action, use_obs = inputs
0070     if self._debug:
0071       with tf.control_dependencies([tf.assert_equal(use_obs, use_obs[0, 0])]):
0072         use_obs = tf.identity(use_obs)
0073     use_obs = use_obs[0, 0]
0074     zero_obs = tools.nested.map(tf.zeros_like, obs)
0075     prior = self._transition_tpl(prev_state, prev_action, zero_obs)
0076     posterior = tf.cond(
0077         use_obs,
0078         lambda: self._posterior_tpl(prev_state, prev_action, obs),
0079         lambda: prior)
0080     return (prior, posterior), posterior
