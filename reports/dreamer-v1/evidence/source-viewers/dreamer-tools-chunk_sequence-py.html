0019 import tensorflow as tf

0021 from dreamer.tools import nested

0024 def chunk_sequence(sequence, chunk_length, randomize=True, num_chunks=None):
0025   if 'length' in sequence:
0026     length = sequence.pop('length')
0027   else:
0028     length = tf.shape(nested.flatten(sequence)[0])[0]
0029   if randomize:
0030     if not num_chunks:
0031       num_chunks = tf.maximum(1, length // chunk_length - 1)
0032     else:
0033       num_chunks = num_chunks + 0 * length
0034     used_length = num_chunks * chunk_length
0035     max_offset = length - used_length
0036     offset = tf.random_uniform((), 0, max_offset + 1, dtype=tf.int32)
0037   else:
0038     if num_chunks is None:
0039       num_chunks = length // chunk_length
0040     else:
0041       num_chunks = num_chunks + 0 * length
0042     used_length = num_chunks * chunk_length
0043     offset = 0
0044   clipped = nested.map(
0045       lambda tensor: tensor[offset: offset + used_length],
0046       sequence)
0047   chunks = nested.map(
0048       lambda tensor: tf.reshape(
0049           tensor, [num_chunks, chunk_length] + tensor.shape[1:].as_list()),
0050       clipped)
0051   return chunks
