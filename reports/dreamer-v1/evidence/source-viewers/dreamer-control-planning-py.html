0024 def cross_entropy_method(
0025     cell, objective, state, obs_shape, action_shape, horizon, graph,
0026     beams=1000, topk=100, iterations=10, min_action=-1, max_action=1):
0027   obs_shape, action_shape = tuple(obs_shape), tuple(action_shape)
0028   batch = tools.shape(tools.nested.flatten(state)[0])[0]
0029   initial_state = tools.nested.map(lambda tensor: tf.tile(
0030       tensor, [beams] + [1] * (tensor.shape.ndims - 1)), state)
0031   extended_batch = tools.shape(tools.nested.flatten(initial_state)[0])[0]
0032   use_obs = tf.zeros([extended_batch, horizon, 1], tf.bool)
0033   obs = tf.zeros((extended_batch, horizon) + obs_shape)
0034 
0035   def iteration(index, mean, stddev):
0036     # Sample action proposals from belief.
0037     normal = tf.random_normal((batch, beams, horizon) + action_shape)
0038     action = normal * stddev[:, None] + mean[:, None]
0039     action = tf.clip_by_value(action, min_action, max_action)
0040     # Evaluate proposal actions.
0041     action = tf.reshape(
0042         action, (extended_batch, horizon) + action_shape)
0043     (_, state), _ = tf.nn.dynamic_rnn(
0044         cell, (0 * obs, action, use_obs), initial_state=initial_state)
0045     return_ = objective(state)
0046     return_ = tf.reshape(return_, (batch, beams))
0047     # Re-fit belief to the best ones.
0048     _, indices = tf.nn.top_k(return_, topk, sorted=False)
0049     indices += tf.range(batch)[:, None] * beams
0050     best_actions = tf.gather(action, indices)
0051     mean, variance = tf.nn.moments(best_actions, 1)
0052     stddev = tf.sqrt(variance + 1e-6)
0053     return index + 1, mean, stddev
0054 
0055   mean = tf.zeros((batch, horizon) + action_shape)
0056   stddev = tf.ones((batch, horizon) + action_shape)
0057   _, mean, std = tf.while_loop(
0058       lambda index, mean, stddev: index < iterations, iteration,
0059       (0, mean, stddev), back_prop=False)
0060   return mean

0063 def action_head_policy(
0064     cell, objective, state, obs_shape, action_shape, graph, config, strategy):
0065   features = cell.features_from_state(state)
0066   policy = graph.heads.action(features)
0067   if strategy == 'sample':
0068     action = policy.sample()
0069   elif strategy == 'mode':
0070     action = policy.mode()
0071   else:
0072     raise NotImplementedError(strategy)
0073   plan = action[:, None, :]
0074   return plan
