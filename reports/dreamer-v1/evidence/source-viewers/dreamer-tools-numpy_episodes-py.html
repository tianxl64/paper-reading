0032 def numpy_episodes(
0033     train_dir, test_dir, shape, reader=None, loader=None, num_chunks=None,
0034     preprocess_fn=None, gpu_prefetch=False):
0035   reader = reader or episode_reader
0036   loader = loader or cache_loader
0037   try:
0038     dtypes, shapes = _read_spec(reader, train_dir)
0039   except ZeroDivisionError:
0040     dtypes, shapes = _read_spec(reader, test_dir)
0041   train = tf.data.Dataset.from_generator(
0042       functools.partial(loader, reader, train_dir),
0043       dtypes, shapes)
0044   test = tf.data.Dataset.from_generator(
0045       functools.partial(cache_loader, reader, test_dir, shape[0]),
0046       dtypes, shapes)
0047   chunking = lambda x: tf.data.Dataset.from_tensor_slices(
0048       chunk_sequence.chunk_sequence(x, shape[1], True, num_chunks))
0049   train = train.flat_map(chunking)
0050   train = train.batch(shape[0], drop_remainder=True)
0051   if preprocess_fn:
0052     train = train.map(preprocess_fn, tf.data.experimental.AUTOTUNE)
0053   if gpu_prefetch:
0054     train = train.apply(tf.data.experimental.copy_to_device('/gpu:0'))
0055   train = train.prefetch(tf.data.experimental.AUTOTUNE)
0056   test = test.flat_map(chunking)
0057   test = test.batch(shape[0], drop_remainder=True)
0058   if preprocess_fn:
0059     test = test.map(preprocess_fn, tf.data.experimental.AUTOTUNE)
0060   if gpu_prefetch:
0061     test = test.apply(tf.data.experimental.copy_to_device('/gpu:0'))
0062   test = test.prefetch(tf.data.experimental.AUTOTUNE)
0063   return attr_dict.AttrDict(train=train, test=test)

0066 def cache_loader(reader, directory, every):
0067   cache = {}
0068   while True:
0069     episodes = _sample(cache.values(), every)
0070     for episode in _permuted(episodes, every):
0071       yield episode
0072     filenames = tf.gfile.Glob(os.path.join(directory, '*.npz'))
0073     filenames = [filename for filename in filenames if filename not in cache]
0074     for filename in filenames:
0075       cache[filename] = reader(filename)

0078 def recent_loader(reader, directory, every):
0079   recent = {}
0080   cache = {}
0081   while True:
0082     episodes = []
0083     episodes += _sample(recent.values(), every // 2)
0084     episodes += _sample(cache.values(), every // 2)
0085     for episode in _permuted(episodes, every):
0086       yield episode
0087     cache.update(recent)
0088     recent = {}
0089     filenames = tf.gfile.Glob(os.path.join(directory, '*.npz'))
0090     filenames = [filename for filename in filenames if filename not in cache]
0091     for filename in filenames:

0095 def window_loader(reader, directory, window, every):
0096   cache = {}
0097   while True:
0098     episodes = _sample(cache.values(), every)
0099     for episode in _permuted(episodes, every):
0100       yield episode
0101     filenames = tf.gfile.Glob(os.path.join(directory, '*.npz'))
0102     filenames = sorted(filenames)[-window:]
0103     for filename in filenames:
0104       if filename not in cache:
0105         cache[filename] = reader(filename)
0106     for key in list(cache.keys()):
0107       if key not in filenames:
0108         del cache[key]

0111 def reload_loader(reader, directory):
0112   directory = os.path.expanduser(directory)
0113   while True:
0114     filenames = tf.gfile.Glob(os.path.join(directory, '*.npz'))
0115     random.shuffle(filenames)
0116     for filename in filenames:
0117       yield reader(filename)

0120 def dummy_loader(reader, directory):
0121   random = np.random.RandomState(seed=0)
0122   dtypes, shapes, length = _read_spec(reader, directory, True, True)
0123   while True:
0124     episode = {}
0125     for key in dtypes:
0126       dtype, shape = dtypes[key], (length,) + shapes[key][1:]
0127       if dtype in (np.float32, np.float64):
0128         episode[key] = random.uniform(0, 1, shape).astype(dtype)
0129       elif dtype in (np.int32, np.int64, np.uint8):
0130         episode[key] = random.uniform(0, 255, shape).astype(dtype)
0131       else:
0132         raise NotImplementedError('Unsupported dtype {}.'.format(dtype))
0133     yield episode

0136 def episode_reader(
0137     filename, resize=None, max_length=None, action_noise=None,
0138     clip_rewards=False, pcont_scale=None):
0139   try:
0140     with tf.gfile.Open(filename, 'rb') as file_:
0141       episode = np.load(file_)
0142   except (IOError, ValueError):
0143     # Try again one second later, in case the file was still being written.
0144     time.sleep(1)
0145     with tf.gfile.Open(filename, 'rb') as file_:
0146       episode = np.load(file_)
0147   episode = {key: _convert_type(episode[key]) for key in episode.keys()}
0148   episode['return'] = np.cumsum(episode['reward'])
0149   if 'reward_mask' not in episode:
0150     episode['reward_mask'] = np.ones_like(episode['reward'])[..., None]
0151   if max_length:
0152     episode = {key: value[:max_length] for key, value in episode.items()}
0153   if resize and resize != 1:
0154     factors = (1, resize, resize, 1)
0155     episode['image'] = interpolation.zoom(episode['image'], factors)
0156   if action_noise:
0157     seed = np.fromstring(filename, dtype=np.uint8)
0158     episode['action'] += np.random.RandomState(seed).normal(
0159         0, action_noise, episode['action'].shape)
0160   if clip_rewards is False:
0161     pass
0162   elif clip_rewards == 'sign':
0163     episode['reward'] = np.sign(episode['reward'])
0164   elif clip_rewards == 'tanh':
0165     episode['reward'] = np.tanh(episode['reward'])
0166   else:
0167     raise NotImplementedError(clip_rewards)
0168   if pcont_scale is not None:
0169     episode['pcont'] *= pcont_scale
0170   return episode

0173 def _read_spec(
0174     reader, directory, return_length=False, numpy_types=False):
0175   episodes = reload_loader(reader, directory)
0176   episode = next(episodes)
0177   episodes.close()
0178   dtypes = {key: value.dtype for key, value in episode.items()}
0179   if not numpy_types:
0180     dtypes = {key: tf.as_dtype(value) for key, value in dtypes.items()}
0181   shapes = {key: value.shape for key, value in episode.items()}
0182   shapes = {key: (None,) + shape[1:] for key, shape in shapes.items()}
0183   if return_length:
0184     length = len(episode[list(shapes.keys())[0]])
0185     return dtypes, shapes, length
0186   else:
0187     return dtypes, shapes

0190 def _convert_type(array):
0191   if array.dtype == np.float64:
0192     return array.astype(np.float32)
0193   if array.dtype == np.int64:
0194     return array.astype(np.int32)
0195   return array

0198 def _sample(sequence, amount):
0199   sequence = list(sequence)
0200   amount = min(amount, len(sequence))
0201   return random.sample(sequence, amount)

0204 def _permuted(sequence, amount):
0205   sequence = list(sequence)
0206   if not sequence:
0207     return
0208   index = 0
0209   while True:
0210     for element in np.random.permutation(sequence):
0211       if index >= amount:
0212         return
0213       yield element
0214       index += 1
