0024 class TanhNormal(object):
0025 
0026   def __init__(self, mean, std, samples=100):
0027     dist = tfd.Normal(mean, std)
0028     dist = tfd.TransformedDistribution(dist, TanhBijector())
0029     dist = tfd.Independent(dist, 1)
0030     self._dist = dist
0031     self._samples = samples
0032 
0033   @property
0034   def name(self):
0035     return 'TanhNormalDistribution'
0036 
0037   def __getattr__(self, name):
0038     return getattr(self._dist, name)
0039 
0040   def mean(self):
0041     samples = self._dist.sample(self._samples)
0042     return tf.reduce_mean(samples, 0)
0043 
0044   def stddev(self):
0045     samples = self._dist.sample(self._samples)
0046     mean = tf.reduce_mean(samples, 0, keep_dims=True)
0047     return tf.reduce_mean(tf.pow(samples - mean, 2), 0)
0048 
0049   def mode(self):
0050     samples = self._dist.sample(self._samples)
0051     logprobs = self._dist.log_prob(samples)
0052     mask = tf.one_hot(tf.argmax(logprobs, axis=0), self._samples, axis=0)
0053     return tf.reduce_sum(samples * mask[..., None], 0)
0054 
0055   def entropy(self):
0056     sample = self._dist.sample(self._samples)
0057     logprob = self.log_prob(sample)
0058     return -tf.reduce_mean(logprob, 0)
0059

0061 class TanhBijector(tfp.bijectors.Bijector):
0062 
0063   def __init__(self, validate_args=False, name='tanh'):
0064     super(TanhBijector, self).__init__(
0065         forward_min_event_ndims=0,
0066         validate_args=validate_args,
0067         name=name)
0068 
0069   def _forward(self, x):
0070     return tf.nn.tanh(x)
0071 
0072   def _inverse(self, y):
0073     precision = 0.99999997
0074     clipped = tf.where(
0075         tf.less_equal(tf.abs(y), 1.),
0076         tf.clip_by_value(y, -precision, precision), y)
0077     # y = tf.stop_gradient(clipped) + y - tf.stop_gradient(y)
0078     return tf.atanh(clipped)
0079 
0080   def _forward_log_det_jacobian(self, x):
0081     log2 = tf.math.log(tf.constant(2.0, dtype=x.dtype))
0082     return 2.0 * (log2 - x - tf.nn.softplus(-2.0 * x))
