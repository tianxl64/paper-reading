0027 _Phase = collections.namedtuple(

0033 class Trainer(object):
0034 
0035   def __init__(self, logdir, config=None):
0036     self._logdir = logdir
0037     self._global_step = tf.train.get_or_create_global_step()
0038     self._step = tf.placeholder(tf.int32, name='step')
0039     self._phase = tf.placeholder(tf.string, name='phase')
0040     self._log = tf.placeholder(tf.bool, name='log')
0041     self._report = tf.placeholder(tf.bool, name='report')
0042     self._reset = tf.placeholder(tf.bool, name='reset')
0043     self._phases = []
0044     # Checkpointing.
0045     self._loaders = []
0046     self._savers = []
0047     self._logdirs = []
0048     self._checkpoints = []
0049     self._config = config or tools.AttrDict()

0071   def add_saver(
0072       self, include=r'.*', exclude=r'.^', logdir=None, load=True, save=True,
0073       checkpoint=None):
0074     variables = tools.filter_variables(include, exclude)
0075     saver = tf.train.Saver(variables, max_to_keep=1)
0076     if load:
0077       self._loaders.append(saver)
0078     if save:
0079       self._savers.append(saver)
0080     self._logdirs.append(logdir or self._logdir)
0081     if checkpoint is None and self._config.checkpoint_to_load:
0082       self._checkpoints.append(
0083           os.path.join(self._logdirs[-1], self._config.checkpoint_to_load))
0084     else:
0085       self._checkpoints.append(checkpoint)

0087   def add_phase(
0088       self, name, steps, score, summary, batch_size=1,
0089       report_every=None, log_every=None, checkpoint_every=None,
0090       restore_every=None, feed=None):
0091     score = tf.convert_to_tensor(score, tf.float32)
0092     summary = tf.convert_to_tensor(summary, tf.string)
0093     feed = feed or {}
0094     if not score.shape.ndims:
0095       score = score[None]
0096     writer = self._logdir and tf.summary.FileWriter(
0097         os.path.join(self._logdir, name),
0098         tf.get_default_graph(), flush_secs=30)
0099     op = self._define_step(name, batch_size, score, summary)
0100     self._phases.append(_Phase(
0101         name, writer, op, batch_size, int(steps), feed, report_every,
0102         log_every, checkpoint_every, restore_every))
0103 
0104   def run(self, max_step=None, sess=None, unused_saver=None):
0105     for _ in self.iterate(max_step, sess):
0106       pass
0107 
0108   def iterate(self, max_step=None, sess=None):
0109     sess = sess or self._create_session()
0110     with sess:
0111       self._initialize_variables(
0112           sess, self._loaders, self._logdirs, self._checkpoints)
0113       sess.graph.finalize()
0114       while True:
0115         global_step = sess.run(self._global_step)
0116         if max_step and global_step >= max_step:
0117           break
0118         phase, epoch, steps_in = self._find_current_phase(global_step)
0119         phase_step = epoch * phase.steps + steps_in
0120         if steps_in % phase.steps < phase.batch_size:
0121           message = '\n' + ('-' * 50) + '\n'
0122           message += 'Epoch {} phase {} (phase step {}, global step {}).'
0123           print(message.format(epoch + 1, phase.name, phase_step, global_step))
0124         # Populate book keeping tensors.
0125         phase.feed[self._step] = phase_step
0126         phase.feed[self._phase] = phase.name
0127         phase.feed[self._reset] = (steps_in < phase.batch_size)
0128         phase.feed[self._log] = phase.writer and self._is_every_steps(
0129             phase_step, phase.batch_size, phase.log_every)
0130         phase.feed[self._report] = self._is_every_steps(
0131             phase_step, phase.batch_size, phase.report_every)
0132         summary, mean_score, global_step = sess.run(phase.op, phase.feed)
0133         if self._is_every_steps(
0134             phase_step, phase.batch_size, phase.checkpoint_every):
0135           for saver in self._savers:
0136             self._store_checkpoint(sess, saver, global_step)
0137         if self._is_every_steps(
0138             phase_step, phase.batch_size, phase.report_every):
0139           print('Score {}.'.format(mean_score))
0140           yield mean_score
0141         if summary and phase.writer:
0142           # We want smaller phases to catch up at the beginnig of each epoch so
0143           # that their graphs are aligned.
0144           longest_phase = max(phase_.steps for phase_ in self._phases)
0145           summary_step = epoch * longest_phase + steps_in
0146           phase.writer.add_summary(summary, summary_step)
0147         if self._is_every_steps(
0148             phase_step, phase.batch_size, phase.restore_every):
0149           self._initialize_variables(
0150               sess, self._loaders, self._logdirs, self._checkpoints)
0151 
0152   def _is_every_steps(self, phase_step, batch, every):
0153     if not every:
0154       return False
0155     covered_steps = range(phase_step, phase_step + batch)
0156     return any((step + 1) % every == 0 for step in covered_steps)
0157 
0158   def _find_current_phase(self, global_step):
0159     epoch_size = sum(phase.steps for phase in self._phases)
0160     epoch = int(global_step // epoch_size)
0161     steps_in = global_step % epoch_size
0162     for phase in self._phases:
0163       if steps_in < phase.steps:
0164         return phase, epoch, steps_in
0165       steps_in -= phase.steps
0166 
0167   def _define_step(self, name, batch_size, score, summary):
0168     with tf.variable_scope('phase_{}'.format(name)):
0169       score_mean = tools.StreamingMean((), tf.float32, 'score_mean')
0170       score.set_shape((None,))
0171       with tf.control_dependencies([score, summary]):
0172         submit_score = score_mean.submit(score)
0173       with tf.control_dependencies([submit_score]):
0174         mean_score = tf.cond(self._report, score_mean.clear, float)
0175         summary = tf.cond(
0176             self._report,
0177             lambda: tf.summary.merge([summary, tf.summary.scalar(
0178                 name + '/score', mean_score, family='trainer')]),
0179             lambda: summary)
0180         next_step = self._global_step.assign_add(batch_size)
0181       with tf.control_dependencies([summary, mean_score, next_step]):
0182         return (
0183             tf.identity(summary),
0184             tf.identity(mean_score),
0185             tf.identity(next_step))

0187   def _create_session(self):
0188     config = tf.ConfigProto()
0189     config.gpu_options.allow_growth = True
0190     try:
0191       return tf.Session('local', config=config)
0192     except tf.errors.NotFoundError:
0193       return tf.Session(config=config)
0194 
0195   def _initialize_variables(self, sess, savers, logdirs, checkpoints):
0196     sess.run(tf.group(
0197         tf.local_variables_initializer(),
0198         tf.global_variables_initializer()))
0199     assert len(savers) == len(logdirs) == len(checkpoints)
0200     for i, (saver, logdir, checkpoint) in enumerate(
0201         zip(savers, logdirs, checkpoints)):
0202       logdir = os.path.expanduser(logdir)
0203       state = tf.train.get_checkpoint_state(logdir)
0204       if checkpoint:
0205         checkpoint = os.path.join(logdir, checkpoint)
0206       if not checkpoint and state and state.model_checkpoint_path:
0207         checkpoint = state.model_checkpoint_path
0208       if checkpoint:
0209         saver.restore(sess, checkpoint)
0210 
0211   def _store_checkpoint(self, sess, saver, global_step):
0212     if not self._logdir or not saver:
0213       return
0214     tf.gfile.MakeDirs(self._logdir)
0215     filename = os.path.join(self._logdir, 'model.ckpt')
0216     saver.save(sess, filename, global_step)
