0015 from __future__ import absolute_import
0016 from __future__ import division
0017 from __future__ import print_function
0018 
0019 import tensorflow as tf
0020 from tensorflow_probability import distributions as tfd
0021 
0022 
0023 class OneHot(object):
0024 
0025   def __init__(self, logits=None, probs=None, gradient='score'):
0026     self._gradient = gradient
0027     self._dist = tfd.Categorical(logits=logits, probs=probs)
0028     self._num_classes = self.mean().shape[-1].value
0029 
0030   @property
0031   def name(self):
0032     return 'OneHotDistribution'
0033 
0034   def __getattr__(self, name):
0035     return getattr(self._dist, name)
0036 
0037   def prob(self, events):
0038     indices = tf.argmax(events, axis=-1)
0039     return self._dist.prob(indices)
0040 
0041   def log_prob(self, events):
0042     indices = tf.argmax(events, axis=-1)
0043     return self._dist.log_prob(indices)
0044 
0045   def mean(self):
0046     return self._dist.probs_parameter()
0047 
0048   def stddev(self):
0049     values = tf.one_hot(tf.range(self._num_classes), self._num_classes)
0050     distances = tf.reduce_sum((values - self.mean()[..., None, :]) ** 2, -1)
0051     return tf.sqrt(distances)
0052 
0053   def mode(self):
0054     return tf.one_hot(self._dist.mode(), self._num_classes)
0055 
0056   def sample(self, amount=None):
0057     amount = [amount] if amount else []
0058     sample = tf.one_hot(self._dist.sample(*amount), self._num_classes)
0059     if self._gradient == 'score':  # Implemented as DiCE.
0060       logp = self.log_prob(sample)[..., None]
0061       sample *= tf.exp(logp - tf.stop_gradient(logp))
0062     elif self._gradient == 'straight':  # Gradient for all classes.
0063       probs = self._dist.probs_parameter()
0064       sample += probs - tf.stop_gradient(probs)
0065     else:
0066       raise NotImplementedError(self._gradient)
0067     return sample
