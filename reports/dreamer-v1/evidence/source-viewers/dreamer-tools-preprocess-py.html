0015 from __future__ import absolute_import
0016 from __future__ import division
0017 from __future__ import print_function

0019 import tensorflow as tf

0021 
0022 def preprocess(observ, bits):
0023   if 'image' in observ:
0024     bins = 2 ** bits
0025     image = tf.cast(observ['image'], tf.float32)
0026     if bits < 8:
0027       image = tf.floor(image / 2 ** (8 - bits))
0028     image = image / bins
0029     image = image + tf.random_uniform(tf.shape(image), 0, 1.0 / bins)
0030     image = image - 0.5
0031     observ['image'] = image
0032   return observ
0033 
0034 
0035 def postprocess(image, bits, dtype=tf.float32):
0036   bins = 2 ** bits
0037   if dtype == tf.float32:
0038     image = tf.floor(bins * (image + 0.5)) / bins
0039   elif dtype == tf.uint8:
0040     image = image + 0.5
0041     image = tf.floor(bins * image)
0042     image = image * (256.0 / bins)
0043     image = tf.cast(tf.clip_by_value(image, 0, 255), tf.uint8)
0044   else:
0045     raise NotImplementedError(dtype)
0046   return image
