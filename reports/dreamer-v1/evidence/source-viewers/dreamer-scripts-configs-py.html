0112 def make_config(params):
0113   config = tools.AttrDict()
0114   config.debug = params.get('debug', False)
0115   with params.unlocked:
0116     for name in params.get('defaults', ['dreamer']):
0117       for key, value in DEFAULTS[name].items():
0118         if key not in params:
0119           params[key] = value
0120   config.loss_scales = tools.AttrDict()
0121   config = _data_processing(config, params)
0122   config = _model_components(config, params)
0123   config = _tasks(config, params)
0124   config = _loss_functions(config, params)
0125   config = _training_schedule(config, params)
0126   # Mark params as used which are only accessed at run-time.
0127   run_time_keys = [
0128       'planner_discount', 'planner_lambda', 'objective_entropy_scale',
0129       'normalize_actions', 'max_length', 'render_size', 'atari_lifes',
0130       'atari_noops', 'atari_sticky', 'atari_train_max_length',
0131       'atari_grayscale']
0132   for key in run_time_keys:
0133     params.get(key, None)
0134   if params.untouched:
0135     message = 'Found unused config overrides: {}'
0136     raise KeyError(message.format(', '.join(params.untouched)))
0137   return config

0259 def _tasks(config, params):
0260   config.isolate_envs = params.get('isolate_envs', 'thread')
0261   train_tasks, test_tasks = [], []
0262   for name in params.get('tasks', ['cheetah_run']):
0263     try:
0264       train_tasks.append(getattr(tasks_lib, name)(config, params, 'train'))
0265       test_tasks.append(getattr(tasks_lib, name)(config, params, 'test'))
0266     except TypeError:
0267       train_tasks.append(getattr(tasks_lib, name)(config, params))
0268       test_tasks.append(getattr(tasks_lib, name)(config, params))
0269   def common_spaces_ctor(task, action_spaces):
0270     env = task.env_ctor()
0271     env = control.wrappers.SelectObservations(env, ['image'])
0272     env = control.wrappers.PadActions(env, action_spaces)
0273     return env
0274   if len(train_tasks) > 1:
0275     action_spaces = [task.env_ctor().action_space for task in train_tasks]
0276     for index, task in enumerate(train_tasks):
0277       env_ctor = tools.bind(common_spaces_ctor, task, action_spaces)
0278       train_tasks[index] = tasks_lib.Task(task.name, env_ctor, [])
0279   if len(test_tasks) > 1:
0280     action_spaces = [task.env_ctor().action_space for task in test_tasks]
0281     for index, task in enumerate(test_tasks):
0282       env_ctor = tools.bind(common_spaces_ctor, task, action_spaces)
0283       test_tasks[index] = tasks_lib.Task(task.name, env_ctor, [])
0284   if config.gradient_heads == 'all_but_image':
0285     config.gradient_heads = train_tasks[0].state_components
0286   diags = params.get('state_diagnostics', True)
0287   for name in train_tasks[0].state_components + ['reward', 'pcont']:
0288     if name not in config.gradient_heads + ['reward', 'pcont'] and not diags:
0289       continue
0290     kwargs = {}
0291     kwargs['stop_gradient'] = name not in config.gradient_heads
0292     if name == 'pcont':
0293       kwargs['dist'] = 'binary'
0294     default = dict(reward=2).get(name, config.num_layers)
0295     kwargs['num_layers'] = params.get(name + '_layers', default)
0296     config.heads[name] = tools.bind(config.head_network, **kwargs)
0297     config.loss_scales[name] = 1.0
0298   config.train_tasks = train_tasks
0299   config.test_tasks = test_tasks
0300   return config

0462 def _active_collection(tasks, collects, defaults, config, params):
0463   sims = tools.AttrDict()
0464   for task in tasks:
0465     for user_collect in collects:
0466       for key in user_collect:
0467         if key not in defaults:
0468           message = 'Invalid key {} in activation collection config.'
0469           raise KeyError(message.format(key))
0470       collect = tools.AttrDict(defaults, _unlocked=True)
0471       collect.update(user_collect)
0472       collect.planner = _define_planner(
0473           collect.planner, collect.horizon, config, params)
0474       collect.objective = tools.bind(
0475           getattr(objectives_lib, collect.objective), params=params)
0476       if collect.give_rewards:
0477         collect.task = task
0478       else:
0479         env_ctor = tools.bind(
0480             lambda ctor: control.wrappers.NoRewardHint(ctor()),
0481             task.env_ctor)
0482         collect.task = tasks_lib.Task(task.name, env_ctor)
0483       collect.exploration = tools.AttrDict(
0484           scale=collect.action_noise_scale,
0485           type=collect.action_noise_type,
0486           schedule=tools.bind(
0487               tools.schedule.linear,
0488               ramp=collect.action_noise_ramp,
0489               min=collect.action_noise_min),
0490           factors=collect.action_noise_factors)
0491       name = '{}_{}_{}'.format(collect.prefix, collect.name, task.name)
0492       assert name not in sims, (set(sims.keys()), name)
0493       sims[name] = collect
0494   return sims
