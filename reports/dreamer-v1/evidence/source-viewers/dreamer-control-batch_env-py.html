0024 class TFBatchEnv(object):
0025 
0026   def __init__(self, envs, blocking):
0027     self._batch_env = PyBatchEnv(envs, blocking, flatten=True)
0028     spaces = self._batch_env.observation_space.spaces
0029     self._dtypes = [self._parse_dtype(spaces[key]) for key in self._keys[:-2]]
0030     self._dtypes += [tf.float32, tf.bool]  # Reward and done flag.
0031     self._shapes = [self._parse_shape(spaces[key]) for key in self._keys[:-2]]
0032     self._shapes += [(), ()]  # Reward and done flag.
0033 
0034   def __getattr__(self, name):
0035     return getattr(self._batch_env, name)
0036 
0037   def __len__(self):
0038     return len(self._batch_env)
0039 
0040   def __getitem__(self, index):
0041     return self._batch_env[index]
0042 
0043   def step(self, action):
0044     output = tf.py_func(
0045         self._batch_env.step, [action], self._dtypes, name='step')
0046     return self._process_output(output, len(self._batch_env))
0047 
0048   def reset(self, indices=None):
0049     if indices is None:
0050       indices = tf.range(len(self._batch_env))
0051     output = tf.py_func(
0052         self._batch_env.reset, [indices], self._dtypes, name='reset')
0053     return self._process_output(output, None)
0054 
0055   def _process_output(self, output, batch_size):
0056     for tensor, shape in zip(output, self._shapes):
0057       tensor.set_shape((batch_size,) + shape)
0058     return {key: tensor for key, tensor in zip(self._keys, output)}
0059 
0060   def _parse_dtype(self, space):
0061     if isinstance(space, gym.spaces.Discrete):
0062       return tf.int32
0063     if isinstance(space, gym.spaces.Box):
0064       if space.low.dtype == np.uint8:
0065         return tf.uint8
0066       else:
0067         return tf.float32
0068     raise NotImplementedError()
0069 
0070   def _parse_shape(self, space):
0071     if isinstance(space, gym.spaces.Discrete):
0072       return ()
0073     if isinstance(space, gym.spaces.Box):
0074       return space.shape
0075     raise NotImplementedError("Unsupported space '{}.'".format(space))

0078 class PyBatchEnv(object):
0079 
0080   def __init__(self, envs, blocking, flatten=False):
0081     observ_space = envs[0].observation_space
0082     if not all(env.observation_space == observ_space for env in envs):
0083       raise ValueError('All environments must use the same observation space.')
0084     action_space = envs[0].action_space
0085     if not all(env.action_space == action_space for env in envs):
0086       raise ValueError('All environments must use the same observation space.')
0087     self._envs = envs
0088     self._blocking = blocking
0089     self._flatten = flatten
0090     self._keys = list(sorted(observ_space.spaces.keys())) + ['reward', 'done']
0091 
0092   def __len__(self):
0093     return len(self._envs)
0094 
0095   def __getitem__(self, index):
0096     return self._envs[index]
0097 
0098   def __getattr__(self, name):
0099     return getattr(self._envs[0], name)
0100 
0101   def step(self, actions):
0102     for index, (env, action) in enumerate(zip(self._envs, actions)):
0103       if not env.action_space.contains(action):
0104         message = 'Invalid action for batch index {}: {}'
0105         raise ValueError(message.format(index, action))
0106     if self._blocking:
0107       transitions = [
0108           env.step(action)
0109           for env, action in zip(self._envs, actions)]
0110     else:
0111       transitions = [
0112           env.step(action, blocking=False)
0113           for env, action in zip(self._envs, actions)]
0114       transitions = [transition() for transition in transitions]
0115     outputs = {key: [] for key in self._keys}
0116     for observ, reward, done, _ in transitions:
0117       for key, value in observ.items():
0118         outputs[key].append(np.array(value))
0119       outputs['reward'].append(np.array(reward, np.float32))
0120       outputs['done'].append(np.array(done, np.bool))
0121     outputs = {key: np.stack(value) for key, value in outputs.items()}
0122     if self._flatten:
0123       outputs = tuple(outputs[key] for key in self._keys)
0124     return outputs
0125 
0126   def reset(self, indices=None):
0127     if indices is None:
0128       indices = range(len(self._envs))
0129     if self._blocking:
0130       observs = [self._envs[index].reset() for index in indices]
0131     else:
0132       observs = [self._envs[index].reset(blocking=False) for index in indices]
0133       observs = [observ() for observ in observs]
0134     outputs = {key: [] for key in self._keys}
0135     for observ in observs:
0136       for key, value in observ.items():
0137         outputs[key].append(np.array(value))
0138       outputs['reward'].append(np.array(0.0, np.float32))
0139       outputs['done'].append(np.array(False, np.bool))
0140     outputs = {key: np.stack(value) for key, value in outputs.items()}
0141     if self._flatten:
0142       outputs = tuple(outputs[key] for key in self._keys)
0143     return outputs
0144 
0145   def close(self):
0146     for env in self._envs:
0147       if hasattr(env, 'close'):
0148         env.close()
