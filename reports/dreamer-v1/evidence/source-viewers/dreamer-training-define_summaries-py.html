0027 def define_summaries(graph, config, cleanups):
0028   summaries = []
0029   # plot_summaries = []  # Control dependencies for non thread-safe matplot.
0030   heads = graph.heads.copy(_unlocked=True)
0031   last_time = tf.Variable(lambda: tf.timestamp(), trainable=False)
0032   last_step = tf.Variable(lambda: 0.0, trainable=False, dtype=tf.float64)
0033 
0034   def transform(dist):
0035     mean = config.postprocess_fn(dist.mean())
0036     mean = tf.clip_by_value(mean, 0.0, 1.0)
0037     return tfd.Independent(tfd.Normal(mean, 1.0), len(dist.event_shape))
0038   heads['image'] = lambda features: transform(graph.heads['image'](features))
0039   heads.pop('cpc', None)  # Not applied to RNN states.
0040 
0041   with tf.variable_scope('general'):
0042     summaries += summary.data_summaries(graph.data, config.postprocess_fn)
0043     summaries += summary.dataset_summaries(config.train_dir, 'train_dataset')
0044     summaries += summary.dataset_summaries(config.test_dir, 'test_dataset')
0045     summaries += summary.objective_summaries(graph.objectives)
0046     summaries.append(tf.summary.scalar('step', graph.step))
0047     new_time, new_step = tf.timestamp(), tf.cast(graph.global_step, tf.float64)
0048     delta_time, delta_step = new_time - last_time, new_step - last_step
0049     with tf.control_dependencies([delta_time, delta_step]):
0050       assign_ops = [last_time.assign(new_time), last_step.assign(new_step)]
0051       with tf.control_dependencies(assign_ops):
0052         summaries.append(tf.summary.scalar(
0053             'steps_per_second', delta_step / delta_time))
0054         summaries.append(tf.summary.scalar(
0055             'seconds_per_step', delta_time / delta_step))
0056 
0057   with tf.variable_scope('closedloop'):
0058     prior, posterior = tools.unroll.closed_loop(
0059         graph.cell, graph.embedded, graph.data['action'], config.debug)
0060     summaries += summary.state_summaries(graph.cell, prior, posterior)
0061     # with tf.variable_scope('prior'):
0062     #   prior_features = graph.cell.features_from_state(prior)
0063     #   prior_dists = {
0064     #       name: head(prior_features)
0065     #       for name, head in heads.items()}
0066     #   summaries += summary.dist_summaries(prior_dists, graph.data)
0067     #   summaries += summary.image_summaries(
0068     #       prior_dists['image'], config.postprocess_fn(graph.data['image']))
0069     # with tf.variable_scope('posterior'):
0070     #   posterior_features = graph.cell.features_from_state(posterior)
0071     #   posterior_dists = {
0072     #       name: head(posterior_features)
0073     #       for name, head in heads.items()}
0074     #   summaries += summary.dist_summaries(
0075     #       posterior_dists, graph.data)
0076     #   summaries += summary.image_summaries(
0077     #       posterior_dists['image'],
0078     #       config.postprocess_fn(graph.data['image']))
0079 
0080   with tf.variable_scope('openloop'):
0081     state = tools.unroll.open_loop(
0082         graph.cell, graph.embedded, graph.data['action'],
0083         config.open_loop_context, config.debug)
0084     state_features = graph.cell.features_from_state(state)
0085     state_dists = {name: head(state_features) for name, head in heads.items()}
0086     if 'cpc' in graph.heads:
0087       state_dists['cpc'] = graph.heads.cpc(graph.embedded)
0088     summaries += summary.dist_summaries(state_dists, graph.data)
0089     summaries += summary.image_summaries(
0090         state_dists['image'], config.postprocess_fn(graph.data['image']))
0091     summaries += summary.state_summaries(graph.cell, state, posterior)
0092     # with tf.control_dependencies(plot_summaries):
0093     #   plot_summary = summary.prediction_summaries(
0094     #       state_dists, graph.data, state)
0095     #   plot_summaries += plot_summary
0096     #   summaries += plot_summary
0097 
0098   with tf.variable_scope('simulation'):
0099     sim_returns = []
0100     for name, params in config.test_collects.items():
0101       # These are expensive and equivalent for train and test phases, so only
0102       # do one of them.
0103       sim_summary, sim_return = tf.cond(
0104           tf.equal(graph.phase, 'test'),
0105           lambda: utility.simulate(
0106               graph.metrics, config, params, graph, cleanups,
0107               gif_summary=True, name=name),
0108           lambda: ('', 0.0),
0109           name='should_simulate_' + params.task.name)
0110       summaries.append(sim_summary)
0111       sim_returns.append(sim_return)
0112 
0113   summaries = tf.summary.merge(summaries)
0114   score = tf.reduce_mean(sim_returns)[None]
0115   return summaries, score
