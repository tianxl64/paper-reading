0046 class Experiment(object):

0048   def __init__(
0049       self, basedir, process_fn, start_fn=None, resume_fn=None,
0050       num_runs=None, worker_name=None, ping_every=30, resume_runs=True):
0051     self._basedir = basedir
0052     self._process_fn = process_fn
0053     self._start_fn = start_fn
0054     self._resume_fn = resume_fn
0055     self._num_runs = num_runs
0056     self._worker_name = worker_name or str(uuid.uuid4())
0057     self._ping_every = ping_every
0058     self._ping_stale = ping_every and 2 * ping_every
0059     self._resume_runs = resume_runs
0060 
0061   def __iter__(self):
0062     for current_run in self._generate_run_numbers():
0063       logdir = self._basedir and os.path.join(
0064           self._basedir, '{:05}'.format(current_run))
0065       try:
0066         run = Run(
0067             logdir, self._process_fn, self._start_fn, self._resume_fn,
0068             self._worker_name, self._ping_every, self._ping_stale,
0069             self._resume_runs)
0070         yield run
0071       except SkipRun:
0072         continue
0073       except StopExperiment:
0074         print('Stopping.')
0075         break
0076     print('All runs completed.')
0077 
0078   def _generate_run_numbers(self):
0079     if self._num_runs:
0080       # Don't wait initially and see if there are runs that are already stale.
0081       runs = np.random.permutation(range(self._num_runs))
0082       for run in runs:
0083         yield run + 1
0084       # At the end, wait for all dead runs to become stale, and pick them up.
0085       # This is necessary for complete runs of workers that died very recently.
0086       if self._ping_stale:
0087         time.sleep(self._ping_stale)
0088         for run in runs:
0089           yield run + 1
0090     else:
0091       # For infinite runs, we want to always finish started jobs first.
0092       # Therefore, we need to wait for them to become stale in the beginning.
0093       if self._ping_stale:
0094         time.sleep(self._ping_stale)
0095       for run in itertools.count():
0096         yield run + 1

0099 class Run(object):
0100 
0101   def __init__(
0102       self, logdir, process_fn, start_fn, resume_fn, worker_name,
0103       ping_every=30, ping_stale=60, reuse_if_exists=True):
0104     self._logdir = os.path.expanduser(logdir)
0105     self._process_fn = process_fn
0106     self._worker_name = worker_name
0107     self._ping_every = ping_every
0108     self._ping_stale = ping_stale
0109     self._logger = self._create_logger()
0110     try:
0111       if self._should_start():
0112         self._claim()
0113         self._logger.info('Start.')
0114         self._init_fn = start_fn
0115       elif reuse_if_exists and self._should_resume():
0116         self._claim()
0117         self._logger.info('Resume.')
0118         self._init_fn = resume_fn
0119       else:
0120         raise SkipRun
0121     except WorkerConflict:
0122       self._logger.info('Leave to other worker.')
0123       raise SkipRun
0124     self._thread = None
0125     self._running = [True]
0126     self._thread = threading.Thread(target=self._store_ping_thread)
0127     self._thread.daemon = True  # Terminate with main thread.
0128     self._thread.start()
0129 
0130   def __iter__(self):
0131     try:
0132       args = self._init_fn and self._init_fn(self._logdir)
0133       if args is None:
0134         args = ()
0135       if not isinstance(args, tuple):
0136         args = (args,)
0137       for value in self._process_fn(self._logdir, *args):
0138         if not self._running[0]:
0139           break
0140         yield value
0141       self._logger.info('Done.')
0142       self._store_done()
0143     except WorkerConflict:
0144       self._logging.warn('Unexpected takeover.')
0145       raise SkipRun
0146     except Exception as e:
0147       exc_info = sys.exc_info()
0148       self._handle_exception(e)
0149       six.reraise(*exc_info)
0150     finally:
0151       self._running[0] = False
0152       self._thread and self._thread.join()
0153 
0154   def _should_start(self):
0155     if not self._logdir:
0156       return True
0157     if tf.gfile.Exists(os.path.join(self._logdir, 'PING')):
0158       return False
0159     if tf.gfile.Exists(os.path.join(self._logdir, 'DONE')):
0160       return False
0161     return True
0162 
0163   def _should_resume(self):
0164     if not self._logdir:
0165       return False
0166     if tf.gfile.Exists(os.path.join(self._logdir, 'DONE')):
0167       # self._logger.debug('Already done.')
0168       return False
0169     if not tf.gfile.Exists(os.path.join(self._logdir, 'PING')):
0170       # self._logger.debug('Not started yet.')
0171       return False
0172     last_worker, last_ping = self._read_ping()
0173     if last_worker != self._worker_name and last_ping < self._ping_stale:
0174       # self._logger.debug('Already in progress.')
0175       return False
0176     return True
0177 
0178   def _claim(self):
0179     if not self._logdir:
0180       return False
0181     self._store_ping(overwrite=True)
0182     if self._ping_every:
0183       time.sleep(self._ping_every)
0184     if self._read_ping()[0] != self._worker_name:
0185       raise WorkerConflict
0186     self._store_ping()
0187 
0188   def _store_done(self):
0189     if not self._logdir:
0190       return
0191     with tf.gfile.Open(os.path.join(self._logdir, 'DONE'), 'w') as file_:
0192       file_.write('\n')
0193 
0194   def _store_fail(self, message):
0195     if not self._logdir:
0196       return
0197     with tf.gfile.Open(os.path.join(self._logdir, 'FAIL'), 'w') as file_:
0198       file_.write(message + '\n')
0199 
0200   def _read_ping(self):
0201     if not tf.gfile.Exists(os.path.join(self._logdir, 'PING')):
0202       return None, None
0203     try:
0204       with tf.gfile.Open(os.path.join(self._logdir, 'PING'), 'rb') as file_:
0205         last_worker, last_ping = pickle.load(file_)
0206       duration = (datetime.datetime.utcnow() - last_ping).total_seconds()
0207       return last_worker, duration
0208     except (EOFError, IOError, tf.errors.NotFoundError):
0209       raise WorkerConflict
0210 
0211   def _store_ping(self, overwrite=False):
0212     if not self._logdir:
0213       return
0214     try:
0215       last_worker, _ = self._read_ping()
0216       if last_worker is None:
0217         self._logger.info("Create directory '{}'.".format(self._logdir))
0218         tf.gfile.MakeDirs(self._logdir)
0219       elif last_worker != self._worker_name and not overwrite:
0220         raise WorkerConflict
0221       # self._logger.debug('Store ping.')
0222       with tf.gfile.Open(os.path.join(self._logdir, 'PING'), 'wb') as file_:
0223         pickle.dump((self._worker_name, datetime.datetime.utcnow()), file_)
0224     except (EOFError, IOError, tf.errors.NotFoundError):
0225       raise WorkerConflict
0226 
0227   def _store_ping_thread(self):
0228     if not self._ping_every:
0229       return
0230     try:
0231       last_write = time.time()
0232       self._store_ping(self._logdir)
0233       while self._running[0]:
0234         if time.time() >= last_write + self._ping_every:
0235             last_write = time.time()
0236             self._store_ping(self._logdir)
0237         # Only wait short times to quickly react to abort.
0238         time.sleep(0.01)
0239     except WorkerConflict:
0240       self._running[0] = False

0242   def _handle_exception(self, exception):
0243     message = ''.join(traceback.format_exception(*sys.exc_info()))
0244     self._logger.warning('Exception:\n{}'.format(message))
0245     self._logger.warning('Failed.')
0246     try:
0247       self._store_done()
0248       self._store_fail(message)
0249     except Exception:
0250       message = ''.join(traceback.format_exception(*sys.exc_info()))
0251       template = 'Exception in exception handler:\n{}'
0252       self._logger.warning(template.format(message))
0253 
0254   def _create_logger(self):
0255     run_name = self._logdir and os.path.basename(self._logdir)
0256     methods = {}
0257     for name in 'debug info warning'.split():
0258       methods[name] = lambda unused_self, message: print(
0259           'Worker {} run {}: {}'.format(self._worker_name, run_name, message))
0260     return type('PrefixedLogger', (object,), methods)()
