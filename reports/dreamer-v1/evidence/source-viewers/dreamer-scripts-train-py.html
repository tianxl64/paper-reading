0035 def process(logdir, args):
0036   with args.params.unlocked:
0037     args.params.logdir = logdir
0038   config = configs.make_config(args.params)
0039   logdir = pathlib.Path(logdir)
0040   metrics = tools.Metrics(logdir / 'metrics', workers=5)
0041   training.utility.collect_initial_episodes(metrics, config)
0042   tf.reset_default_graph()
0043   dataset = tools.numpy_episodes.numpy_episodes(
0044       config.train_dir, config.test_dir, config.batch_shape,
0045       reader=config.data_reader,
0046       loader=config.data_loader,
0047       num_chunks=config.num_chunks,
0048       preprocess_fn=config.preprocess_fn,
0049       gpu_prefetch=config.gpu_prefetch)
0050   metrics = tools.InGraphMetrics(metrics)
0051   build_graph = tools.bind(training.define_model, logdir, metrics)
0052   for score in training.utility.train(build_graph, dataset, logdir, config):
0053     yield score
