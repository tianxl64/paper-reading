0132 def compute_objectives(posterior, prior, target, graph, config):
0133   raw_features = graph.cell.features_from_state(posterior)
0134   heads = graph.heads
0135   if config.imagination_horizon:
0136     imagination_start = posterior
0137     if config.imagination_skip_last:
0138       imagination_start = tools.nested.map(
0139           lambda x: x[:, :-config.imagination_skip_last], imagination_start)
0140     raw_states = imagine_forward(
0141         imagination_start, config.imagination_horizon, graph, config,
0142         graph.heads.action, stop_grad_post_action=False,
0143         stop_grad_pre_action=config.stop_grad_pre_action)
0144   else:
0145     raw_states = None
0146   objectives = []
0147   for name, scale in sorted(config.loss_scales.items(), key=lambda x: x[0]):
0148     if config.loss_scales[name] == 0.0:
0149       continue
0150     if name in config.heads and name not in config.gradient_heads:
0151       features = tf.stop_gradient(raw_features)
0152       include = r'.*/head_{}/.*'.format(name)
0153       exclude = None
0154     else:
0155       features = raw_features
0156       include = None
0157       exclude = None
0158 
0159     if name == 'divergence':
0160       loss = graph.cell.divergence_from_states(posterior, prior)
0161       if config.free_nats is not None:
0162         loss = tf.maximum(0.0, loss - float(config.free_nats))
0163       objectives.append(Objective('divergence', loss, min, include, exclude))
0164 
0165     elif name == 'cpc':
0166       pred = heads.cpc(graph.embedded)
0167       objective = compute_cpc_loss(pred, features, config)
0168       objectives.append(Objective('cpc', objective, max, include, exclude))
0169 
0170     elif name == 'overshooting':
0171       shape = tools.shape(graph.data['action'])
0172       length = tf.tile(tf.constant(shape[1])[None], [shape[0]])
0173       _, priors, posteriors, mask = tools.overshooting(
0174           graph.cell, {}, graph.embedded, graph.data['action'], length,
0175           config.overshooting_distance, posterior)
0176       posteriors, priors, mask = tools.nested.map(
0177           lambda x: x[:, :, 1:-1], (posteriors, priors, mask))
0178       if config.os_stop_posterior_grad:
0179         posteriors = tools.nested.map(tf.stop_gradient, posteriors)
0180       loss = graph.cell.divergence_from_states(posteriors, priors)
0181       if config.free_nats is not None:
0182         loss = tf.maximum(0.0, loss - float(config.free_nats))
0183       objectives.append(Objective('overshooting', loss, min, include, exclude))
0184 
0185     elif name == 'value':
0186       if config.value_source == 'dataset':
0187         loss = compute_value_loss(
0188             config, graph, priors, features, target['reward'])
0189       elif config.value_source == 'model':
0190         if 'action_target' in graph.heads or not config.imagination_horizon:
0191           if 'action_target' in graph.heads:
0192             policy = graph.heads.action_target
0193           else:
0194             policy = graph.heads.action
0195           states = imagine_forward(
0196               posterior, config.value_model_horizon, graph, config, policy)
0197         else:
0198           states = raw_states
0199         feat = graph.cell.features_from_state(states)
0200         loss = compute_value_loss(config, graph, states, feat, None)
0201       else:
0202         raise NotImplementedError(config.value_source)
0203       objectives.append(Objective('value', loss, min, include, exclude))
0204 
0205     elif name == 'action':
0206       if config.action_source == 'model':
0207         if not config.imagination_horizon:
0208           states = imagine_forward(
0209               posterior, config.action_model_horizon, graph, config,
0210               policy=graph.heads.action, stop_grad_post_action=False)
0211         else:
0212           states = raw_states
0213         feat = graph.cell.features_from_state(states)
0214         objective = compute_action_values(config, graph, states, feat)
0215         objectives.append(Objective(
0216             'action', objective, max, include, exclude))
0217       elif config.action_source == 'dataset':
0218         objective = heads.action(features).log_prob(target[name])
0219         objective -= compute_action_divergence(features, graph, config)
0220         objectives.append(Objective(
0221             'action', objective, max, include, exclude))
0222       else:
0223         raise NotImplementedError(config.action_source)
0224 
0225     elif name == 'reward':
0226       reward_mask = tf.squeeze(target['reward_mask'], [-1])
0227       logprob = heads.reward(features).log_prob(target[name]) * reward_mask
0228       objectives.append(Objective('reward', logprob, max, include, exclude))
0229 
0230     elif name == 'pcont' and config.pcont_label_weight:
0231       terminal = tf.cast(tf.less(target[name], 0.5), tf.float32)
0232       logprob = heads[name](features).log_prob(target[name])
0233       logprob *= 1 + terminal * (config.pcont_label_weight - 1)
0234       objectives.append(Objective(name, logprob, max, include, exclude))
0235 
0236     else:
0237       logprob = heads[name](features).log_prob(target[name])
0238       objectives.append(Objective(name, logprob, max, include, exclude))
0239 
0240   objectives = [o._replace(value=tf.reduce_mean(o.value)) for o in objectives]
0241   return objectives

0244 def imagine_forward(
0245     initial_state, distance, graph, config, policy,
0246     stop_grad_post_action=True, stop_grad_pre_action=True):
0247   extended_batch = np.prod(tools.shape(
0248       tools.nested.flatten(initial_state)[0])[:2])
0249   obs = tf.zeros([extended_batch] + list(graph.embedded.shape[2:]))
0250   use_obs = tf.zeros([extended_batch, 1], tf.bool)
0251   new_shape = lambda t: [
0252       tf.reduce_prod(tools.shape(t)[:2])] + tools.shape(t)[2:]
0253   initial_state = tools.nested.map(
0254       lambda tensor: tf.reshape(tensor, new_shape(tensor)),
0255       initial_state)
0256   def step_fn(prev, index):
0257     feature = graph.cell.features_from_state(prev)
0258     if stop_grad_pre_action:
0259       feature = tf.stop_gradient(feature)
0260     action = policy(feature).sample()
0261     if stop_grad_post_action:
0262       action = tf.stop_gradient(action)
0263     (_, state), _ = graph.cell((obs, action, use_obs), prev)
0264     return state
0265   states = tf.scan(step_fn, tf.range(distance), initial_state, back_prop=True)
0266   states = tools.nested.map(lambda x: tf.transpose(x, [1, 0, 2]), states)
0267   return states

0270 def compute_value_loss(config, graph, states, features, reward):
0271   if reward is None:
0272     reward = graph.heads.reward(features).mode()
0273   if config.value_maxent:
0274     reward -= compute_action_divergence(features, graph, config)
0275     reward -= compute_state_divergence(states, graph, config)
0276   if config.value_loss_pcont:
0277     pcont = tf.stop_gradient(graph.heads.pcont(features).mean())
0278   else:
0279     pcont = tf.ones_like(reward)
0280   pcont *= config.value_discount
0281   pred = graph.heads.value(features)
0282   if 'value_target' in graph.heads:
0283     value = graph.heads.value_target(features).mode()
0284   else:
0285     value = pred.mode()
0286   bootstrap = None
0287   if config.value_bootstrap:
0288     reward = reward[:, :-1]
0289     value = value[:, :-1]
0290     pcont = pcont[:, :-1]
0291     bootstrap = value[:, -1]
0292   return_ = control.lambda_return(
0293       reward, value, bootstrap, pcont,
0294       config.value_lambda, axis=1, stop_gradient=True)
0295   return_ = tf.concat([return_, tf.zeros_like(return_[:, -1:])], 1)
0296   loss = -pred.log_prob(return_)[:, :-1]
0297   if config.value_pcont_weight:
0298     loss *= tf.stop_gradient(tf.cumprod(tf.concat([
0299         tf.ones_like(pcont[:, :1]), pcont[:, :-1]], 1), 1))
0300   return loss

0303 def compute_action_values(config, graph, states, features):
0304   reward = graph.heads.reward(features).mode()
0305   reward -= compute_action_divergence(features, graph, config)
0306   reward -= compute_state_divergence(states, graph, config)
0307   if config.action_loss_pcont:
0308     pcont = graph.heads.pcont(features).mean()
0309     if config.action_pcont_stop_grad:
0310       pcont = tf.stop_gradient(pcont)
0311   else:
0312     pcont = tf.ones_like(reward)
0313   pcont *= config.action_discount
0314   if 'value' not in graph.heads:
0315     return control.discounted_return(
0316         reward, pcont, bootstrap=None, axis=1, stop_gradient=False)
0317   value = graph.heads.value(features).mode()
0318   bootstrap = None
0319   if config.action_bootstrap:
0320     reward = reward[:, :-1]
0321     value = value[:, :-1]
0322     pcont = pcont[:, :-1]
0323     bootstrap = value[:, -1]
0324   return_ = control.lambda_return(
0325       reward, value, bootstrap, pcont,
0326       config.action_lambda,
0327       axis=1, stop_gradient=False)
0328   if config.action_pcont_weight:
0329     return_ *= tf.stop_gradient(tf.cumprod(tf.concat([
0330         tf.ones_like(pcont[:, :1]), pcont[:, :-1]], 1), 1))
0331   return return_

0334 def compute_action_divergence(features, graph, config):
0335   features = tf.stop_gradient(features)
0336   pred = graph.heads.action(features)
0337   if not config.action_beta:
0338     return 0.0
0339   try:
0340     amount = -pred.entropy()
0341   except NotImplementedError:
0342     samples = pred.sample(100)
0343     amount = tf.reduce_mean(pred.log_prob(samples), 0)
0344   amount *= config.action_beta
0345   value = config.action_beta_dims_value
0346   if value and value < 0:
0347     amount /= value * float(pred.event_shape[-1].value)
0348   if value and value > 0:
0349     amount *= value * float(pred.event_shape[-1].value)
0350   return amount

0366 def compute_cpc_loss(pred, features, config):
0367   if config.cpc_contrast == 'batch':
0368     ta = tf.TensorArray(tf.float32, 0, True, element_shape=[None, None])
0369     _, _, ta = tf.while_loop(
0370         lambda i, f, ta: tf.less(i, tf.shape(f)[0]),
0371         lambda i, f, ta: (
0372             i + 1, f, ta.write(ta.size(), pred.log_prob(tf.roll(f, i, 0)))),
0373         (0, features, ta), back_prop=True, swap_memory=True)
0374     positive = pred.log_prob(features)
0375     negative = tf.reduce_logsumexp(ta.stack(), 0)
0376     return positive - negative
0377   elif config.cpc_contrast == 'time':
0378     ta = tf.TensorArray(tf.float32, 0, True, element_shape=[None, None])
0379     _, _, ta = tf.while_loop(
0380         lambda i, f, ta: tf.less(i, tf.shape(f)[1]),
0381         lambda i, f, ta: (
0382             i + 1, f, ta.write(ta.size(), pred.log_prob(tf.roll(f, i, 1)))),
0383         (0, features, ta), back_prop=True, swap_memory=True)
0384     positive = pred.log_prob(features)
0385     negative = tf.reduce_logsumexp(ta.stack(), 0)
0386     return positive - negative
0387   elif config.cpc_contrast == 'window':
0388     assert config.cpc_batch_amount <= config.batch_shape[0]
0389     assert config.cpc_time_amount <= config.batch_shape[1]
0390     total_amount = config.cpc_batch_amount * config.cpc_time_amount
0391     ta = tf.TensorArray(tf.float32, 0, True, element_shape=[None, None])
0392     def compute_negatives(index, ta):
0393       batch_shift = tf.math.floordiv(index, config.cpc_time_amount)
0394       time_shift = tf.mod(index, config.cpc_time_amount)
0395       batch_shift -= config.cpc_batch_amount // 2
0396       time_shift -= config.cpc_time_amount // 2
0397       rolled = tf.roll(tf.roll(features, batch_shift, 0), time_shift, 1)
0398       return ta.write(ta.size(), pred.log_prob(rolled))
0399     _, ta = tf.while_loop(
0400         lambda index, ta: tf.less(index, total_amount),
0401         lambda index, ta: (index + 1, compute_negatives(index, ta)),
0402         (0, ta), back_prop=True, swap_memory=True)
0403     positive = pred.log_prob(features)
0404     negative = tf.reduce_logsumexp(ta.stack(), 0)
0405     return positive - negative
0406   else:
0407     raise NotImplementedError(config.cpc_contrast)

0410 def apply_optimizers(objectives, trainer, config):
0411   # Make sure all losses are computed and apply loss scales.
0412   processed = []
0413   values = [ob.value for ob in objectives]
0414   for ob in objectives:
0415     loss = {min: ob.value, max: -ob.value}[ob.goal]
0416     loss *= config.loss_scales[ob.name]
0417     with tf.control_dependencies(values):
0418       loss = tf.identity(loss)
0419     processed.append(ob._replace(value=loss, goal=min))
0420   # Merge objectives that operate on the whole model to compute only one
0421   # backward pass and to share optimizer statistics.
0422   objectives = []
0423   losses = []
0424   for ob in processed:
0425     if ob.include is None and ob.exclude is None:
0426       assert ob.goal == min
0427       losses.append(ob.value)
0428     else:
0429       objectives.append(ob)
0430   loss = tf.reduce_sum(losses)
0431   # modules = ['encoder', 'rnn']
0432   # modules += ['head_{}'.format(name) for name in config.gradient_heads]
0433   # include = r'graph/({})/.*'.format('|'.join(modules))
0434   include = r'.*'
0435   objectives.append(Objective('model', loss, min, include, None))
0436   # Apply optimizers and collect loss summaries.
0437   summaries = []
0438   grad_norms = {}
0439   for ob in objectives:
0440     assert ob.name in list(config.loss_scales.keys()) + ['model'], ob
0441     assert ob.goal == min, ob
0442     if ob.name not in config.optimizers:
0443       with config.optimizers.unlocked:
0444         config.optimizers[ob.name] = config.optimizers.default.copy()
0445     optimizer = config.optimizers[ob.name](
0446         include=ob.include,
0447         exclude=ob.exclude,
0448         step=trainer.step,
0449         log=trainer.log,
0450         debug=config.debug,
0451         name=ob.name)
0452     condition = tf.equal(trainer.phase, 'train')
0453     summary, grad_norm = optimizer.maybe_minimize(condition, ob.value)
0454     summaries.append(summary)
0455     grad_norms[ob.name] = grad_norm
0456   return summaries, grad_norms

0460   def env_ctor():

0467   agent_config = tools.AttrDict(
0468       cell=cell,
0469       encoder=graph.encoder,
0470       planner=functools.partial(params.planner, graph=graph),
0471       objective=bind_or_none(params.objective, graph=graph),
0472       exploration=params.exploration,

0474       postprocess_fn=config.postprocess_fn)

0476   with params.unlocked:
0477     params.update(agent_config)

0481     summaries = []

0483         env_ctor, params.num_envs, config.isolate_envs)

0492           'gif', data['image'], max_outputs=1, fps=20))

0498     with tf.control_dependencies(write_metrics):

0504 def print_metrics(metrics, step, every, decimals=2, name='metrics'):
0505   factor = 10 ** decimals
0506   means, updates = [], []
0507   for key, value in metrics.items():
0508     key = 'metrics_{}_{}'.format(name, key)
0509     mean = tools.StreamingMean((), tf.float32, key)
0510     means.append(mean)
0511     updates.append(mean.submit(value))
0512   with tf.control_dependencies(updates):
0513     message = '{}: step/{} ='.format(name, '/'.join(metrics.keys()))
0514     print_metrics = tf.cond(
0515         tf.equal(step % every, 0),
0516         lambda: tf.print(message, [step] + [
0517             tf.round(mean.clear() * factor) / factor for mean in means]),
0518         tf.no_op)
0519   return print_metrics
