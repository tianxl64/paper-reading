0102 def _worker_dying(basedir, die_at_step, worker_name):
0103   experiment = running.Experiment(
0104       basedir, _process_fn, _start_fn, _resume_fn,
0105       num_runs=100, worker_name=worker_name, ping_every=1.0)
0106   step = 0
0107   for run in experiment:
0108     for score in run:
0109       step += 1
0110       if step >= die_at_step:
0111         return

0114 def _start_fn(logdir):
0115   assert not tf.gfile.Exists(os.path.join(logdir, 'DONE'))
0116   assert not tf.gfile.Exists(os.path.join(logdir, 'started'))
0117   assert not tf.gfile.Exists(os.path.join(logdir, 'resumed'))
0118   with tf.gfile.GFile(os.path.join(logdir, 'started'), 'w') as file_:
0119     file_.write('\n')
0120   with tf.gfile.GFile(os.path.join(logdir, 'numbers'), 'wb') as file_:
0121     pickle.dump([], file_)
0122   return []
0123 
0124 
0125 def _resume_fn(logdir):
0126   assert not tf.gfile.Exists(os.path.join(logdir, 'DONE'))
0127   assert tf.gfile.Exists(os.path.join(logdir, 'started'))
0128   with tf.gfile.GFile(os.path.join(logdir, 'resumed'), 'w') as file_:
0129     file_.write('\n')
0130   with tf.gfile.GFile(os.path.join(logdir, 'numbers'), 'rb') as file_:
0131     numbers = pickle.load(file_)
0132   if len(numbers) != 5:
0133     raise Exception('Expected to be resumed in the middle for this test.')
0134   return numbers

0137 def _process_fn(logdir, numbers):
0138   assert tf.gfile.Exists(os.path.join(logdir, 'started'))
0139   while len(numbers) < 10:
0140     number = np.random.uniform(0, 0.1)
0141     time.sleep(number)
0142     numbers.append(number)
0143     with tf.gfile.GFile(os.path.join(logdir, 'numbers'), 'wb') as file_:
0144       pickle.dump(numbers, file_)
0145     yield number
