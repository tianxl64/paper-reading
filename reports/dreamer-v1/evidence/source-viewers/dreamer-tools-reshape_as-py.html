0015 from __future__ import absolute_import
0016 from __future__ import division
0017 from __future__ import print_function
0018 
0019 import tensorflow as tf
0020 
0021 from dreamer.tools import nested
0022 
0023 
0024 def reshape_as(tensor, reference):
0025   if isinstance(tensor, (list, tuple, dict)):
0026     return nested.map(tensor, lambda x: reshape_as(x, reference))
0027   tensor = tf.convert_to_tensor(tensor)
0028   reference = tf.convert_to_tensor(reference)
0029   statics = reference.shape.as_list()
0030   dynamics = tf.shape(reference)
0031   shape = [
0032       static if static is not None else dynamics[index]
0033       for index, static in enumerate(statics)]
0034   return tf.reshape(tensor, shape)
