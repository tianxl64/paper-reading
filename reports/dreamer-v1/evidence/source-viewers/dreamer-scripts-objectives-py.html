0015 from __future__ import absolute_import

0019 import tensorflow as tf
0020 
0021 from dreamer import tools
0022 from dreamer.control import temporal_difference as td
0023 
0024 
0025 def reward(state, graph, params):
0026   features = graph.cell.features_from_state(state)
0027   reward = graph.heads.reward(features).mean()
0028   return tf.reduce_sum(reward, 1)
0029 
0030 
0031 def reward_value(state, graph, params):
0032   features = graph.cell.features_from_state(state)
0033   reward = graph.heads.reward(features).mean()
0034   value = graph.heads.value(features).mean()
0035   value *= tools.schedule.linear(
0036       graph.step, params.get('objective_value_ramp', 0))
0037   return_ = td.lambda_return(
0038       reward[:, :-1], value[:, :-1], value[:, -1],
0039       params.get('planner_discount', 0.99),
0040       params.get('planner_lambda', 0.95),
0041       axis=1)
0042   return return_[:, 0]
